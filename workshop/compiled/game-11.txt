rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        Set Global Variable(learner, Host Player);
        Teleport(Global.learner, Vector(0, 0, -10));
        Wait(0.5, Ignore Condition);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,11,1,0"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(0, 0, -10));
        Wait(0.2, Ignore Condition);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,11,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,11,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        Wait Until(Or(Compare(Global.qAnswer, >=, 0), Global.gameOver), 90);
        Set Global Variable(qActive, False);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,11,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(5, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
    }
}

rule ("Game 11: the best plan, by value iteration over the beliefs a ping can reach") {
    event {
        Subroutine;
        solvePlan;
    }
    actions {
        "Every rung starts at \"enter now\". Then, sweep after sweep, each rung takes the better of entering now and\npaying for one more ping and continuing with the best plan from wherever the reading lands."
        For Global Variable(gi, 0, 21, 1);
            Set Global Variable At Index(Vt, Global.gi, Max(Subtract(Multiply(10, Subtract(1, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))))), Multiply(30, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))))), Subtract(Multiply(10, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10)))))), Multiply(30, Subtract(1, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))))))));
        End;
        For Global Variable(gsweep, 0, 40, 1);
            For Global Variable(gi, 1, 20, 1);
                Set Global Variable(gp, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))));
                Set Global Variable(pl, Add(Multiply(Global.gp, 0.8), Multiply(Subtract(1, Global.gp), 0.199999999999999)));
                Set Global Variable At Index(Vt, Global.gi, Max(Max(Subtract(Multiply(10, Subtract(1, Global.gp)), Multiply(30, Global.gp)), Subtract(Multiply(10, Global.gp), Multiply(30, Subtract(1, Global.gp)))), Add(Add(-1, Multiply(Global.pl, Value In Array(Global.Vt, Add(Global.gi, 1)))), Multiply(Subtract(1, Global.pl), Value In Array(Global.Vt, Subtract(Global.gi, 1))))));
            End;
            Wait(0.016, Ignore Condition);
        End;
        "The plan's threshold: the first rung (counting up from 50/50) where entering is at least as good as pinging."
        Set Global Variable(nStar, -1);
        For Global Variable(gi, 10, 21, 1);
            If(And(Compare(Global.nStar, <, 0), Compare(Max(Subtract(Multiply(10, Subtract(1, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))))), Multiply(30, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))))), Subtract(Multiply(10, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10)))))), Multiply(30, Subtract(1, Divide(1, Add(1, Raise To Power(4, Subtract(0, Subtract(Global.gi, 10))))))))), >=, Subtract(Value In Array(Global.Vt, Global.gi), 0.0001))));
                Set Global Variable(nStar, Subtract(Global.gi, 10));
            End;
        End;
        Set Global Variable(pStar, Divide(1, Add(1, Raise To Power(4, Subtract(0, Global.nStar)))));
    }
}

rule ("Game 11: at the current belief, what entering now and pinging once more are worth") {
    event {
        Subroutine;
        judge;
    }
    actions {
        Set Global Variable(enterNow, Max(Subtract(Multiply(10, Subtract(1, Global.belief)), Multiply(30, Global.belief)), Subtract(Multiply(10, Global.belief), Multiply(30, Subtract(1, Global.belief)))));
        If(Compare(Global.acc, >, 0.99));
            "A perfect ping: after it the belief is 1 or 0, and you enter the empty room."
            Set Global Variable(pingQ, Add(Add(-1, Multiply(Global.belief, 10)), Multiply(Subtract(1, Global.belief), 10)));
        Else;
            Set Global Variable(pl, Add(Multiply(Global.belief, Global.acc), Multiply(Subtract(1, Global.belief), Subtract(1, Global.acc))));
            Set Global Variable(pingQ, Add(Add(-1, Multiply(Global.pl, Value In Array(Global.Vt, Add(Add(Global.nLat, 10), 1)))), Multiply(Subtract(1, Global.pl), Value In Array(Global.Vt, Subtract(Add(Global.nLat, 10), 1)))));
        End;
    }
}

rule ("Game 11: one ping: draw the reading, log it, move the belief by Bayes' rule") {
    event {
        Subroutine;
        doPing;
    }
    actions {
        Call Subroutine(judge);
        If(Or(And(Compare(Global.roundN, ==, 1), Compare(Global.pingCount, <, 2)), And(Compare(Global.roundN, ==, 2), Compare(Global.pingCount, <, 1))));
            Set Global Variable(barShown, False);
        End;
        Set Global Variable(pBefore, Global.belief);
        Set Global Variable(reading, If-Then-Else(Compare(Random Real(0, 1), <, Global.acc), Global.hidden, Subtract(1, Global.hidden)));
        Set Global Variable(belief, If-Then-Else(Compare(Global.reading, ==, 0), Divide(Multiply(Global.belief, Global.acc), Add(Multiply(Global.belief, Global.acc), Multiply(Subtract(1, Global.belief), Subtract(1, Global.acc)))), Divide(Multiply(Global.belief, Subtract(1, Global.acc)), Add(Multiply(Global.belief, Subtract(1, Global.acc)), Multiply(Subtract(1, Global.belief), Global.acc)))));
        Modify Global Variable(nLat, Add, If-Then-Else(Compare(Global.reading, ==, 0), 1, -1));
        Modify Global Variable(pingCount, Add, 1);
        Modify Global Variable(score, Subtract, 1);
        If(Compare(Global.reading, ==, 0));
            Modify Global Variable(nSaidL, Add, 1);
        Else;
            Modify Global Variable(nSaidR, Add, 1);
        End;
        Set Global Variable(lastSaid, If-Then-Else(Compare(Global.reading, ==, 0), Custom String("LEFT"), Custom String("RIGHT")));
        If(Compare(Global.pingCount, ==, 1));
            Set Global Variable(firstReading, Global.reading);
        End;
        "ping: round, ping number, reading (0 LEFT, 1 RIGHT), belief before, belief after, entering-now value, ping value"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,11,ping,{0},{1},{2}", Global.roundN, Global.pingCount, Custom String("{0},{1},{2}", Global.reading, Global.pBefore, Custom String("{0},{1},{2}", Global.belief, Global.enterNow, Global.pingQ))));
        Set Global Variable(barText, Custom String("chance Sombra is in the LEFT room: {0}%", Round To Integer(Multiply(Global.belief, 100), To Nearest)));
        Set Global Variable(hudPings, Custom String("pings this round: {0} said LEFT, {1} said RIGHT. Latest: {2}", Global.nSaidL, Global.nSaidR, Global.lastSaid));
        Set Global Variable(hudRound, Custom String("round {0} of 5   score {1}   the ping is right {2}% of the time", Global.roundN, Global.score, Round To Integer(Multiply(Global.acc, 100), To Nearest)));
        Small Message(All Players(All Teams), Custom String("PING: {0}", Global.lastSaid));
        Create Effect(All Players(All Teams), Sphere, Color(Yellow), Add(If-Then-Else(Compare(Global.reading, ==, 0), Global.padL, Global.padR), Up), 1.4, None);
        Set Global Variable(fx, Last Created Entity);
        Wait(1, Ignore Condition);
        Destroy Effect(Global.fx);
    }
}

rule ("Game 11: one round: hide Sombra, let him ping or enter, reveal, judge his choice") {
    event {
        Subroutine;
        playRound;
    }
    actions {
        Modify Global Variable(roundN, Add, 1);
        Set Global Variable(acc, If-Then-Else(Compare(Global.roundN, ==, 5), 1, 0.8));
        Set Global Variable(hidden, Random Integer(0, 1));
        "hide: round, her room (0 LEFT, 1 RIGHT), the ping's accuracy this round"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,11,hide,{0},{1},{2}", Global.roundN, Global.hidden, Global.acc));
        Set Global Variable(belief, 0.5);
        Set Global Variable(nLat, 0);
        Set Global Variable(pingCount, 0);
        Set Global Variable(nSaidL, 0);
        Set Global Variable(nSaidR, 0);
        Set Global Variable(lastSaid, Custom String("none yet"));
        Set Global Variable(roundDone, False);
        Set Global Variable(roomsOpen, Compare(Global.roundN, >=, 3));
        Set Global Variable(barShown, Compare(Global.roundN, >=, 3));
        Set Global Variable(barText, Custom String("chance Sombra is in the LEFT room: 50%"));
        Set Global Variable(hudPings, Custom String("pings this round: none yet"));
        Set Global Variable(hudRound, Custom String("round {0} of 5   score {1}   the ping is right {2}% of the time", Global.roundN, Global.score, Round To Integer(Multiply(Global.acc, 100), To Nearest)));
        If(Compare(Global.roundN, ==, 1));
            Set Global Variable(coach, Custom String("Round 1. Press F to ping. The rooms open after two pings."));
        Else If(Compare(Global.roundN, ==, 2));
            Set Global Variable(coach, Custom String("Round 2. She has moved. Press F to ping. The rooms open after one ping."));
        Else If(Compare(Global.roundN, ==, 5));
            Set Global Variable(coach, Custom String("Last round. The ping is now always right. Ping, or walk into a room."));
        Else;
            Set Global Variable(coach, Custom String("Round {0}. Your call: ping with F, or walk onto the pad of the room you think is empty.", Global.roundN));
        End;
        "The stand-in's plan for each round: 0 follow the best plan, 1 enter after one ping (the gamble at 80%)."
        Set Global Variable(autoStyle, If-Then-Else(Or(Compare(Global.roundN, ==, 3), Compare(Global.roundN, ==, 5)), 1, 0));
        While(Not(Global.roundDone));
            Set Global Variable(deciding, True);
            Wait Until(Or(Global.pingReq, And(Global.roomsOpen, Or(Compare(Distance Between(Position Of(Global.learner), Global.padL), <, 2), Compare(Distance Between(Position Of(Global.learner), Global.padR), <, 2)))), 900);
            Set Global Variable(deciding, False);
            Stop Throttle In Direction(Global.learner);
            Set Global Variable(onPad, And(Global.roomsOpen, Or(Compare(Distance Between(Position Of(Global.learner), Global.padL), <, 2), Compare(Distance Between(Position Of(Global.learner), Global.padR), <, 2))));
            If(Global.pingReq);
                Set Global Variable(pingReq, False);
                If(Compare(Global.pingCount, >=, 8));
                    Set Global Variable(coach, Custom String("That is the most pings for one round. Walk into a room."));
                    Set Global Variable(roomsOpen, True);
                Else;
                    Call Subroutine(doPing);
                    If(And(Compare(Global.roundN, ==, 1), Compare(Global.pingCount, ==, 1)));
                        Set Global Variable(qId, 1);
                        Set Global Variable(qText, Custom String("The ping said {0}. It is right 8 times in 10. How sure should you be she is there?", Global.lastSaid));
                        Set Global Variable(qOpts, Array(Custom String("50%"), Custom String("80%"), Custom String("94%")));
                        Set Global Variable(qCorrect, 1);
                        Set Global Variable(qAuto, 1);
                        Set Global Variable(qWhy, Custom String("Before it you were 50/50, so the ping's own accuracy is the answer: 80%."));
                        Call Subroutine(ask);
                        Set Global Variable(barShown, True);
                        Set Global Variable(coach, Custom String("The bar is your belief: one number, moved by one rule after each ping. Press F again."));
                    Else If(And(Compare(Global.roundN, ==, 1), Compare(Global.pingCount, ==, 2)));
                        Set Global Variable(qId, 2);
                        Set Global Variable(qText, Custom String("The second ping said {0}. How sure should you be she is in the {0} room now?", Global.lastSaid));
                        Set Global Variable(qOpts, Array(Custom String("50%"), Custom String("80%"), Custom String("94%")));
                        Set Global Variable(qCorrect, If-Then-Else(Compare(Global.reading, ==, Global.firstReading), 2, 0));
                        Set Global Variable(qAuto, 2);
                        Set Global Variable(qWhy, If-Then-Else(Compare(Global.reading, ==, Global.firstReading), Custom String("Two that agree: 0.8 x 0.8 against 0.2 x 0.2, about 94%."), Custom String("One each way cancels out: back to 50%, as if you had not pinged at all.")));
                        Call Subroutine(ask);
                        Set Global Variable(barShown, True);
                        Set Global Variable(roomsOpen, True);
                        Set Global Variable(coach, Custom String("Rooms open. Ping more with F, or walk onto the pad of the room you think is EMPTY."));
                    Else If(And(Compare(Global.roundN, ==, 2), Compare(Global.pingCount, ==, 1)));
                        Set Global Variable(qId, 3);
                        Set Global Variable(qText, Custom String("She is 80% likely in the {0} room. Walk into the other now, or ping again: which pays more?", Global.lastSaid));
                        Set Global Variable(qOpts, Array(Custom String("enter now"), Custom String("ping again")));
                        Call Subroutine(judge);
                        Set Global Variable(qCorrect, If-Then-Else(Compare(Global.pingQ, >, Global.enterNow), 1, 0));
                        Set Global Variable(qAuto, 1);
                        Set Global Variable(qWhy, Custom String("Entering pays 0.8 x 10 - 0.2 x 30 = {0} on average. Pinging on (then the best plan): {1}.", Global.enterNow, Global.pingQ));
                        Call Subroutine(ask);
                        Set Global Variable(barShown, True);
                        Set Global Variable(roomsOpen, True);
                        Set Global Variable(coach, Custom String("At these prices 80% is not sure enough: one wrong room costs three right ones. Your call."));
                    Else;
                        Set Global Variable(coach, Custom String("That ping was worth {0} on average; entering instead, {1}. {2}", Global.pingQ, Global.enterNow, If-Then-Else(Compare(Global.pingQ, >, Add(Global.enterNow, 0.0001)), Custom String("Worth its cost."), Custom String("Entering was the better bet."))));
                    End;
                End;
            Else If(Global.onPad);
                Call Subroutine(judge);
                Set Global Variable(room, If-Then-Else(Compare(Distance Between(Position Of(Global.learner), Global.padL), <, 2), 0, 1));
                Set Global Variable(roomEv, If-Then-Else(Compare(Global.room, ==, 0), Subtract(Multiply(10, Subtract(1, Global.belief)), Multiply(30, Global.belief)), Subtract(Multiply(10, Global.belief), Multiply(30, Subtract(1, Global.belief)))));
                Set Global Variable(reward, If-Then-Else(Compare(Global.room, !=, Global.hidden), 10, -30));
                Modify Global Variable(score, Add, Global.reward);
                If(Compare(Global.room, !=, Global.hidden));
                    Modify Global Variable(nGood, Add, 1);
                End;
                Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,11,enter,{0},{1},{2}", Global.roundN, Global.room, Custom String("{0},{1},{2}", Global.hidden, Global.belief, Custom String("{0},{1},{2}", Global.roomEv, Global.enterNow, Custom String("{0},{1},{2}", Global.pingQ, Global.reward, Global.pingCount)))));
                Create Effect(All Players(All Teams), Sphere, Custom Color(210, 127, 243, 255), Add(If-Then-Else(Compare(Global.hidden, ==, 0), Global.padL, Global.padR), Up), 1.2, None);
                Set Global Variable(sombraFx, Last Created Entity);
                Create In-World Text(All Players(All Teams), Custom String("SOMBRA WAS HERE"), Add(If-Then-Else(Compare(Global.hidden, ==, 0), Global.padL, Global.padR), Vector(0, 2.8, 0)), 2, Do Not Clip, None, Custom Color(210, 127, 243, 255), Default Visibility);
                Set Global Variable(sombraTx, Last Text ID);
                If(Compare(Global.room, ==, Global.hidden));
                    Set Status(Global.learner, Null, Hacked, 2);
                End;
                Set Global Variable(hudRound, Custom String("round {0} of 5   score {1}   the ping is right {2}% of the time", Global.roundN, Global.score, Round To Integer(Multiply(Global.acc, 100), To Nearest)));
                Set Global Variable(coach, Custom String("She was in the {0} room. You walked into the {1}: {2}.", If-Then-Else(Compare(Global.hidden, ==, 0), Custom String("LEFT"), Custom String("RIGHT")), If-Then-Else(Compare(Global.room, ==, 0), Custom String("LEFT"), Custom String("RIGHT")), If-Then-Else(Compare(Global.reward, >, 0), Custom String("+10, empty"), Custom String("-30, hacked"))));
                Set Global Variable(barShown, True);
                Wait(4, Ignore Condition);
                Set Global Variable(coach, Custom String("That room was empty with chance {0}%: worth {1} on average. Pinging on: {2}", Round To Integer(Multiply(100, If-Then-Else(Compare(Global.room, ==, 0), Subtract(1, Global.belief), Global.belief)), To Nearest), Global.roomEv, Custom String("{0}. {1}", Global.pingQ, If-Then-Else(Compare(Global.pingQ, >, Add(Global.roomEv, 0.0001)), Custom String("One more ping was worth it."), Custom String("Entering was the better bet.")))));
                Wait(6, Ignore Condition);
                Destroy Effect(Global.sombraFx);
                Destroy In-World Text(Global.sombraTx);
                Call Subroutine(putHome);
                Set Facing(Global.learner, Forward, To World);
                Set Global Variable(roundDone, True);
            End;
        End;
    }
}
