rule ("Calibration: find one edge (binary search on downward rays)") {
    event {
        Subroutine;
        probeEdge;
    }
    actions {
        Set Global Variable(probeLo, 0);
        Set Global Variable(probeHi, 4);
        While(And(Compare(Absolute Value(Subtract(Y Component Of(Ray Cast Hit Position(Add(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeHi)), Vector(0, 2, 0)), Subtract(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeHi)), Vector(0, 2, 0)), Null, Null, False)), Global.floorY)), <, 0.3), Compare(Global.probeHi, <, 512)));
            Set Global Variable(probeLo, Global.probeHi);
            Modify Global Variable(probeHi, Multiply, 2);
        End;
        For Global Variable(i, 0, 16, 1);
            Set Global Variable(probeMid, Divide(Add(Global.probeLo, Global.probeHi), 2));
            If(Compare(Absolute Value(Subtract(Y Component Of(Ray Cast Hit Position(Add(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeMid)), Vector(0, 2, 0)), Subtract(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeMid)), Vector(0, 2, 0)), Null, Null, False)), Global.floorY)), <, 0.3));
                Set Global Variable(probeLo, Global.probeMid);
            Else;
                Set Global Variable(probeHi, Global.probeMid);
            End;
        End;
        Wait(0.016, Ignore Condition);
    }
}

rule ("Calibration: measure the island") {
    event {
        Ongoing - Global;
    }
    actions {
        "Skirmish has a 30 minute match clock. Freeze it so a long run is not cut off."
        Pause Match Time;
        Wait(1, Ignore Condition);
        Set Global Variable(isleCenter, Ray Cast Hit Position(Vector(0, 60, 0), Vector(0, -60, 0), Null, Null, False));
        If(Compare(Y Component Of(Global.isleCenter), <, -59));
            Set Global Variable(calMsg, Custom String("no floor under the map origin: stand on the island (host)"));
            Wait Until(And(Has Spawned(Host Player), Is On Ground(Host Player)), 99999);
            Wait(0.5, Ignore Condition);
            Set Global Variable(isleCenter, Position Of(Host Player));
        End;
        Set Global Variable(floorY, Y Component Of(Global.isleCenter));
        Set Global Variable(probeDir, Left);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeXMax, Add(X Component Of(Global.isleCenter), Global.probeLo));
        Set Global Variable(probeDir, Right);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeXMin, Subtract(X Component Of(Global.isleCenter), Global.probeLo));
        Set Global Variable(probeDir, Forward);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeZMax, Add(Z Component Of(Global.isleCenter), Global.probeLo));
        Set Global Variable(probeDir, Backward);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeZMin, Subtract(Z Component Of(Global.isleCenter), Global.probeLo));
        If(Or(Or(Compare(Subtract(Global.edgeXMax, Global.edgeXMin), <, 22), Compare(Subtract(Global.edgeZMax, Global.edgeZMin), <, 14)), Compare(Subtract(Global.edgeXMax, Global.edgeXMin), >, 500)));
            Set Global Variable(calStage, -1);
            Set Global Variable(calMsg, Custom String("island measurement looks wrong (see calibration HUD). Is the map Workshop Island?"));
            Log To Inspector(Custom String("RLLAB,FAIL,island,{0},{1},{2}", Global.edgeXMin, Global.edgeXMax, Custom String("{0},{1},{2}", Global.edgeZMin, Global.edgeZMax, Global.floorY)));
            Abort;
        End;
        Call Subroutine(layoutArena);
        Set Global Variable(calStage, 1);
        Set Global Variable(calMsg, Custom String("spawning bots"));
    }
}

rule ("Arena: place the grid from the measured edges") {
    event {
        Subroutine;
        layoutArena;
    }
    actions {
        Set Global Variable(cell, Min(Global.cfgCell, Divide(Subtract(Subtract(Global.edgeXMax, Global.edgeXMin), 4), 12)));
        Set Global Variable(gridOrigin, Vector(Subtract(Divide(Add(Global.edgeXMin, Global.edgeXMax), 2), Divide(Multiply(12, Global.cell), 2)), Global.floorY, Add(Global.edgeZMin, Global.cfgMargin)));
        For Global Variable(i, 0, 48, 1);
            Set Global Variable At Index(cellPos, Global.i, Add(Global.gridOrigin, Vector(Multiply(Add(Modulo(Global.i, 12), 0.5), Global.cell), 0, Multiply(Add(Round To Integer(Divide(Global.i, 12), Down), 0.5), Global.cell))));
        End;
        Set Global Variable(startPos, First Of(Global.cellPos));
        Set Global Variable(goalPos, Value In Array(Global.cellPos, 11));
        Set Global Variable(stripCenter, Add(Global.gridOrigin, Vector(Divide(Multiply(12, Global.cell), 2), 0, Divide(Multiply(4, Global.cell), 2))));
        Set Global Variable(widowPos, Vector(X Component Of(Global.stripCenter), Global.floorY, Min(Add(Add(Z Component Of(Global.gridOrigin), Multiply(4, Global.cell)), Global.cfgWidowStandoff), Subtract(Global.edgeZMax, 2))));
        Set Global Variable(hostPos, Vector(Add(X Component Of(Global.stripCenter), 3), Global.floorY, Min(Add(Z Component Of(Global.widowPos), 3), Subtract(Global.edgeZMax, 1))));
        Set Global Variable(camEye, Add(Global.stripCenter, Vector(0, 21, 15)));
        Set Global Variable(camLook, Add(Global.stripCenter, Vector(0, 0, 1.5)));
    }
}

rule ("Arena: spawn Tracer and Widowmaker, then measure Tracer's run speed") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.calStage == 1;
    }
    actions {
        "(Widowmaker did appear) and then spawned first time in the next seven. The cause was not found; game\nspeed, cell size and a restart after slow motion were each ruled out in the game."
        Set Global Variable(spawnCalls, 0);
        For Global Variable(spawnTry, 0, 5, 1);
            Set Global Variable(tracer, First Of(Filtered Array(Players On Hero(Hero(Tracer), Team 1), Is Dummy Bot(Current Array Element))));
            If(Not(Entity Exists(Global.tracer)));
                Create Dummy Bot(Hero(Tracer), Team 1, -1, Global.startPos, Left);
                Modify Global Variable(spawnCalls, Add, 1);
                Wait(1, Ignore Condition);
            End;
        End;
        For Global Variable(spawnTry, 0, 5, 1);
            Set Global Variable(widow, First Of(Filtered Array(Players On Hero(Hero(Widowmaker), Team 2), Is Dummy Bot(Current Array Element))));
            If(Not(Entity Exists(Global.widow)));
                Create Dummy Bot(Hero(Widowmaker), Team 2, -1, Global.widowPos, Backward);
                Modify Global Variable(spawnCalls, Add, 10);
                Wait(1, Ignore Condition);
            End;
        End;
        Set Global Variable(tracer, First Of(Filtered Array(Players On Hero(Hero(Tracer), Team 1), Is Dummy Bot(Current Array Element))));
        Set Global Variable(widow, First Of(Filtered Array(Players On Hero(Hero(Widowmaker), Team 2), Is Dummy Bot(Current Array Element))));
        If(Not(And(Entity Exists(Global.tracer), Entity Exists(Global.widow))));
            Set Global Variable(calStage, -1);
            Set Global Variable(calMsg, Custom String("the bots did not spawn. Leave a free slot on Team 1 and Team 2, then Esc > Show Lobby > Restart Match"));
            Log To Inspector(Custom String("RLLAB,FAIL,spawn,{0},{1},{2}", Number Of Players(Team 1), Number Of Players(Team 2), Custom String("{0},{1},{2}", Number Of Heroes(Hero(Tracer), Team 1), Number Of Heroes(Hero(Widowmaker), Team 2), Custom String("{0},{1}", Global.startPos, Global.widowPos))));
            Abort;
        End;
        "spawnCalls = Tracer creations + 10 x Widowmaker creations; 11 means each bot appeared on the first try."
        Log To Inspector(Custom String("RLLAB,SPAWN,{0}", Global.spawnCalls));
        If(Compare(Global.cfgSlowMo, <, 100));
            Set Slow Motion(Global.cfgSlowMo);
            Wait(0.25, Ignore Condition);
        End;
        Set Move Speed(Global.tracer, Global.cfgMoveSpeed);
        Set Damage Dealt(Global.widow, If-Then-Else(Compare(Global.cfgExperiment, ==, 1), Global.cfgWidowDamage, 0));
        Start Facing(Global.widow, Direction Towards(Eye Position(Global.widow), Add(Position Of(Global.tracer), Up)), 720, To World, Direction and Turn Rate);
        "Speed run: from the middle row, along +x."
        Set Global Variable(calMsg, Custom String("measuring Tracer's run speed"));
        Start Forcing Player Position(Global.tracer, Value In Array(Global.cellPos, 24), False);
        Wait(0.25, Ignore Condition);
        Stop Forcing Player Position(Global.tracer);
        Set Facing(Global.tracer, Left, To World);
        Wait(0.4, Ignore Condition);
        Start Throttle In Direction(Global.tracer, Left, 1, To World, Replace existing throttle, None);
        Wait(0.4, Ignore Condition);
        Set Global Variable(calP0, Position Of(Global.tracer));
        Set Global Variable(calT0, Total Time Elapsed);
        Wait(0.8, Ignore Condition);
        Set Global Variable(calP1, Position Of(Global.tracer));
        Set Global Variable(runSpeed, Divide(Distance Between(Global.calP0, Global.calP1), Subtract(Total Time Elapsed, Global.calT0)));
        Stop Throttle In Direction(Global.tracer);
        Wait(0.5, Ignore Condition);
        Set Global Variable(coastDist, Distance Between(Global.calP1, Position Of(Global.tracer)));
        If(Compare(Global.runSpeed, <, 1));
            Set Global Variable(calStage, -1);
            Set Global Variable(calMsg, Custom String("Tracer did not move under throttle control"));
            Log To Inspector(Custom String("RLLAB,FAIL,throttle,{0},{1},{2}", Global.runSpeed, Global.calP0, Global.calP1));
            Abort;
        End;
        Set Global Variable(actDuration, Divide(Multiply(Divide(Global.cell, Global.runSpeed), Global.cfgDurationScale), 100));
        Set Global Variable(actTimeout, Add(Divide(Multiply(2.5, Global.cell), Global.runSpeed), 0.25));
        Set Global Variable(calStage, 2);
        Set Global Variable(calMsg, Custom String("ready"));
    }
}

rule ("Arena: put Tracer back on the start cell") {
    event {
        Subroutine;
        resetTracer;
    }
    actions {
        Stop Throttle In Direction(Global.tracer);
        If(Is Dead(Global.tracer));
            Resurrect(Global.tracer);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.tracer));
                Respawn(Global.tracer);
                Wait Until(Is Alive(Global.tracer), 3);
            End;
        End;
        Start Forcing Player Position(Global.tracer, Global.startPos, False);
        Wait(0.15, Ignore Condition);
        Stop Forcing Player Position(Global.tracer);
        Set Facing(Global.tracer, Left, To World);
        Heal(Global.tracer, Null, 9999);
        If(Global.cfgBlink);
            Set Ability Charge(Global.tracer, Button(Ability 1), 3);
        End;
        Wait Until(Is On Ground(Global.tracer), 1);
        Set Global Variable(tracerDied, False);
        Set Global Variable(tracerDiedEnv, False);
    }
}

rule ("Environment: carry out action curA and observe the result") {
    event {
        Subroutine;
        stepEnv;
    }
    actions {
        Set Global Variable(stepCol, Modulo(Global.curCell, 12));
        Set Global Variable(stepRow, Round To Integer(Divide(Global.curCell, 12), Down));
        Set Global Variable(actDir, Value In Array(Array(Left, Right, Forward, Backward, Left), Global.curA));
        Set Global Variable(tgtCol, Add(Global.stepCol, Value In Array(Array(1, -1, 0, 0, 0), Global.curA)));
        Set Global Variable(tgtRow, Add(Global.stepRow, Value In Array(Array(0, 0, 1, -1, 0), Global.curA)));
        Set Global Variable(reward, Global.rStep);
        Set Global Variable(terminal, False);
        Set Global Variable(outcome, 0);
        Set Global Variable(stepHits0, Global.epHits);
        If(Compare(Global.curA, ==, 4));
            Set Facing(Global.tracer, Global.actDir, To World);
            Start Throttle In Direction(Global.tracer, Global.actDir, 1, To World, Replace existing throttle, None);
            Wait(0.064, Ignore Condition);
            Press Button(Global.tracer, Button(Ability 1));
            Wait(0.1, Ignore Condition);
            Stop Throttle In Direction(Global.tracer);
            Wait Until(Or(Or(Compare(Horizontal Speed Of(Global.tracer), <, 0.2), Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2))), Global.tracerDied), 0.4);
        Else If(Or(Or(Compare(Global.tgtCol, <, 0), Compare(Global.tgtCol, >=, 12)), Compare(Global.tgtRow, >=, 4)));
            "Blocked: no movement, the step still counts."
            Wait(0.064, Ignore Condition);
        Else;
            Set Global Variable(tgtPos, Add(Global.gridOrigin, Vector(Multiply(Add(Global.tgtCol, 0.5), Global.cell), 0, Multiply(Add(Global.tgtRow, 0.5), Global.cell))));
            Set Facing(Global.tracer, Global.actDir, To World);
            Start Throttle In Direction(Global.tracer, Global.actDir, 1, To World, Replace existing throttle, None);
            If(Not(Global.cfgClosedLoop));
                "Open loop: run for a fixed time, wherever that leaves her."
                Wait Until(Or(Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2)), Global.tracerDied), Global.actDuration);
            Else If(Compare(Global.tgtRow, <, 0));
                "Closed loop, off the edge: keep running until she is actually falling."
                Wait Until(Or(Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2)), Global.tracerDied), Global.actTimeout);
            Else;
                "Closed loop: cut the throttle so she coasts to a stop on the cell centre."
                Wait Until(Or(Or(Compare(Dot Product(Subtract(Global.tgtPos, Position Of(Global.tracer)), Global.actDir), <=, Global.coastDist), Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2))), Global.tracerDied), Global.actTimeout);
            End;
            Stop Throttle In Direction(Global.tracer);
            If(Compare(Global.tgtRow, <, 0));
                Wait Until(Or(Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2)), Global.tracerDied), 1);
            Else;
                Wait Until(Or(Or(Compare(Horizontal Speed Of(Global.tracer), <, 0.2), Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2))), Global.tracerDied), 0.4);
            End;
        End;
        Call Subroutine(observeState);
        Set Global Variable(nextCell, Global.obsCell);
        Set Global Variable(nextS, Global.obsS);
        If(And(And(Global.tracerDied, Not(Global.tracerDiedEnv)), Compare(Y Component Of(Position Of(Global.tracer)), >=, Subtract(Global.floorY, 1.2))));
            Set Global Variable(terminal, True);
            Set Global Variable(outcome, 4);
            Modify Global Variable(reward, Add, Global.rKilled);
        Else If(Or(Compare(Y Component Of(Position Of(Global.tracer)), <, Subtract(Global.floorY, 1.2)), Global.tracerDied));
            Set Global Variable(terminal, True);
            Set Global Variable(outcome, 2);
            Modify Global Variable(reward, Add, Global.rFall);
        Else If(Compare(Global.nextCell, ==, 11));
            Set Global Variable(terminal, True);
            Set Global Variable(outcome, 1);
            Modify Global Variable(reward, Add, Global.rGoal);
            Modify Global Variable(epEdgeSum, Add, Subtract(Z Component Of(Global.obsP), Global.edgeZMin));
        Else;

rule ("Experiment: one episode") {
    event {
        Subroutine;
        runEpisode;
    }
    actions {
        Set Global Variable(isTraining, And(Global.wantTraining, Compare(Global.cfgExperiment, !=, 2)));
        If(Compare(Global.isTraining, !=, Global.lastTraining));
            Set Global Variable(recentRet, Empty Array);
            Set Global Variable(recentSum, 0);
            Set Global Variable(lastTraining, Global.isTraining);
        End;
        Call Subroutine(resetTracer);
        Set Global Variable(epSteps, 0);
        Set Global Variable(epReturn, 0);
        Set Global Variable(epEdgeSum, 0);
        Set Global Variable(epHits, 0);
        Call Subroutine(observeState);
        Set Global Variable(curS, Global.obsS);
        Set Global Variable(curCell, Global.obsCell);
        Set Global Variable(selS, Global.curS);
        Call Subroutine(chooseAction);
        Set Global Variable(curA, Global.selA);
        Set Global Variable(curExplore, Global.selExplore);
        Set Global Variable(terminal, False);
        Set Global Variable(nextShotT, Add(Total Time Elapsed, Global.cfgWidowInterval));
        Set Global Variable(episodeActive, True);
        While(Not(Global.terminal));
            Call Subroutine(stepEnv);
            Modify Global Variable(epSteps, Add, 1);
            Modify Global Variable(epReturn, Add, Global.reward);
            If(Global.terminal);
                "Terminal step: the target is the reward alone."
                Set Global Variable(tdTarget, Global.reward);
            Else;
                "Choose a' now, with the behaviour policy, for both algorithms."
                Set Global Variable(selS, Global.nextS);
                Call Subroutine(chooseAction);
                Set Global Variable(nextA, Global.selA);
                Set Global Variable(nextExplore, Global.selExplore);
                If(Compare(Global.algo, ==, 0));
                    Set Global Variable(tdTarget, Add(Global.reward, Multiply(Global.gamma, Value In Array(Global.Q, Add(Multiply(Global.nextS, 5), Global.nextA)))));
                Else;
                    Set Global Variable(tdTarget, Add(Global.reward, Multiply(Global.gamma, Last Of(Sorted Array(Array Slice(Global.Q, Multiply(Global.nextS, 5), Global.nAct), Current Array Element)))));
                End;
            End;
            Set Global Variable(qOld, Value In Array(Global.Q, Add(Multiply(Global.curS, 5), Global.curA)));
            If(Global.isTraining);
                Set Global Variable At Index(Q, Add(Multiply(Global.curS, 5), Global.curA), Add(Global.qOld, Multiply(Global.alpha, Subtract(Global.tdTarget, Global.qOld))));
                If(And(Compare(Global.curS, <, 48), Compare(Global.curS, !=, 11)));
                    Set Global Variable(glyphS, Global.curS);
                    Call Subroutine(updateGlyph);
                End;
            End;
            If(Global.cfgLogSteps);
                Call Subroutine(logStep);
            End;
            Set Global Variable(curS, Global.nextS);
            Set Global Variable(curCell, Global.nextCell);
            Set Global Variable(curA, Global.nextA);
            Set Global Variable(curExplore, Global.nextExplore);
            If(And(Global.paused, Not(Global.terminal)));
                Wait Until(Not(Global.paused), 99999);
            End;
        End;
        Set Global Variable(episodeActive, False);
        Stop Throttle In Direction(Global.tracer);
        Modify Global Variable(epCount, Add, 1);
        Modify Global Variable At Index(cnt, Add(Multiply(If-Then-Else(Global.isTraining, 0, 1), 5), Global.outcome), Add, 1);
        Set Global Variable(lastReturn, Global.epReturn);
        Set Global Variable(lastOutcome, Global.outcome);
        Set Global Variable(lastSteps, Global.epSteps);
        Modify Global Variable(recentRet, Append To Array, Global.epReturn);
        Modify Global Variable(recentSum, Add, Global.epReturn);
        If(Compare(Count Of(Global.recentRet), >, 20));
            Modify Global Variable(recentSum, Subtract, First Of(Global.recentRet));
            Modify Global Variable(recentRet, Remove From Array By Index, 0);
        End;
        If(Compare(Count Of(Global.histReturn), <, 990));
            Modify Global Variable(histReturn, Append To Array, Global.epReturn);
            Modify Global Variable(histOutcome, Append To Array, Global.outcome);
        End;
        Call Subroutine(logEpisode);
        If(And(Compare(Global.cfgLogQEvery, >, 0), Compare(Modulo(Global.epCount, Global.cfgLogQEvery), ==, 0)));
            Call Subroutine(logQTable);
        End;
    }
}
