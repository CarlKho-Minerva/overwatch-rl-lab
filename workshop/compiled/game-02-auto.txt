rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        "The automatic check: a bot plays the learner's part. The host stands aside, out of the way and unseen."
        Create Dummy Bot(Hero(Tracer), Team 1, -1, Vector(-8, 0, -4), Left);
        Wait(1, Ignore Condition);
        Set Global Variable(learner, First Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
        Set Status(Host Player, Null, Phased Out, 99999);
        Set Invisible(Host Player, All);
        Teleport(Host Player, Vector(0, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Left, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,2,1,1"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(-8, 0, -4));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Left, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,2,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,2,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        "The moment the question is on screen, so the log shows how long the prediction took (and so a script that plays\nthe human build knows when to press, as a person does by looking)."
        Set Global Variable(qShown, Total Time Elapsed);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,Q,2,{0}", Global.qId));
        Wait(1.2, Ignore Condition);
        Set Global Variable(qAnswer, Global.qAuto);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,2,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1},{2}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0), Subtract(Total Time Elapsed, Global.qShown))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(0.6, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

rule ("Game 2: the bot Tracer walks to the pack worth more under this discount") {
    event {
        Subroutine;
        botWalk;
    }
    actions {
        If(Not(Entity Exists(Global.bot)));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,walked,{0},{1},{2},-1", Global.rnd, Global.g, Global.best));
            Abort;
        End;
        Set Global Variable(walkDir, If-Then-Else(Compare(Global.best, ==, 1), Left, Right));
        Set Global Variable(tgtPos, Add(If-Then-Else(Compare(Global.best, ==, 1), Vector(14, 0, -4), Vector(-14.600000000000001, 0, -4)), Vector(0, 0, 1.8)));
        Teleport(Global.bot, Vector(-8, 0, -2.2));
        Wait(0.3, Ignore Condition);
        Set Facing(Global.bot, Global.walkDir, To World);
        Start Throttle In Direction(Global.bot, Global.walkDir, 1, To World, Replace existing throttle, None);
        Wait Until(Compare(Dot Product(Subtract(Global.tgtPos, Position Of(Global.bot)), Global.walkDir), <=, 0.3), 10);
        Stop Throttle In Direction(Global.bot);
        Wait(0.4, Ignore Condition);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,walked,{0},{1},{2}", Global.rnd, Global.g, Custom String("{0},{1}", Global.best, Compare(Distance Between(Position Of(Global.bot), Global.tgtPos), <, 1.5))));
    }
}

        Set Global Variable(bot, Filtered Array(All Players(Team Of(Global.learner)), And(Is Dummy Bot(Current Array Element), Compare(Current Array Element, !=, Global.learner))));
        Set Global Variable(bot, If-Then-Else(Compare(Count Of(Global.bot), >, 0), First Of(Global.bot), Null));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,bot,{0}", If-Then-Else(Entity Exists(Global.bot), 1, 0)));
        Set Global Variable(coach, Custom String("A mini pack 3 pads behind you, a mega 10 pads ahead. Take the small thing now, or walk for the big one?"));
        Wait(1, Ignore Condition);
        "---------------- Part 1: the dial ----------------"
        For Global Variable(rnd, 0, 4, 1);
            If((Global.learner).fell);
                Call Subroutine(putHome);
            End;
            Set Global Variable(dialOpen, True);
            Set Global Variable(dialLocked, False);
            For Global Variable(dialJ, 0, 4, 1);
                If(Not(Value In Array(Global.tried, Global.dialIdx)));
                    Break;
                End;
                Set Global Variable(dialIdx, Modulo(Add(Global.dialIdx, 1), 4));
            End;
            Call Subroutine(buildDial);
            Set Global Variable(ctrlLine, Custom String("F: turn the dial    R: lock it in    then answer with a click"));
            Set Global Variable(coach, If-Then-Else(Compare(Global.rnd, ==, 0), Custom String("The dial says how much a pack shrinks for every pad she must walk. Set it, then lock it."), Custom String("Next setting: {0} of 4. F to turn, R to lock.", Add(Global.rnd, 1))));
            Wait(0.5, Ignore Condition);
            Set Global Variable(dialIdx, Value In Array(Global.autoOrder, Global.rnd));
            Call Subroutine(buildDial);
            Wait(0.5, Ignore Condition);
            Set Global Variable(dialLocked, True);
            Set Global Variable(dialOpen, False);
            Set Global Variable(g, Value In Array(Global.dialVals, Global.dialIdx));
            Set Global Variable At Index(tried, Global.dialIdx, True);
            Call Subroutine(buildDial);
            Set Global Variable(vMini, Multiply(15, Raise To Power(Global.g, 3)));
            Set Global Variable(vMega, Multiply(50, Raise To Power(Global.g, 10)));
            Set Global Variable(best, If-Then-Else(Compare(Global.vMega, >, Global.vMini), 1, 0));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,dial,{0},{1},{2}", Global.rnd, Global.g, Custom String("{0},{1},{2}", Global.vMini, Global.vMega, Global.best)));
            Set Global Variable(qId, Add(1, Global.rnd));
            Set Global Variable(qText, Custom String("Dial at {0}. Which pack does she walk to?", Global.g));
            Set Global Variable(qOpts, Array(Custom String("MINI (15, 3 pads)"), Custom String("MEGA (50, 10 pads)")));
            Set Global Variable(qCorrect, Global.best);
            Set Global Variable(qAuto, Global.best);
            Set Global Variable(qWhy, Custom String("Mini 15 x {0}^3 = {1}. Mega 50 x {0}^10 = {2}. She takes the larger.", Global.g, Global.vMini, Global.vMega));
            Call Subroutine(ask);
            Set Global Variable(miniLbl, Custom String("MINI worth {0} to her", Global.vMini));
            Set Global Variable(megaLbl, Custom String("MEGA worth {0} to her", Global.vMega));
            Call Subroutine(drawTexts);
            Set Global Variable(coach, Custom String("Discount {0}: she walks to the {1}.", Global.g, If-Then-Else(Compare(Global.best, ==, 1), Custom String("MEGA"), Custom String("MINI"))));
            Call Subroutine(botWalk);
            If(Compare(Global.rnd, ==, 0));
                Set Global Variable(dialName, Custom String("DISCOUNT (gamma)"));
                Call Subroutine(buildDial);
                Set Global Variable(coach, Custom String("That dial is the discount factor, gamma: what one step later is worth compared with now."));
            Else If(Compare(Global.rnd, ==, 1));
                Set Global Variable(coach, Custom String("What she collects, shrunk by gamma once per step of delay and added up, is her return."));
            Else If(Compare(Global.rnd, ==, 2));
                Set Global Variable(coach, Custom String("Same packs, same pads. Only gamma changed, and the best way to act can change with it."));
            Else;
                Set Global Variable(coach, Custom String("The switch sits near gamma 0.84, where 15 x g^3 equals 50 x g^10."));
                Wait(1, Ignore Condition);
                Set Global Variable(coach, Custom String("Why discount: later is less certain, endless sums stay finite, and sooner is usually worth more."));
            End;
            Wait(1, Ignore Condition);
            Set Global Variable(miniLbl, Custom String("MINI 15, 3 pads behind you"));
            Set Global Variable(megaLbl, Custom String("MEGA 50, 10 pads ahead"));
            Call Subroutine(drawTexts);
        End;
        "Interleaving: lecture 1's state and the Markov property."
        Set Global Variable(ctrlLine, Custom String("Answer with a click or melee"));
        Set Global Variable(coach, Custom String("A question from lecture 1: what does her state need to hold?"));
        Set Global Variable(qId, 5);
        Set Global Variable(qText, Custom String("Is her pad enough to predict what comes next, here and in a match?"));
        Set Global Variable(qOpts, Array(Custom String("yes, both"), Custom String("here yes, match no"), Custom String("no, not here")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("Here only her pad changes. In a match a taken pack returns on a timer: add it to the state."));
        Call Subroutine(ask);
        Wait(0.5, Ignore Condition);
        "---------------- Part 2: the one-step rule ----------------"
        If((Global.learner).fell);
            Call Subroutine(putHome);
        End;
        Set Global Variable(dialLine, Custom String("DISCOUNT 0.9, EACH STEP COSTS 1"));
        Set Global Variable(miniLbl, Custom String("MINI 15"));
        Set Global Variable(megaLbl, Custom String("MEGA 50"));
        Set Global Variable(coach, Custom String("New rule: each step costs 1, time out of the fight. Pads show their worth if she always walks to the mega."));
        Set Global Variable At Index(padOn, 13, 1);
        Set Global Variable At Index(padV, 13, Value In Array(Global.vToMega, 13));
        Call Subroutine(drawTexts);
        Wait(1, Ignore Condition);
        Set Global Variable(coach, Custom String("Each pad: minus 1 for the step, plus 0.9 times the pad ahead of it."));
        For Global Variable(loopI, 12, 10, -1);
            Set Global Variable At Index(padOn, Global.loopI, 1);
            Set Global Variable At Index(padV, Global.loopI, Value In Array(Global.vToMega, Global.loopI));
            Call Subroutine(drawTexts);
            Wait(0.3, Ignore Condition);
        End;
        Set Global Variable At Index(padOn, 10, 2);
        Call Subroutine(drawTexts);
        Set Global Variable(qId, 6);
        Set Global Variable(qText, Custom String("This pad = -1 + 0.9 x the pad ahead. What goes on the ? pad?"));
        Set Global Variable(qOpts, Array(Custom String("37.60"), Custom String("33.74"), Custom String("34.74")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("-1 + 0.9 x 38.6 = 33.74. 37.60 forgets the discount; 34.74 forgets the step cost."));
        Call Subroutine(ask);
        For Global Variable(loopI, 10, 0, -1);
            Set Global Variable At Index(padOn, Global.loopI, 1);
            Set Global Variable At Index(padV, Global.loopI, Value In Array(Global.vToMega, Global.loopI));
            Call Subroutine(drawTexts);
            Wait(0.1, Ignore Condition);
        End;
        For Global Variable(loopI, 1, 14, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,pad,0,{0},{1}", Global.loopI, Value In Array(Global.vToMega, Global.loopI)));
        End;
        "He walks the row; the screen checks the rule on the pad he stands on."
        Call Subroutine(putHome);
        Set Facing(Global.learner, Left, To World);
        Set Global Variable(ctrlLine, Custom String("WASD: walk along the row to the mega. Watch the line on the right."));
        Set Global Variable(coach, Custom String("Walk the row to the mega. On every pad, check: this pad = -1 + 0.9 x the pad ahead."));
        Set Global Variable(walkN, 0);
        Set Global Variable(walkOk, 0);
        Set Global Variable(padLast, -1);
        Set Global Variable(walkT0, Total Time Elapsed);
        Wait(0.5, Ignore Condition);
        Start Throttle In Direction(Global.learner, Left, 1, To World, Replace existing throttle, None);
        While(And(Compare(Global.padLast, !=, 13), Compare(Subtract(Total Time Elapsed, Global.walkT0), <, 25)));
            If((Global.learner).fell);
                Call Subroutine(putHome);
                Set Global Variable(padLast, -1);
            End;
            Set Global Variable(padNow, Add(Round To Integer(Divide(Subtract(X Component Of(Position Of(Global.learner)), -8), 2.2), To Nearest), 3));
            If(And(And(And(Compare(Absolute Value(Subtract(Z Component Of(Position Of(Global.learner)), -4)), <, 1.3), Compare(Global.padNow, >=, 1)), Compare(Global.padNow, <=, 13)), Compare(Global.padNow, !=, Global.padLast)));
                Set Global Variable(padLast, Global.padNow);
                Modify Global Variable(walkN, Add, 1);
                If(Compare(Global.padNow, ==, 13));
                    Set Global Variable(checkLine, Custom String("MEGA PAD: worth 50. She picks it up."));
                    Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,walk,{0},{1},0", Global.padNow, Value In Array(Global.vToMega, Global.padNow)));
                Else;
                    Set Global Variable(ruleV, Add(-1, Multiply(0.9, Value In Array(Global.vToMega, Add(Global.padNow, 1)))));
                    If(Compare(Absolute Value(Subtract(Global.ruleV, Value In Array(Global.vToMega, Global.padNow))), <, 0.011));
                        Modify Global Variable(walkOk, Add, 1);
                    End;
                    Set Global Variable(checkLine, Custom String("THIS PAD {0}  =  -1 + 0.9 x {1} (the pad ahead)  =  {2}", Value In Array(Global.vToMega, Global.padNow), Value In Array(Global.vToMega, Add(Global.padNow, 1)), Global.ruleV));
                    Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,walk,{0},{1},{2}", Global.padNow, Value In Array(Global.vToMega, Global.padNow), Value In Array(Global.vToMega, Add(Global.padNow, 1))));
                End;
            End;
            Wait(0.1, Ignore Condition);
        End;
        Stop Throttle In Direction(Global.learner);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,walkdone,{0},{1},{2}", Global.walkN, Global.walkOk, If-Then-Else(Compare(Global.padLast, ==, 13), 1, 0)));
        Set Global Variable(coach, Custom String("That one-step rule is the Bellman equation: value here = reward now + discount x value next."));
        Wait(1, Ignore Condition);
        Set Global Variable(checkLine, Custom String(""));
        "A value belongs to a pad and a way of walking."
        Call Subroutine(putHome);
        Set Global Variable(ctrlLine, Custom String("Answer with a click"));
        Set Global Variable(coach, Custom String("Same start pad, same discount 0.9. Now she always walks to the MINI."));
        Set Global Variable(qId, 7);
        Set Global Variable(qText, Custom String("Walking to the MINI: start pad more or less than {0}?", Value In Array(Global.vToMega, 3)));
        Set Global Variable(qOpts, Array(Custom String("more"), Custom String("less")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("Less: {0}. The same pad has another value under another way of walking.", Value In Array(Global.vToMini, 3)));
        Call Subroutine(ask);
        Set Global Variable At Index(padOn, 13, 0);
        For Global Variable(loopI, 1, 13, 1);
            Set Global Variable At Index(padOn, Global.loopI, 1);
            Set Global Variable At Index(padV, Global.loopI, Value In Array(Global.vToMini, Global.loopI));
        End;
        Call Subroutine(drawTexts);
        For Global Variable(loopI, 0, 13, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,pad,1,{0},{1}", Global.loopI, Value In Array(Global.vToMini, Global.loopI)));
        End;
        Set Global Variable(coach, Custom String("One pad behind the start, the mini is the better walk: {0} against {1}.", Value In Array(Global.vToMini, 2), Value In Array(Global.vToMega, 2)));
        Wait(1, Ignore Condition);
        "Debrief."
        Set Global Variable(ctrlLine, Custom String(""));
        Set Global Variable(debrief1, Custom String("You did: set the discount, called her pack, walked a row checked one step at a time."));
        Set Global Variable(debrief2, Custom String("Called: discount, return (a discounted sum), value under a policy, Bellman equation."));
        Set Global Variable(debrief3, Custom String("In an agent: low gamma answers now, high gamma spends tool calls on a better answer."));
        Set Global Variable(coach, Custom String("Your predictions: {0} of {1} right.", Global.predRight, Global.predTotal));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,2,end,{0},{1}", Global.predRight, Global.predTotal));
        Wait(1, Ignore Condition);
        Set Global Variable(coach, Custom String("Simplified: one straight row and no dice. The lecture averages over chance; here every step is certain."));
        Call Subroutine(finish);
    }
