rule ("Calibration: find one edge (binary search on downward rays)") {
    event {
        Subroutine;
        probeEdge;
    }
    actions {
        Set Global Variable(probeLo, 0);
        Set Global Variable(probeHi, 4);
        While(And(Compare(Absolute Value(Subtract(Y Component Of(Ray Cast Hit Position(Add(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeHi)), Vector(0, 2, 0)), Subtract(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeHi)), Vector(0, 2, 0)), Null, Null, False)), Global.floorY)), <, 0.3), Compare(Global.probeHi, <, 512)));
            Set Global Variable(probeLo, Global.probeHi);
            Modify Global Variable(probeHi, Multiply, 2);
        End;
        For Global Variable(probeI, 0, 16, 1);
            Set Global Variable(probeMid, Divide(Add(Global.probeLo, Global.probeHi), 2));
            If(Compare(Absolute Value(Subtract(Y Component Of(Ray Cast Hit Position(Add(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeMid)), Vector(0, 2, 0)), Subtract(Add(Global.isleCenter, Multiply(Global.probeDir, Global.probeMid)), Vector(0, 2, 0)), Null, Null, False)), Global.floorY)), <, 0.3));
                Set Global Variable(probeLo, Global.probeMid);
            Else;
                Set Global Variable(probeHi, Global.probeMid);
            End;
        End;
        Wait(0.016, Ignore Condition);
    }
}

rule ("Init: measure the island and lay out both strips") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1, Ignore Condition);
        Set Global Variable(isleCenter, Ray Cast Hit Position(Vector(0, 60, 0), Vector(0, -60, 0), Null, Null, False));
        Set Global Variable(floorY, Y Component Of(Global.isleCenter));
        Set Global Variable(probeDir, Left);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeXMax, Add(X Component Of(Global.isleCenter), Global.probeLo));
        Set Global Variable(probeDir, Right);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeXMin, Subtract(X Component Of(Global.isleCenter), Global.probeLo));
        Set Global Variable(probeDir, Forward);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeZMax, Add(Z Component Of(Global.isleCenter), Global.probeLo));
        Set Global Variable(probeDir, Backward);
        Call Subroutine(probeEdge);
        Set Global Variable(edgeZMin, Subtract(Z Component Of(Global.isleCenter), Global.probeLo));
        If(Or(Or(Or(Compare(Y Component Of(Global.isleCenter), <, -59), Compare(Subtract(Global.edgeXMax, Global.edgeXMin), <, 26)), Compare(Subtract(Global.edgeZMax, Global.edgeZMin), <, 22)), Compare(Subtract(Global.edgeXMax, Global.edgeXMin), >, 500)));
            Set Global Variable(calStage, -1);
            Set Global Variable(calMsg, Custom String("island measurement looks wrong. Is the map Workshop Island?"));
            Enable Inspector Recording;
            Log To Inspector(Custom String("NS,FAIL,island,{0},{1},{2}", Global.edgeXMin, Global.edgeXMax, Custom String("{0},{1},{2}", Global.edgeZMin, Global.edgeZMax, Global.floorY)));
            Disable Inspector Recording;
            Abort;
        End;
        Set Global Variable(cell, Min(Global.cfgCell, Divide(Subtract(Subtract(Global.edgeXMax, Global.edgeXMin), 4), 12)));
        Set Global Variable(gridX0, Subtract(Divide(Add(Global.edgeXMin, Global.edgeXMax), 2), Divide(Multiply(12, Global.cell), 2)));
        Set Global Variable(zBase, Array(Add(Global.edgeZMin, 0.5), Subtract(Global.edgeZMax, 0.5)));
        For Global Variable(probeI, 0, 96, 1);
            Set Global Variable At Index(cellPos, Global.probeI, Vector(Add(Global.gridX0, Multiply(Add(Modulo(Global.probeI, 12), 0.5), Global.cell)), Global.floorY, Add(Value In Array(Global.zBase, Round To Integer(Divide(Global.probeI, 48), Down)), Multiply(Multiply(Value In Array(Global.zSign, Round To Integer(Divide(Global.probeI, 48), Down)), Add(Round To Integer(Divide(Modulo(Global.probeI, 48), 12), Down), 0.5)), Global.cell))));
        End;
        Set Global Variable(startPos, Array(First Of(Global.cellPos), Value In Array(Global.cellPos, 48)));
        Set Global Variable(goalPos, Array(Value In Array(Global.cellPos, 11), Value In Array(Global.cellPos, 59)));
        Set Global Variable(camEye, Vector(Divide(Add(Global.edgeXMin, Global.edgeXMax), 2), Add(Global.floorY, 38), Add(Divide(Add(Global.edgeZMin, Global.edgeZMax), 2), 16)));
        Set Global Variable(camLook, Vector(Divide(Add(Global.edgeXMin, Global.edgeXMax), 2), Global.floorY, Add(Divide(Add(Global.edgeZMin, Global.edgeZMax), 2), 1)));
        Set Global Variable(calStage, 1);
        Set Global Variable(calMsg, Custom String("spawning bots"));
    }
}

rule ("Init: spawn the bots, measure their run speed, start the block") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.calStage == 1;
    }
    actions {
        "Up to five tries: on 2026-11-21 a bot failed to appear in two matches in a row, cause not found."
        For Global Variable(spawnTry, 0, 5, 1);
            Set Global Variable(n1, Count Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
            Set Global Variable(n2, Count Of(Filtered Array(All Players(Team 2), Is Dummy Bot(Current Array Element))));
            If(Or(Compare(Global.n1, <, Global.cfgBotsPerTeam), Compare(Global.n2, <, Global.cfgBotsPerTeam)));
                For Global Variable(spawnI, 0, Subtract(Global.cfgBotsPerTeam, Global.n1), 1);
                    Create Dummy Bot(Hero(Tracer), Team 1, -1, First Of(Global.startPos), Left);
                    Modify Global Variable(spawnCalls, Add, 1);
                    Wait(0.25, Ignore Condition);
                End;
                For Global Variable(spawnI, 0, Subtract(Global.cfgBotsPerTeam, Global.n2), 1);
                    Create Dummy Bot(Hero(Tracer), Team 2, -1, Value In Array(Global.startPos, 1), Left);
                    Modify Global Variable(spawnCalls, Add, 1);
                    Wait(0.25, Ignore Condition);
                End;
                Wait(1.5, Ignore Condition);
            End;
        End;
        Set Global Variable(bots, Append To Array(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element)), Filtered Array(All Players(Team 2), Is Dummy Bot(Current Array Element))));
        Set Global Variable(nBots, Count Of(Global.bots));
        If(Compare(Global.nBots, <, 1));
            Set Global Variable(calStage, -1);
            Set Global Variable(calMsg, Custom String("no bot spawned. Leave free slots on both teams, then Esc > Show Lobby > Restart Match"));
            Enable Inspector Recording;
            Log To Inspector(Custom String("NS,FAIL,spawn,{0},{1}", Number Of Players(Team 1), Number Of Players(Team 2)));
            Disable Inspector Recording;
            Abort;
        End;
        Set Move Speed(Global.bots, Global.cfgMoveSpeed);
        "Speed run with the first bot: its real run speed, and how far it coasts after the throttle stops."
        Set Global Variable(calMsg, Custom String("measuring run speed"));
        Start Forcing Player Position(First Of(Global.bots), Value In Array(Global.cellPos, 24), False);
        Wait(0.25, Ignore Condition);
        Stop Forcing Player Position(First Of(Global.bots));
        Set Facing(First Of(Global.bots), Left, To World);
        Wait(0.4, Ignore Condition);
        Start Throttle In Direction(First Of(Global.bots), Left, 1, To World, Replace existing throttle, None);
        Wait(0.4, Ignore Condition);
        Set Global Variable(calP0, Position Of(First Of(Global.bots)));
        Set Global Variable(calT0, Total Time Elapsed);
        Wait(0.6, Ignore Condition);
        Set Global Variable(calP1, Position Of(First Of(Global.bots)));
        Set Global Variable(runSpeed, Divide(Distance Between(Global.calP0, Global.calP1), Subtract(Total Time Elapsed, Global.calT0)));
        Stop Throttle In Direction(First Of(Global.bots));
        Wait(0.5, Ignore Condition);
        Set Global Variable(coastDist, Distance Between(Global.calP1, Position Of(First Of(Global.bots))));
        If(Compare(Global.runSpeed, <, 1));
            Set Global Variable(calStage, -1);
            Set Global Variable(calMsg, Custom String("the bot did not move under throttle control"));
            Enable Inspector Recording;
            Log To Inspector(Custom String("NS,FAIL,throttle,{0},{1},{2}", Global.runSpeed, Global.calP0, Global.calP1));
            Disable Inspector Recording;
            Abort;
        End;
        Set Global Variable(actTimeout, Add(Divide(Multiply(2.5, Global.cell), Global.runSpeed), 0.25));
        Set Global Variable(condOffset, Random Integer(0, 11));
        Set Global Variable(feat, Array(First Of(Global.bots), Last Of(Global.bots)));
        Set Global Variable(matchT0, Total Time Elapsed);
        "M: version, bots, team1 bots, cell, run speed, coast, move speed %, eval episodes, step limit, eval step limit,\ngamma*1000, r_step, r_goal, r_fall, r_timeout, condition offset, spawn calls, conditions"
        Enable Inspector Recording;
        "M: version, bots, team1 bots, cell, run speed, coast, move speed %, eval episodes, step limit, eval step limit,\ngamma*1000, r_step, r_goal, r_fall, r_timeout, condition offset, spawn calls, conditions"
        Log To Inspector(Custom String("NS,M,202,{0},{1},{2}", Global.nBots, Count Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))), Custom String("{0},{1},{2}", Global.cell, Global.runSpeed, Custom String("{0},{1},{2}", Global.coastDist, Global.cfgMoveSpeed, Custom String("{0},{1},{2}", Global.cfgEvalEps, Global.cfgMaxSteps, Custom String("{0},{1},{2}", Global.cfgEvalMaxSteps, Round To Integer(Multiply(Global.gamma, 1000), To Nearest), Custom String("{0},{1},{2}", Global.rStep, Global.rGoal, Custom String("{0},{1},{2}", Global.rFall, Global.rTimeout, Custom String("{0},{1},12", Global.condOffset, Global.spawnCalls)))))))));
        "M: version, bots, team1 bots, cell, run speed, coast, move speed %, eval episodes, step limit, eval step limit,\ngamma*1000, r_step, r_goal, r_fall, r_timeout, condition offset, spawn calls, conditions"
        Disable Inspector Recording;
        Enable Inspector Recording;
        Log To Inspector(Custom String("NS,CAL,{0},{1},{2}", Global.floorY, Global.edgeXMin, Custom String("{0},{1},{2}", Global.edgeXMax, Global.edgeZMin, Custom String("{0},{1},{2}", Global.edgeZMax, Global.gridX0, Global.actTimeout))));
        Disable Inspector Recording;
        Modify Global Variable(logBytes, Add, 260);
        For Global Variable(spawnI, 0, Global.nBots, 1);
            Set Player Variable(Value In Array(Global.bots, Global.spawnI), side, If-Then-Else(Compare(Team Of(Value In Array(Global.bots, Global.spawnI)), ==, Team 1), 0, 1));
            Set Player Variable(Value In Array(Global.bots, Global.spawnI), bid, Global.spawnI);
        End;
        Set Global Variable(calStage, 2);
        Set Global Variable(calMsg, Custom String("running"));
    }
}

rule ("Environment: put the bot back on its start cell") {
    event {
        Subroutine;
        resetBot;
    }
    actions {
        Stop Throttle In Direction(Event Player);
        If(Is Dead(Event Player));
            Resurrect(Event Player);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Event Player));
                Respawn(Event Player);
                Wait Until(Is Alive(Event Player), 3);
            End;
        End;
        Start Forcing Player Position(Event Player, Value In Array(Global.startPos, (Event Player).side), False);
        Wait(0.15, Ignore Condition);
        Stop Forcing Player Position(Event Player);
        Set Facing(Event Player, Left, To World);
        Wait Until(Is On Ground(Event Player), 1);
        Set Player Variable(Event Player, died, False);
    }
}

rule ("Environment: carry out action curA with the real bot and observe the result") {
    event {
        Subroutine;
        stepEnv;
    }
    actions {
        Set Player Variable(Event Player, stepCol, Modulo((Event Player).curS, 12));
        Set Player Variable(Event Player, stepRow, Round To Integer(Divide((Event Player).curS, 12), Down));
        Set Player Variable(Event Player, actDir, Value In Array(Array(Left, Right, Vector(0, 0, Value In Array(Global.zSign, (Event Player).side)), Vector(0, 0, Subtract(0, Value In Array(Global.zSign, (Event Player).side)))), (Event Player).curA));
        Set Player Variable(Event Player, tgtCol, Add((Event Player).stepCol, Value In Array(Array(1, -1, 0, 0), (Event Player).curA)));
        Set Player Variable(Event Player, tgtRow, Add((Event Player).stepRow, Value In Array(Array(0, 0, 1, -1), (Event Player).curA)));
        Set Player Variable(Event Player, reward, Global.rStep);
        Set Player Variable(Event Player, terminal, False);
        Set Player Variable(Event Player, outcome, 0);
        If(Or(Or(Compare((Event Player).tgtCol, <, 0), Compare((Event Player).tgtCol, >=, 12)), Compare((Event Player).tgtRow, >=, 4)));
            "Blocked: that side of the grid is open floor and the rule says no. The step still costs its reward."
            Set Player Variable(Event Player, intended, (Event Player).curS);
            Wait(0.032, Ignore Condition);
        Else;
            Set Player Variable(Event Player, intended, Add(Multiply((Event Player).tgtRow, 12), (Event Player).tgtCol));
            Set Player Variable(Event Player, tgtPos, Vector(Add(Global.gridX0, Multiply(Add((Event Player).tgtCol, 0.5), Global.cell)), Global.floorY, Add(Value In Array(Global.zBase, (Event Player).side), Multiply(Multiply(Value In Array(Global.zSign, (Event Player).side), Add((Event Player).tgtRow, 0.5)), Global.cell))));
            Set Facing(Event Player, (Event Player).actDir, To World);
            Start Throttle In Direction(Event Player, (Event Player).actDir, 1, To World, Replace existing throttle, None);
            If(Compare((Event Player).tgtRow, <, 0));
                "Off the edge: keep running until the bot is really falling."
                Wait Until(Or(Compare(Y Component Of(Position Of(Event Player)), <, Subtract(Global.floorY, 1.2)), (Event Player).died), Global.actTimeout);
            Else;
                "Cut the throttle one coast distance short, so the bot slides to a stop on the cell centre."
                Wait Until(Or(Or(Compare(Dot Product(Subtract((Event Player).tgtPos, Position Of(Event Player)), (Event Player).actDir), <=, Global.coastDist), Compare(Y Component Of(Position Of(Event Player)), <, Subtract(Global.floorY, 1.2))), (Event Player).died), Global.actTimeout);
            End;
            Stop Throttle In Direction(Event Player);
            If(Compare((Event Player).tgtRow, <, 0));
                Wait Until(Or(Compare(Y Component Of(Position Of(Event Player)), <, Subtract(Global.floorY, 1.2)), (Event Player).died), 1);
            Else;
                Wait Until(Or(Or(Compare(Horizontal Speed Of(Event Player), <, 0.2), Compare(Y Component Of(Position Of(Event Player)), <, Subtract(Global.floorY, 1.2))), (Event Player).died), 0.4);
            End;
        End;
        Call Subroutine(observe);
        Set Player Variable(Event Player, nextS, (Event Player).obsCell);
        If(Or(Compare(Y Component Of(Position Of(Event Player)), <, Subtract(Global.floorY, 1.2)), (Event Player).died));
            Set Player Variable(Event Player, terminal, True);
            Set Player Variable(Event Player, outcome, 2);
            Modify Player Variable(Event Player, reward, Add, Global.rFall);
            If(Compare((Event Player).tgtRow, >=, 0));
                Modify Player Variable(Event Player, epSlips, Add, 1);
            End;
        Else;
            Modify Player Variable(Event Player, epEdgeSum, Add, Add(Multiply(Value In Array(Global.zSign, (Event Player).side), Subtract(Z Component Of((Event Player).obsP), Value In Array(Global.zBase, (Event Player).side))), 0.5));
            If(Compare((Event Player).nextS, !=, (Event Player).intended));
                Modify Player Variable(Event Player, epSlips, Add, 1);
            End;
            If(Compare((Event Player).nextS, ==, 11));
                Set Player Variable(Event Player, terminal, True);
                Set Player Variable(Event Player, outcome, 1);
                Modify Player Variable(Event Player, reward, Add, Global.rGoal);
            Else If(Compare(Add((Event Player).epSteps, 1), >=, If-Then-Else((Event Player).isTraining, Global.cfgMaxSteps, Global.cfgEvalMaxSteps)));
                Set Player Variable(Event Player, terminal, True);
                Set Player Variable(Event Player, outcome, 3);
                Modify Player Variable(Event Player, reward, Add, Global.rTimeout);
            End;
        End;
    }
}

rule ("Agent: the update after one real step") {
    event {
        Subroutine;
        learnStep;
    }
    actions {
        Set Player Variable(Event Player, idx, Add(Multiply((Event Player).curS, 4), (Event Player).curA));
        Set Player Variable(Event Player, qOld, Value In Array((Event Player).Q, (Event Player).idx));
        Set Player Variable(Event Player, qDirect, (Event Player).qOld);
        Set Player Variable(Event Player, tgt, 0);
        If(Compare((Event Player).algo, ==, 0));
            "SARSA: score the move by the action the bot will really take next."
            Set Player Variable(Event Player, tgt, Add((Event Player).reward, If-Then-Else((Event Player).terminal, 0, Multiply(Global.gamma, Value In Array((Event Player).Q, Add(Multiply((Event Player).nextS, 4), (Event Player).nextA))))));
            Set Player Variable At Index(Event Player, Q, (Event Player).idx, Add((Event Player).qOld, Multiply((Event Player).alpha, Subtract((Event Player).tgt, (Event Player).qOld))));
        Else If(Or(Compare((Event Player).algo, ==, 1), Compare((Event Player).algo, ==, 3)));
            "Q-learning: score the move by the best action available next."
            Set Player Variable(Event Player, tgt, Add((Event Player).reward, If-Then-Else((Event Player).terminal, 0, Multiply(Global.gamma, Last Of(Sorted Array(Array Slice((Event Player).Q, Multiply((Event Player).nextS, 4), 4), Current Array Element))))));
            Set Player Variable At Index(Event Player, Q, (Event Player).idx, Add((Event Player).qOld, Multiply((Event Player).alpha, Subtract((Event Player).tgt, (Event Player).qOld))));
            Set Player Variable(Event Player, qDirect, Value In Array((Event Player).Q, (Event Player).idx));
            If(And(Compare((Event Player).algo, ==, 3), Compare((Event Player).outcome, !=, 3)));
                "Dyna-Q (Lecture 8): remember what this move did, then replay remembered moves in imagination.\nA timeout is a property of the clock, not of the move, so it is kept out of the model."
                If(Not(Value In Array((Event Player).mKnown, (Event Player).idx)));
                    Set Player Variable At Index(Event Player, mKnown, (Event Player).idx, True);
                    Modify Player Variable(Event Player, mSeen, Append To Array, (Event Player).idx);
                End;
                Set Player Variable At Index(Event Player, mS2, (Event Player).idx, If-Then-Else((Event Player).terminal, -1, (Event Player).nextS));
                Set Player Variable At Index(Event Player, mR, (Event Player).idx, (Event Player).reward);
                For Player Variable(Event Player, k, 0, (Event Player).planN, 1);
                    Set Player Variable(Event Player, pIdx, Random Value In Array((Event Player).mSeen));
                    Set Player Variable(Event Player, s2, Value In Array((Event Player).mS2, (Event Player).pIdx));
                    Set Player Variable(Event Player, qNew, Add(Value In Array((Event Player).mR, (Event Player).pIdx), If-Then-Else(Compare((Event Player).s2, <, 0), 0, Multiply(Global.gamma, Last Of(Sorted Array(Array Slice((Event Player).Q, Multiply((Event Player).s2, 4), 4), Current Array Element))))));
                    Modify Player Variable At Index(Event Player, Q, (Event Player).pIdx, Add, Multiply((Event Player).alpha, Subtract((Event Player).qNew, Value In Array((Event Player).Q, (Event Player).pIdx))));
                End;
            End;
        Else If(Compare((Event Player).algo, ==, 2));
            "Expected SARSA: score the move by the average over what the bot might do next."
            Set Player Variable(Event Player, rowQ, Array Slice((Event Player).Q, Multiply((Event Player).nextS, 4), 4));
            Set Player Variable(Event Player, tgt, Add((Event Player).reward, If-Then-Else((Event Player).terminal, 0, Multiply(Global.gamma, Add(Multiply(Subtract(1, (Event Player).eps), Last Of(Sorted Array((Event Player).rowQ, Current Array Element))), Divide(Multiply((Event Player).eps, Add(Add(Add(First Of((Event Player).rowQ), Value In Array((Event Player).rowQ, 1)), Value In Array((Event Player).rowQ, 2)), Value In Array((Event Player).rowQ, 3))), 4))))));
            Set Player Variable At Index(Event Player, Q, (Event Player).idx, Add((Event Player).qOld, Multiply((Event Player).alpha, Subtract((Event Player).tgt, (Event Player).qOld))));
        Else If(Compare((Event Player).algo, ==, 5));
            "SARSA(lambda), replacing traces: one surprise updates every recently visited move, fading with age."
            Set Player Variable(Event Player, tgt, Add((Event Player).reward, If-Then-Else((Event Player).terminal, 0, Multiply(Global.gamma, Value In Array((Event Player).Q, Add(Multiply((Event Player).nextS, 4), (Event Player).nextA))))));
            Set Player Variable(Event Player, delta, Subtract((Event Player).tgt, (Event Player).qOld));
            Set Player Variable At Index(Event Player, E, (Event Player).idx, 1);
            Set Player Variable(Event Player, Q, Mapped Array((Event Player).Q, Add(Current Array Element, Multiply(Multiply((Event Player).alpha, (Event Player).delta), Value In Array((Event Player).E, Current Array Index)))));
            Set Player Variable(Event Player, E, Mapped Array((Event Player).E, Multiply(Multiply(Current Array Element, Global.gamma), (Event Player).lam)));
        Else If(Or(Compare((Event Player).algo, ==, 4), Compare((Event Player).algo, ==, 8)));
            Modify Player Variable(Event Player, epS, Append To Array, (Event Player).idx);
            Modify Player Variable(Event Player, epR, Append To Array, (Event Player).reward);
        Else If(Compare((Event Player).algo, ==, 6));
            "Prediction (Lecture 4): the policy is fixed (uniform random); TD(0) learns its value every step."
            Set Player Variable(Event Player, tgt, Add((Event Player).reward, If-Then-Else((Event Player).terminal, 0, Multiply(Global.gamma, Value In Array((Event Player).Vtd, (Event Player).nextS)))));
            Modify Player Variable At Index(Event Player, Vtd, (Event Player).curS, Add, Multiply((Event Player).alpha, Subtract((Event Player).tgt, Value In Array((Event Player).Vtd, (Event Player).curS))));
            Modify Player Variable(Event Player, epS, Append To Array, (Event Player).curS);
            Modify Player Variable(Event Player, epR, Append To Array, (Event Player).reward);
        End;
        Modify Player Variable At Index(Event Player, Nsa, (Event Player).idx, Add, 1);
        Set Player Variable(Event Player, gS, (Event Player).curS);
        Call Subroutine(updateGlyph);
    }
}

rule ("Agent: the update at the end of an episode (Monte Carlo, REINFORCE, MC prediction)") {
    event {
        Subroutine;
        learnEpisodeEnd;
    }
    actions {
        Set Player Variable(Event Player, g, 0);
        If(Compare((Event Player).algo, ==, 4));
            "Every-visit Monte Carlo control: move each visited Q toward the return that actually followed."
            For Player Variable(Event Player, k, Subtract(Count Of((Event Player).epS), 1), -1, -1);
                Set Player Variable(Event Player, g, Add(Multiply(Global.gamma, (Event Player).g), Value In Array((Event Player).epR, (Event Player).k)));
                Modify Player Variable At Index(Event Player, Q, Value In Array((Event Player).epS, (Event Player).k), Add, Multiply((Event Player).alpha, Subtract((Event Player).g, Value In Array((Event Player).Q, Value In Array((Event Player).epS, (Event Player).k)))));
            End;
        Else If(Compare((Event Player).algo, ==, 6));
            "Every-visit Monte Carlo prediction: V(s) is the running mean of the returns seen from s."
            For Player Variable(Event Player, k, Subtract(Count Of((Event Player).epS), 1), -1, -1);
                Set Player Variable(Event Player, g, Add(Multiply(Global.gamma, (Event Player).g), Value In Array((Event Player).epR, (Event Player).k)));
                Modify Player Variable At Index(Event Player, Nv, Value In Array((Event Player).epS, (Event Player).k), Add, 1);
                Modify Player Variable At Index(Event Player, Vmc, Value In Array((Event Player).epS, (Event Player).k), Add, Divide(Subtract((Event Player).g, Value In Array((Event Player).Vmc, Value In Array((Event Player).epS, (Event Player).k))), Value In Array((Event Player).Nv, Value In Array((Event Player).epS, (Event Player).k))));
            End;
        Else If(Compare((Event Player).algo, ==, 8));
            "REINFORCE with a baseline (Lecture 7): raise the preference of each action taken in proportion to\nhow much better than usual the episode went from there on. Q doubles as the baseline per state-action."
            For Player Variable(Event Player, k, Subtract(Count Of((Event Player).epS), 1), -1, -1);
                Set Player Variable(Event Player, g, Add(Multiply(Global.gamma, (Event Player).g), Value In Array((Event Player).epR, (Event Player).k)));
                Set Player Variable(Event Player, idx, Value In Array((Event Player).epS, (Event Player).k));
                Set Player Variable(Event Player, s2, Round To Integer(Divide((Event Player).idx, 4), Down));
                Set Player Variable(Event Player, rowQ, Array Slice((Event Player).theta, Multiply((Event Player).s2, 4), 4));
                Set Player Variable(Event Player, rowMax, Last Of(Sorted Array((Event Player).rowQ, Current Array Element)));
                Set Player Variable(Event Player, probs, Mapped Array((Event Player).rowQ, Raise To Power(2.718281828, Subtract(Current Array Element, (Event Player).rowMax))));
                Set Player Variable(Event Player, u, Add(Add(Add(First Of((Event Player).probs), Value In Array((Event Player).probs, 1)), Value In Array((Event Player).probs, 2)), Value In Array((Event Player).probs, 3)));
                Set Player Variable(Event Player, delta, Subtract((Event Player).g, Value In Array((Event Player).Vtd, (Event Player).s2)));
                Modify Player Variable At Index(Event Player, Vtd, (Event Player).s2, Add, Multiply(0.2, (Event Player).delta));
                For Player Variable(Event Player, a, 0, 4, 1);
                    Modify Player Variable At Index(Event Player, theta, Add(Multiply((Event Player).s2, 4), (Event Player).a), Add, Multiply(Multiply((Event Player).alpha, (Event Player).delta), Subtract(If-Then-Else(Compare((Event Player).a, ==, Modulo((Event Player).idx, 4)), 1, 0), Divide(Value In Array((Event Player).probs, (Event Player).a), (Event Player).u))));
                End;
                Modify Player Variable At Index(Event Player, Q, (Event Player).idx, Add, Multiply(0.2, Subtract((Event Player).g, Value In Array((Event Player).Q, (Event Player).idx))));
            End;
        End;
    }
}

rule ("Run: one episode") {
    event {
        Subroutine;
        runEpisode;
    }
    actions {
        If(Compare((Event Player).eps0, <, 0));
            Set Player Variable(Event Player, eps, Max(0.01, Divide(1, Add(1, Divide((Event Player).epCount, 5)))));
        End;
        Call Subroutine(resetBot);
        Set Player Variable(Event Player, epSteps, 0);
        Set Player Variable(Event Player, epReturn, 0);
        Set Player Variable(Event Player, epEdgeSum, 0);
        Set Player Variable(Event Player, epSlips, 0);
        Set Player Variable(Event Player, epS, Empty Array);
        Set Player Variable(Event Player, epR, Empty Array);
        If(Compare((Event Player).algo, ==, 5));
            Set Player Variable(Event Player, E, Empty Array);
            Set Player Variable At Index(Event Player, E, 191, 0);
        End;
        Call Subroutine(observe);
        Set Player Variable(Event Player, curS, (Event Player).obsCell);
        Set Player Variable(Event Player, selS, (Event Player).curS);
        Call Subroutine(chooseAction);
        Set Player Variable(Event Player, curA, (Event Player).selA);
        Set Player Variable(Event Player, curExplore, (Event Player).selExplore);
        Set Player Variable(Event Player, terminal, False);
        While(Not((Event Player).terminal));
            Call Subroutine(stepEnv);
            If(Not((Event Player).terminal));
                "Choose the next action now, with the behaviour policy, for every method."
                Set Player Variable(Event Player, selS, (Event Player).nextS);
                Call Subroutine(chooseAction);
                Set Player Variable(Event Player, nextA, (Event Player).selA);
                Set Player Variable(Event Player, nextExplore, (Event Player).selExplore);
            End;
            If(And((Event Player).isTraining, Compare((Event Player).algo, !=, 7)));
                Call Subroutine(learnStep);
                If(And(Compare((Event Player).runSteps, <, 30), Compare((Event Player).algo, <=, 3)));
                    Enable Inspector Recording;
                    Log To Inspector(Custom String("NS,S,{0},{1},{2}", (Event Player).bid, (Event Player).runCount, Custom String("{0},{1},{2}", (Event Player).runSteps, (Event Player).curS, Custom String("{0},{1},{2}", (Event Player).curA, (Event Player).reward, Custom String("{0},{1},{2}", (Event Player).nextS, (Event Player).nextA, Custom String("{0},{1},{2}", (Event Player).terminal, (Event Player).curExplore, Custom String("{0},{1},{2}", (Event Player).qOld, If-Then-Else(Compare((Event Player).algo, ==, 3), (Event Player).qDirect, Value In Array((Event Player).Q, Add(Multiply((Event Player).curS, 4), (Event Player).curA))), Custom String("{0},{1}", (Event Player).tgt, Round To Integer(Multiply((Event Player).eps, 1000), To Nearest)))))))));
                    Disable Inspector Recording;
                    Modify Global Variable(logBytes, Add, 75);
                End;
            End;
            Modify Player Variable(Event Player, epSteps, Add, 1);
            Modify Player Variable(Event Player, runSteps, Add, 1);
            Modify Player Variable(Event Player, epReturn, Add, (Event Player).reward);
            Set Player Variable(Event Player, curS, (Event Player).nextS);
            Set Player Variable(Event Player, curA, (Event Player).nextA);
            Set Player Variable(Event Player, curExplore, (Event Player).nextExplore);
        End;
        Stop Throttle In Direction(Event Player);
        If((Event Player).isTraining);
            Call Subroutine(learnEpisodeEnd);
            If(Or(Or(Or(Compare((Event Player).algo, ==, 3), Compare((Event Player).algo, ==, 4)), Compare((Event Player).algo, ==, 5)), Compare((Event Player).algo, ==, 8)));
                "These methods change many cells per episode: redraw every arrow."
                For Player Variable(Event Player, gS, 0, 48, 1);
                    Call Subroutine(updateGlyph);
                End;
            End;
            Modify Player Variable(Event Player, epCount, Add, 1);
            If(Compare((Event Player).outcome, ==, 1));
                Modify Player Variable(Event Player, cntGoal, Add, 1);
            Else If(Compare((Event Player).outcome, ==, 2));
                Modify Player Variable(Event Player, cntFell, Add, 1);
            Else;
                Modify Player Variable(Event Player, cntTimeout, Add, 1);
            End;
        Else;
            Modify Player Variable(Event Player, evalCount, Add, 1);
            If(Compare((Event Player).outcome, ==, 1));
                Modify Player Variable(Event Player, evGoal, Add, 1);
            Else If(Compare((Event Player).outcome, ==, 2));
                Modify Player Variable(Event Player, evFell, Add, 1);
            Else;
                Modify Player Variable(Event Player, evTimeout, Add, 1);
            End;
        End;
        Modify Player Variable(Event Player, runSlips, Add, (Event Player).epSlips);
        Set Player Variable(Event Player, lastReturn, (Event Player).epReturn);
        Set Player Variable(Event Player, lastOutcome, (Event Player).outcome);
        Modify Global Variable(totalEpisodes, Add, 1);
        Modify Global Variable(totalSteps, Add, (Event Player).epSteps);
        "E: bot, run, episode (evaluation episodes continue the count), mode (0 training, 1 evaluation), return, steps,\noutcome (1 goal, 2 fell, 3 timeout), mean distance from the edge (m), slips, seconds since match start"
        Enable Inspector Recording;
        "E: bot, run, episode (evaluation episodes continue the count), mode (0 training, 1 evaluation), return, steps,\noutcome (1 goal, 2 fell, 3 timeout), mean distance from the edge (m), slips, seconds since match start"
        Log To Inspector(Custom String("NS,E,{0},{1},{2}", (Event Player).bid, (Event Player).runCount, Custom String("{0},{1},{2}", Add((Event Player).epCount, (Event Player).evalCount), If-Then-Else((Event Player).isTraining, 0, 1), Custom String("{0},{1},{2}", (Event Player).epReturn, (Event Player).epSteps, Custom String("{0},{1},{2}", (Event Player).outcome, Divide((Event Player).epEdgeSum, Max(1, (Event Player).epSteps)), Custom String("{0},{1}", (Event Player).epSlips, Subtract(Total Time Elapsed, Global.matchT0)))))));
        "E: bot, run, episode (evaluation episodes continue the count), mode (0 training, 1 evaluation), return, steps,\noutcome (1 goal, 2 fell, 3 timeout), mean distance from the edge (m), slips, seconds since match start"
        Disable Inspector Recording;
        Modify Global Variable(logBytes, Add, 50);
        If(And(And(Compare((Event Player).algo, ==, 6), (Event Player).isTraining), Or(Or(Compare((Event Player).epCount, ==, 10), Compare((Event Player).epCount, ==, 30)), Compare((Event Player).epCount, ==, 100))));
            Call Subroutine(logTables);
        End;
    }
}

rule ("Bot: run after run until the block ends") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Is Dummy Bot(Event Player) == True;
        (Event Player).bid >= 0;
        Global.calStage == 2;
    }
    actions {
        Wait(Add(0.5, Multiply((Event Player).bid, 0.35)), Ignore Condition);
        While(And(And(Compare(Global.calStage, ==, 2), Not(Global.stopping)), Compare(Subtract(Total Time Elapsed, Global.matchT0), <, Multiply(Global.cfgSoftMin, 60))));
            Call Subroutine(startRun);
            While(And(And(Compare((Event Player).epCount, <, (Event Player).trainEps), Not(Global.stopping)), Compare(Subtract(Total Time Elapsed, (Event Player).runT0), <, Multiply(Global.cfgRunMaxMin, 60))));
                Call Subroutine(runEpisode);
                Wait(0.1, Ignore Condition);
            End;
            If(Compare((Event Player).algo, !=, 6));
                Set Player Variable(Event Player, isTraining, False);
                While(And(And(Compare((Event Player).evalCount, <, Global.cfgEvalEps), Not(Global.stopping)), Compare(Subtract(Total Time Elapsed, (Event Player).runT0), <, Add(Multiply(Global.cfgRunMaxMin, 60), 120))));
                    Call Subroutine(runEpisode);
                    Wait(0.1, Ignore Condition);
                End;
            End;
            Call Subroutine(endRun);
            Modify Player Variable(Event Player, runCount, Add, 1);
            Wait(0.5, Ignore Condition);
        End;
        Set Player Variable(Event Player, done, True);
        Start Forcing Player Position(Event Player, Value In Array(Global.startPos, (Event Player).side), False);
    }
}
