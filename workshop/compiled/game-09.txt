rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        Set Global Variable(learner, Host Player);
        Teleport(Global.learner, Vector(0, 0, -8));
        Wait(0.5, Ignore Condition);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,9,1,0"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(0, 0, -8));
        Wait(0.2, Ignore Condition);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: the learner fell off the island") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Event Player == Global.learner;
        Global.ready != False;
        Or(Compare(Y Component Of(Position Of(Event Player)), <, -1.5), Is Dead(Event Player)) == True;
    }
    actions {
        Set Player Variable(Event Player, fell, True);
        Modify Player Variable(Event Player, fellCount, Add, 1);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,9,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,9,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        Wait Until(Or(Compare(Global.qAnswer, >=, 0), Global.gameOver), 90);
        Set Global Variable(qActive, False);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,9,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(5, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
    }
}

rule ("Game 9: let the learner open boxes until the round's last open") {
    event {
        Subroutine;
        openRound;
    }
    actions {
        While(Compare(Global.myT, <, Global.roundEnd));
            Set Global Variable(wantOpen, -1);
            If((Global.learner).fell);
                Call Subroutine(putHome);
            End;
            Set Global Variable(canOpen, True);
            Wait Until(Or(Compare(Global.wantOpen, >=, 0), (Global.learner).fell), 99999);
            Set Global Variable(canOpen, False);
            If(Compare(Global.wantOpen, >=, 0));
                Set Global Variable(oBox, Global.wantOpen);
                Call Subroutine(doOpen);
                Wait(0.6, Ignore Condition);
            End;
        End;
    }
}

rule ("Game 9: open box oBox for the learner, and one open for each strategy") {
    event {
        Subroutine;
        doOpen;
    }
    actions {
        Set Global Variable(oK, Value In Array(Global.myN, Global.oBox));
        Set Global Variable(oPay, Value In Array(Global.payTab, Add(Multiply(Global.oBox, 30), Global.oK)));
        Modify Global Variable At Index(myN, Global.oBox, Add, 1);
        Modify Global Variable At Index(myS, Global.oBox, Add, Global.oPay);
        Modify Global Variable(myT, Add, 1);
        Modify Global Variable(myTot, Add, Global.oPay);
        Modify Global Variable(myReg, Add, Subtract(0.7, Value In Array(Global.odds, Global.oBox)));
        Small Message(All Players(All Teams), Custom String("BOX {0} PAID {1}", Value In Array(Global.label, Global.oBox), Global.oPay));
        "the three strategies take their open number myT"
        Set Global Variable(sTn, Subtract(Global.myT, 1));
        Set Global Variable(sX, Value In Array(Global.coinTab, Global.sTn));
        For Global Variable(sK, 0, 3, 1);
            Call Subroutine(stratChoose);
            Set Global Variable At Index(sA, Global.sK, Global.armPick);
            Set Global Variable At Index(sPay, Global.sK, Value In Array(Global.payTab, Add(Multiply(Global.armPick, 30), Value In Array(Global.sN, Add(Multiply(Global.sK, 4), Global.armPick)))));
            Modify Global Variable At Index(sN, Add(Multiply(Global.sK, 4), Global.armPick), Add, 1);
            Modify Global Variable At Index(sS, Add(Multiply(Global.sK, 4), Global.armPick), Add, Value In Array(Global.sPay, Global.sK));
            Modify Global Variable At Index(sTot, Global.sK, Add, Value In Array(Global.sPay, Global.sK));
            Modify Global Variable At Index(sReg, Global.sK, Add, Subtract(0.7, Value In Array(Global.odds, Global.armPick)));
        End;
        Call Subroutine(refreshView);
        "open: t, box, which open of that box, paid, his total, his regret"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,9,open,{0},{1},{2}", Global.myT, Global.oBox, Custom String("{0},{1},{2}", Global.oK, Global.oPay, Custom String("{0},{1}", Global.myTot, Global.myReg))));
        "step: t, coin draw, then box and pay for greedy, coin, optimist"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,9,step,{0},{1},{2}", Global.myT, Global.sX, Custom String("{0},{1},{2}", First Of(Global.sA), First Of(Global.sPay), Custom String("{0},{1},{2}", Value In Array(Global.sA, 1), Value In Array(Global.sPay, 1), Custom String("{0},{1}", Value In Array(Global.sA, 2), Value In Array(Global.sPay, 2))))));

    }
}

rule ("Game 9: replays") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.setupDone != False;
    }
    actions {
        Set Global Variable(replayT0, Total Time Elapsed);
        Set Global Variable(rpSum, Empty Array);
        Set Global Variable At Index(rpSum, 35, 0);
        Set Global Variable(rpSum, Mapped Array(Global.rpSum, 0));
        For Global Variable(rpK, 0, 3, 1);
            For Global Variable(rpR, 0, 20, 1);
                Set Global Variable(rpN, Array(0, 0, 0, 0));
                Set Global Variable(rpS, Array(0, 0, 0, 0));
                For Global Variable(rpT, 0, 300, 1);
                    "rpT opens so far"
                    If(Compare(Global.rpT, <, 4));
                        Set Global Variable(rpA, Global.rpT);
                    Else;
                        Set Global Variable(rpV, Mapped Array(Global.rpN, Add(Divide(Value In Array(Global.rpS, Current Array Index), Current Array Element), If-Then-Else(Compare(Global.rpK, ==, 2), Square Root(Divide(Multiply(2, Value In Array(Global.lnTab, Global.rpT)), Current Array Element)), 0))));
                        Set Global Variable(rpA, Index Of Array Value(Global.rpV, Last Of(Sorted Array(Global.rpV, Current Array Element))));
                        If(Compare(Global.rpK, ==, 1));
                            Set Global Variable(rpX, Random Integer(0, 39));
                            If(Compare(Global.rpX, <, 4));
                                Set Global Variable(rpA, Global.rpX);
                            End;
                        End;
                    End;
                    Modify Global Variable At Index(rpN, Global.rpA, Add, 1);
                    Modify Global Variable At Index(rpS, Global.rpA, Add, If-Then-Else(Compare(Random Real(0, 1), <, Value In Array(Global.odds, Global.rpA)), 1, 0));
                    If(Compare(Modulo(Global.rpT, 100), ==, 99));
                        For Global Variable(rpI, 0, 4, 1);
                            Modify Global Variable At Index(rpSum, Add(Multiply(Add(Multiply(Global.rpK, 3), Round To Integer(Divide(Global.rpT, 100), Down)), 4), Global.rpI), Add, Value In Array(Global.rpN, Global.rpI));
                        End;
                    End;
                    If(Compare(Modulo(Global.rpT, 15), ==, 14));
                        Wait(0.016, Ignore Condition);
                    End;
                End;
            End;
        End;
        For Global Variable(rpC, 0, 9, 1);
            Set Global Variable At Index(rpReg, Global.rpC, Divide(Add(Add(Add(Multiply(Value In Array(Global.rpSum, Multiply(Global.rpC, 4)), Subtract(0.7, First Of(Global.odds))), Multiply(Value In Array(Global.rpSum, Add(Multiply(Global.rpC, 4), 1)), Subtract(0.7, Value In Array(Global.odds, 1)))), Multiply(Value In Array(Global.rpSum, Add(Multiply(Global.rpC, 4), 2)), Subtract(0.7, Value In Array(Global.odds, 2)))), Multiply(Value In Array(Global.rpSum, Add(Multiply(Global.rpC, 4), 3)), Subtract(0.7, Value In Array(Global.odds, 3)))), 20));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,9,replay,{0},{1},{2}", Round To Integer(Divide(Global.rpC, 3), Down), Modulo(Global.rpC, 3), Custom String("{0},{1},{2}", Value In Array(Global.rpSum, Multiply(Global.rpC, 4)), Value In Array(Global.rpSum, Add(Multiply(Global.rpC, 4), 1)), Custom String("{0},{1},{2}", Value In Array(Global.rpSum, Add(Multiply(Global.rpC, 4), 2)), Value In Array(Global.rpSum, Add(Multiply(Global.rpC, 4), 3)), Value In Array(Global.rpReg, Global.rpC)))));
        End;
        Set Global Variable At Index(hudRp, 0, Custom String("20 replays of 300 opens. Regret added in opens 1-100, 101-200, 201-300:"));
        Set Global Variable At Index(hudRp, 1, Custom String("greedy {0}, {1}, {2}", Divide(Round To Integer(Multiply(First Of(Global.rpReg), 10), To Nearest), 10), Divide(Round To Integer(Multiply(Subtract(Value In Array(Global.rpReg, 1), First Of(Global.rpReg)), 10), To Nearest), 10), Custom String("{0}      coin {1}, {2}", Divide(Round To Integer(Multiply(Subtract(Value In Array(Global.rpReg, 2), Value In Array(Global.rpReg, 1)), 10), To Nearest), 10), Divide(Round To Integer(Multiply(Value In Array(Global.rpReg, 3), 10), To Nearest), 10), Custom String("{0}, {1}", Divide(Round To Integer(Multiply(Subtract(Value In Array(Global.rpReg, 4), Value In Array(Global.rpReg, 3)), 10), To Nearest), 10), Divide(Round To Integer(Multiply(Subtract(Value In Array(Global.rpReg, 5), Value In Array(Global.rpReg, 4)), 10), To Nearest), 10)))));
        Set Global Variable At Index(hudRp, 2, Custom String("optimist {0}, {1}, {2}      the coin can never go under 2.40 per 100", Divide(Round To Integer(Multiply(Value In Array(Global.rpReg, 6), 10), To Nearest), 10), Divide(Round To Integer(Multiply(Subtract(Value In Array(Global.rpReg, 7), Value In Array(Global.rpReg, 6)), 10), To Nearest), 10), Divide(Round To Integer(Multiply(Subtract(Value In Array(Global.rpReg, 8), Value In Array(Global.rpReg, 7)), 10), To Nearest), 10)));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,9,replaytime,{0}", Subtract(Total Time Elapsed, Global.replayT0)));
        Set Global Variable(replayDone, True);
    }
