rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        Set Global Variable(learner, Host Player);
        Teleport(Global.learner, Vector(-12, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Left, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,8,2,0"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(-12, 0, 0));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Left, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,8,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,8,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        Wait Until(Or(Compare(Global.qAnswer, >=, 0), Global.gameOver), 90);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,8,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(5, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

                Create Effect(All Players(All Teams), Sphere, Color(Red), Add(Vector(Add(-12, Multiply(4, Modulo(Global.drawI, 7))), 0, Add(-4, Multiply(4, Round To Integer(Divide(Global.drawI, 7), Down)))), Up), 1.3, None);
            Else If(Compare(Global.drawI, ==, 7));

            Else If(Compare(Global.drawI, ==, 13));

                Create Effect(All Players(All Teams), Light Shaft, Color(Green), Vector(Add(-12, Multiply(4, Modulo(Global.drawI, 7))), 0, Add(-4, Multiply(4, Round To Integer(Divide(Global.drawI, 7), Down)))), 1.4, None);

rule ("Bot: empty value table; the replaying bot's memory starts as the player's walk") {
    event {
        Subroutine;
        freshTables;
    }
    actions {
        Set Player Variable(Event Player, Q, Empty Array);
        For Player Variable(Event Player, j, 0, 84, 1);
            Modify Player Variable(Event Player, Q, Append To Array, 0);
            If(Compare(Modulo((Event Player).j, 28), ==, 27));
                "load: both bots fill their tables in the same frame otherwise"
                Wait(0.016, Ignore Condition);
            End;
        End;
        Set Player Variable(Event Player, mS2, Empty Array);
        Set Player Variable(Event Player, mR, Empty Array);
        Set Player Variable(Event Player, mSeen, Empty Array);
        If(Compare((Event Player).role, ==, 1));
            For Player Variable(Event Player, j, 0, Count Of(Global.walkK), 1);
                If(Not(Array Contains((Event Player).mSeen, Value In Array(Global.walkK, (Event Player).j))));
                    Modify Player Variable(Event Player, mSeen, Append To Array, Value In Array(Global.walkK, (Event Player).j));
                End;
                Set Player Variable At Index(Event Player, mS2, Value In Array(Global.walkK, (Event Player).j), Value In Array(Global.walkS2, (Event Player).j));
                Set Player Variable At Index(Event Player, mR, Value In Array(Global.walkK, (Event Player).j), Value In Array(Global.walkRew, (Event Player).j));
            End;
        End;
    }
}

rule ("Bot: one real step, its Q-learning update, then (replaying bot) remember it and replay memories") {
    event {
        Subroutine;
        realStep;
    }
    actions {
        "Choose: one time in ten a random move, else the best-valued move (ties broken by the dice)."
        Set Player Variable(Event Player, dx, Modulo(Multiply((Event Player).dx, 171), 30269));
        If(Compare((Event Player).dx, <, 3027));
            Set Player Variable(Event Player, dx, Modulo(Multiply((Event Player).dx, 171), 30269));
            Set Player Variable(Event Player, a, Round To Integer(Divide(Multiply((Event Player).dx, 4), 30269), Down));
        Else;
            Set Player Variable(Event Player, row, Array Slice((Event Player).Q, Multiply((Event Player).cur, 4), 4));
            Set Player Variable(Event Player, rmax, Last Of(Sorted Array((Event Player).row, Current Array Element)));
            Set Player Variable(Event Player, ties, Filtered Array(Array(0, 1, 2, 3), Compare(Value In Array((Event Player).row, Current Array Element), >=, Subtract((Event Player).rmax, 0.0001))));
            If(And(Global.showRace, Compare(Count Of(Filtered Array(Array(0, 1, 2, 3), Compare(Absolute Value(Subtract(Subtract((Event Player).rmax, Value In Array((Event Player).row, Current Array Element)), 0.0001)), <, 0.00003))), >, 0)));
                Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,tie,{0},{1},{2}", Global.raceNo, (Event Player).role, Custom String("{0},{1}", Add((Event Player).steps, 1), Add(Add(Add(If-Then-Else(Array Contains((Event Player).ties, 0), 1, 0), If-Then-Else(Array Contains((Event Player).ties, 1), 2, 0)), If-Then-Else(Array Contains((Event Player).ties, 2), 4, 0)), If-Then-Else(Array Contains((Event Player).ties, 3), 8, 0)))));
            End;
            Set Player Variable(Event Player, dx, Modulo(Multiply((Event Player).dx, 171), 30269));
            Set Player Variable(Event Player, a, Value In Array((Event Player).ties, Round To Integer(Divide(Multiply((Event Player).dx, Count Of((Event Player).ties)), 30269), Down)));
        End;
        Set Player Variable(Event Player, k, Add(Multiply((Event Player).cur, 4), (Event Player).a));
        Set Player Variable(Event Player, s2, If-Then-Else(Or(Or(Or(Or(Compare(Add(Modulo((Event Player).cur, 7), Value In Array(Array(1, -1, 0, 0), (Event Player).a)), <, 0), Compare(Add(Modulo((Event Player).cur, 7), Value In Array(Array(1, -1, 0, 0), (Event Player).a)), >=, 7)), Compare(Add(Round To Integer(Divide((Event Player).cur, 7), Down), Value In Array(Array(0, 0, 1, -1), (Event Player).a)), <, 0)), Compare(Add(Round To Integer(Divide((Event Player).cur, 7), Down), Value In Array(Array(0, 0, 1, -1), (Event Player).a)), >=, 3)), And(Compare(Add(Modulo((Event Player).cur, 7), Value In Array(Array(1, -1, 0, 0), (Event Player).a)), ==, 3), Compare(Add(Round To Integer(Divide((Event Player).cur, 7), Down), Value In Array(Array(0, 0, 1, -1), (Event Player).a)), !=, Global.gap))), (Event Player).cur, Add(Multiply(Add(Round To Integer(Divide((Event Player).cur, 7), Down), Value In Array(Array(0, 0, 1, -1), (Event Player).a)), 7), Add(Modulo((Event Player).cur, 7), Value In Array(Array(1, -1, 0, 0), (Event Player).a)))));
        Set Player Variable(Event Player, r, 0);
        If(Compare((Event Player).s2, ==, 13));
            Set Player Variable(Event Player, r, 1);
            Set Player Variable(Event Player, s2, -1);
        End;
        "Q-learning (Lecture 5): nudge the move's value toward its reward plus the best value where it landed."
        Modify Player Variable At Index(Event Player, Q, (Event Player).k, Add, Multiply(0.5, Subtract(Add((Event Player).r, Multiply(0.9, If-Then-Else(Compare((Event Player).s2, <, 0), 0, Last Of(Sorted Array(Array Slice((Event Player).Q, Multiply((Event Player).s2, 4), 4), Current Array Element))))), Value In Array((Event Player).Q, (Event Player).k))));
        Modify Player Variable(Event Player, tripN, Add, 1);
        Modify Player Variable(Event Player, steps, Add, 1);
        If(Compare((Event Player).role, ==, 1));
            "Dyna-Q (Lecture 8): write what the move did into memory, then replay nRep remembered moves."
            If(Not(Array Contains((Event Player).mSeen, (Event Player).k)));
                Modify Player Variable(Event Player, mSeen, Append To Array, (Event Player).k);
            End;
            Set Player Variable At Index(Event Player, mS2, (Event Player).k, (Event Player).s2);
            Set Player Variable At Index(Event Player, mR, (Event Player).k, (Event Player).r);
            For Player Variable(Event Player, j, 0, Global.nRep, 1);
                Set Player Variable(Event Player, dy, Modulo(Multiply((Event Player).dy, 171), 30269));
                Set Player Variable(Event Player, pk, Value In Array((Event Player).mSeen, Round To Integer(Divide(Multiply((Event Player).dy, Count Of((Event Player).mSeen)), 30269), Down)));
                Set Player Variable(Event Player, ps2, Value In Array((Event Player).mS2, (Event Player).pk));
                If(Global.countStale);
                    "Count replays of a memory the moved wall has made untrue (moves from a pad that is now wall excepted)."

        Set Global Variable(sTotD, (Global.botD).total);
        Set Global Variable(sFsP, (Global.botP).firstShort);
        Set Global Variable(sFsD, (Global.botD).firstShort);
        Wait(0.1, Ignore Condition);
        Set Global Variable(qId, 2);
        Set Global Variable(qText, Custom String("{0} replays per real step. Who takes a short trip (10 steps or fewer) first?", Global.nRep));
        Set Global Variable(qOpts, Array(Custom String("the replaying bot"), Custom String("the plain bot"), Custom String("a tie")));
        Set Global Variable(qCorrect, If-Then-Else(Compare(Global.sFsD, <, Global.sFsP), 0, If-Then-Else(Compare(Global.sFsP, <, Global.sFsD), 1, 2)));
        Set Global Variable(qAuto, 0);
        Set Global Variable(qWhy, Custom String("Replays spread its memory, your walk included, through its values between real steps."));
        If(Compare(Global.qCorrect, ==, 2));
            Set Global Variable(qWhy, If-Then-Else(Compare(Global.nRep, ==, 0), Custom String("With 0 replays it is the plain bot: same rule, same dice, same trips."), Custom String("Both got there on the same trip this time.")));
        Else If(Compare(Global.qCorrect, ==, 1));
            Set Global Variable(qWhy, Custom String("The dice favoured the plain bot this time."));
        End;
        Call Subroutine(ask);
        Set Global Variable(coach, Custom String("Race 1. Both bots take one real step at a time. Watch the steps per trip on the right."));
        Set Global Variable(showRace, True);
        Call Subroutine(runRace);
        Set Global Variable(totD1, (Global.botD).total);
        Call Subroutine(raceSummary);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,pass,{0},{1},{2}", Global.raceNo, Global.sTotP, Custom String("{0},{1},{2}", (Global.botP).total, Global.sTotD, Custom String("{0},{1}", (Global.botD).total, Global.nRep))));
        Set Global Variable(coach, If-Then-Else(Compare(Global.nRep, >, 0), Custom String("Each real step costs both bots the same. The replaying bot gets more learning out of each one."), Custom String("0 replays: the second bot is the plain bot. Memory unused is memory wasted.")));
        Wait(8, Ignore Condition);
        "--- Round 3: race 2, a different number of replays ------------------------------------------------------------"
        Set Global Variable(raceNo, 2);
        Set Global Variable(coach, Custom String("Race 2: pick a different number. The plain bot rolls the same dice again: it is your yardstick."));
        Call Subroutine(pickReplays);
        Set Global Variable(r2, Global.nRep);
        Set Global Variable(showRace, False);
        Set Global Variable(coach, Custom String("Race 2 set: {0} replays per real step.", Global.nRep));
        Call Subroutine(runRace);
        Set Global Variable(sTotP, (Global.botP).total);
        Set Global Variable(sTotD, (Global.botD).total);
        Set Global Variable(qId, 3);
        Set Global Variable(qText, Custom String("Now {0} replays, last race {1}. Its real steps in 10 trips, against last race?", Global.r2, Global.r1));
        Set Global Variable(qOpts, Array(Custom String("far fewer"), Custom String("about the same"), Custom String("far more")));
        Set Global Variable(qCorrect, If-Then-Else(Compare(Multiply(Global.sTotD, 3), <, Multiply(Global.totD1, 2)), 0, If-Then-Else(Compare(Multiply(Global.sTotD, 2), >, Multiply(Global.totD1, 3)), 2, 1)));
        Set Global Variable(qAuto, 0);
        If(Compare(Global.qCorrect, ==, 0));
            Set Global Variable(qWhy, Custom String("With more replays each real step is spread further through its memory."));
        Else If(Compare(Global.qCorrect, ==, 2));
            Set Global Variable(qWhy, Custom String("Fewer replays: less of its memory is used per real step, so it needs more real steps."));
        Else;
            Set Global Variable(qWhy, Custom String("Once the way has spread through its memory, more replays cannot make 8 steps shorter."));
        End;
        Call Subroutine(ask);
        Set Global Variable(coach, Custom String("Race 2."));
        Set Global Variable(showRace, True);
        Call Subroutine(runRace);
        Call Subroutine(raceSummary);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,pass,{0},{1},{2}", Global.raceNo, Global.sTotP, Custom String("{0},{1},{2}", (Global.botP).total, Global.sTotD, Custom String("{0},{1}", (Global.botD).total, Global.nRep))));
        Set Global Variable(coach, Custom String("Replay only spreads what it has seen: past a point, more thinking stops paying."));
        Wait(7, Ignore Condition);
        "--- Round 4: the gap moves -----------------------------------------------------------------------------------"
        Set Global Variable(raceNo, 3);
        Set Global Variable(nRep, 20);
        Set Global Variable(gap, 2);
        Destroy All Effects;
        Call Subroutine(drawMaze);
        Set Global Variable(raceHead, Custom String("Race 3, gap moved: {0} replays per step. Steps per trip:", Global.nRep));
        Set Global Variable(hudP, Custom String(""));
        Set Global Variable(hudD, Custom String(""));
        Set Global Variable(hudTot, Custom String(""));
        Set Global Variable(hudTot2, Custom String(""));
        Set Global Variable(coach, Custom String("The gap moved, like the enemy team rotating: your read on where they set up is now out of date."));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,pick,{0},{1}", Global.raceNo, Global.nRep));
        Set Global Variable(showRace, False);
        Set Global Variable(countStale, True);
        Call Subroutine(runRace);
        Set Global Variable(sTotP, (Global.botP).total);
        Set Global Variable(sTotD, (Global.botD).total);
        Set Global Variable(sFix, (Global.botD).fixStep);
        Set Global Variable(sThrough, (Global.botD).throughStep);
        Set Global Variable(qId, 4);
        Set Global Variable(qText, Custom String("The gap moved to the far side. Over 10 trips, which bot needs more real steps?"));
        Set Global Variable(qOpts, Array(Custom String("the replaying bot: stale memory"), Custom String("the plain bot"), Custom String("about the same")));
        Set Global Variable(qCorrect, If-Then-Else(Compare(Multiply(Global.sTotD, 4), >, Multiply(Global.sTotP, 5)), 0, If-Then-Else(Compare(Multiply(Global.sTotP, 4), >, Multiply(Global.sTotD, 5)), 1, 2)));
        Set Global Variable(qAuto, 0);
        If(Compare(Global.qCorrect, ==, 1));
            Set Global Variable(qWhy, Custom String("The plain bot unlearns its old map one real step at a time. Replays fix the other bot faster."));
        Else If(Compare(Global.qCorrect, ==, 0));
            Set Global Variable(qWhy, Custom String("This time its memory kept sending it to the old gap for longer."));
        Else;
            Set Global Variable(qWhy, Custom String("Both had to stumble on the new gap: thinking cannot find what was never seen."));
        End;
        Call Subroutine(ask);
        Set Global Variable(qId, 5);
        Set Global Variable(qText, Custom String("Its memory says the old gap is open. Which happens later?"));
        Set Global Variable(qOpts, Array(Custom String("it fixes the old-gap memory"), Custom String("it finds the new gap"), Custom String("about the same time")));
        If(And(Compare(Global.sFix, <, 0), Compare(Global.sThrough, <, 0)));
            Set Global Variable(qCorrect, 2);
        Else If(Or(Compare(Global.sThrough, <, 0), And(Compare(Global.sFix, >=, 0), Compare(Global.sFix, <, Global.sThrough))));
            Set Global Variable(qCorrect, 1);
        Else;
            Set Global Variable(qCorrect, 0);
        End;
        Set Global Variable(qAuto, 1);
        If(Compare(Global.qCorrect, ==, 1));
            Set Global Variable(qWhy, Custom String("A bump fixes the old line at once. No memory points at the new gap: only exploring finds it."));
        Else If(Compare(Global.qCorrect, ==, 0));
            Set Global Variable(qWhy, Custom String("A random step found the new gap before it tried the old one."));
        Else;
            Set Global Variable(qWhy, Custom String("Neither happened in 10 trips."));
        End;
        Call Subroutine(ask);
        Set Global Variable(coach, Custom String("Race 3, the moved gap."));
        Set Global Variable(showRace, True);
        Call Subroutine(runRace);
        Set Global Variable(countStale, False);
        Call Subroutine(raceSummary);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,pass,{0},{1},{2}", Global.raceNo, Global.sTotP, Custom String("{0},{1},{2}", (Global.botP).total, Global.sTotD, Custom String("{0},{1}", (Global.botD).total, Global.nRep))));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,wall,{0},{1},{2}", (Global.botD).stale, (Global.botD).fixStep, (Global.botD).throughStep));
        Set Global Variable(coach, Custom String("It replayed untrue memories {0} times. A bump fixed the old gap at step {1}; new gap found at step {2}.", (Global.botD).stale, If-Then-Else(Compare((Global.botD).fixStep, >=, 0), (Global.botD).fixStep, Custom String("never")), If-Then-Else(Compare((Global.botD).throughStep, >=, 0), (Global.botD).throughStep, Custom String("never"))));
        Wait(9, Ignore Condition);
        "--- Debrief ---------------------------------------------------------------------------------------------------"
        Set Global Variable(ctrl, Custom String(""));
        Set Global Variable(db1, Custom String("You walked once and coached two bots: one learned from real steps, one also replayed."));
        Set Global Variable(db2, Custom String("That is Dyna-Q: learn a model (its copy of what moves do), then plan by replaying it."));
        Set Global Variable(db3, Custom String("In an LLM agent: practice on a simulated user. A wrong simulation teaches wrong lessons."));
        Set Global Variable(debriefOn, True);
        Set Global Variable(coach, Custom String("Simplified: one tiny maze, an exact memory of each move, no bonus for old moves (Dyna-Q+), no search."));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,8,runs,{0},{1}", Global.r1, Global.r2));
        Call Subroutine(finish);
        Allow Button(Global.learner, Button(Ability 1));
        Allow Button(Global.learner, Button(Ability 2));
        Allow Button(Global.learner, Button(Ultimate));
    }
}
