rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        "The automatic check: a bot plays the learner's part. The host stands aside, out of the way and unseen."
        Create Dummy Bot(Hero(Tracer), Team 1, -1, Vector(-11, 0, -18.5), Left);
        Wait(1, Ignore Condition);
        Set Global Variable(learner, First Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
        Set Status(Host Player, Null, Phased Out, 99999);
        Set Invisible(Host Player, All);
        Teleport(Host Player, Vector(0, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Left, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,5,1,1"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(-11, 0, -18.5));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Left, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,5,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,5,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        Wait(1.2, Ignore Condition);
        Set Global Variable(qAnswer, Global.qAuto);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,5,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(0.6, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

rule ("Game 5: value of one lane's walker by iterative policy evaluation (lecture 3), its own arithmetic") {
    event {
        Subroutine;
        evalLane;
    }
    actions {
        "Inputs: evLane (0 edge, 1 inland), evEps (chance a move is replaced by a random one), evStep, evGoal, evFall.\nOutput: evOut, the expected total from the start pad. With step 0, goal 0, fall 1 it is the chance of a fall."
        Set Global Variable(evV, Empty Array);
        Set Global Variable At Index(evV, 47, 0);
        Set Global Variable(evSweeps, 0);
        For Global Variable(evSweep, 0, 200, 1);
            Set Global Variable(evDelta, 0);
            Modify Global Variable(evSweeps, Add, 1);
            For Global Variable(evK, 47, -1, -1);
                If(Compare(Global.evK, !=, 11));
                    Set Global Variable(evPol, If-Then-Else(Compare(Global.evLane, ==, 0), If-Then-Else(Compare(Round To Integer(Divide(Global.evK, 12), Down), >, 0), 3, 0), If-Then-Else(Compare(Modulo(Global.evK, 12), ==, 11), 3, If-Then-Else(Compare(Round To Integer(Divide(Global.evK, 12), Down), <, 1), 2, If-Then-Else(Compare(Round To Integer(Divide(Global.evK, 12), Down), >, 1), 3, 0)))));
                    Set Global Variable(evSum, 0);
                    For Global Variable(evD, 0, 4, 1);
                        Set Global Variable(evP, If-Then-Else(Compare(Global.evD, ==, Global.evPol), Add(Subtract(1, Global.evEps), Divide(Global.evEps, 4)), Divide(Global.evEps, 4)));
                        If(Compare(Global.evP, >, 0));
                            Set Global Variable(evR2, Add(Round To Integer(Divide(Global.evK, 12), Down), Value In Array(Array(0, 0, 1, -1), Global.evD)));
                            Set Global Variable(evC2, Add(Modulo(Global.evK, 12), Value In Array(Array(1, -1, 0, 0), Global.evD)));
                            If(Compare(Global.evR2, <, 0));
                                Set Global Variable(evX, Add(Global.evStep, Global.evFall));
                            Else;
                                Set Global Variable(evS2, If-Then-Else(Or(Or(Compare(Global.evC2, <, 0), Compare(Global.evC2, >, 11)), Compare(Global.evR2, >, 3)), Global.evK, Add(Multiply(Global.evR2, 12), Global.evC2)));
                                Set Global Variable(evX, Add(Global.evStep, If-Then-Else(Compare(Global.evS2, ==, 11), Global.evGoal, Value In Array(Global.evV, Global.evS2))));
                            End;
                            Modify Global Variable(evSum, Add, Multiply(Global.evP, Global.evX));
                        End;
                    End;
                    Modify Global Variable(evDelta, Max, Absolute Value(Subtract(Global.evSum, Value In Array(Global.evV, Global.evK))));
                    Set Global Variable At Index(evV, Global.evK, Global.evSum);
                End;
                "Every 4 pads, not 8: the stand-in run of 2026-11-22 peaked at 82% server load, and this loop is the\nheaviest work the game does in one frame."
                If(Compare(Modulo(Global.evK, 4), ==, 0));
                    Wait(0.016, Ignore Condition);
                End;
            End;
            If(Compare(Global.evDelta, <, 0.0005));
                Break;
            End;
        End;
        Set Global Variable(evOut, First Of(Global.evV));
    }
}

rule ("Game 5: one crossing, start ring to goal ring, with or without boops") {
    event {
        Subroutine;
        crossOnce;
    }
    actions {
        Call Subroutine(putHome);
        Set Facing(Global.learner, Left, To World);
        Set Global Variable(curCell, 0);
        Set Global Variable(ret, 0);
        Set Global Variable(steps, 0);
        Set Global Variable(boops, 0);
        Set Global Variable(outcome, 0);
        Set Global Variable(edgeSteps, 0);
        Set Global Variable(realFall, 0);
        Set Global Variable(hudRun, Custom String("Now: {0} lane, 0 steps, 0 boops, return 0", Value In Array(Global.laneName, Global.lane)));
        Set Global Variable(coach, Custom String("GO. Walk the {0} lane to the green ring. Each pad you step onto costs 1.", Value In Array(Global.laneName, Global.lane)));
        Set Global Variable(tStart, Total Time Elapsed);
        While(Compare(Global.outcome, ==, 0));
            Wait(0.1, Ignore Condition);
            "The stand-in walks the lane: head for the centre of the pad its lane says comes next."
            Set Global Variable(tgtPos, Vector(Add(Add(-12, Multiply(2, Modulo(Add(Global.curCell, Value In Array(Array(1, -1, 12, -12), If-Then-Else(Compare(Global.lane, ==, 0), If-Then-Else(Compare(Round To Integer(Divide(Global.curCell, 12), Down), >, 0), 3, 0), If-Then-Else(Compare(Modulo(Global.curCell, 12), ==, 11), 3, If-Then-Else(Compare(Round To Integer(Divide(Global.curCell, 12), Down), <, 1), 2, If-Then-Else(Compare(Round To Integer(Divide(Global.curCell, 12), Down), >, 1), 3, 0)))))), 12))), 1), 0, Add(Add(-19.5, Multiply(2, Round To Integer(Divide(Add(Global.curCell, Value In Array(Array(1, -1, 12, -12), If-Then-Else(Compare(Global.lane, ==, 0), If-Then-Else(Compare(Round To Integer(Divide(Global.curCell, 12), Down), >, 0), 3, 0), If-Then-Else(Compare(Modulo(Global.curCell, 12), ==, 11), 3, If-Then-Else(Compare(Round To Integer(Divide(Global.curCell, 12), Down), <, 1), 2, If-Then-Else(Compare(Round To Integer(Divide(Global.curCell, 12), Down), >, 1), 3, 0)))))), 12), Down))), 1)));
            Start Throttle In Direction(Global.learner, Normalize(Vector(Subtract(X Component Of(Global.tgtPos), X Component Of(Position Of(Global.learner))), 0, Subtract(Z Component Of(Global.tgtPos), Z Component Of(Position Of(Global.learner))))), 1, To World, Replace existing throttle, None);
            If((Global.learner).fell);
                "Walked off the island without a boop."
                Set Global Variable(outcome, 2);
                Modify Global Variable(ret, Add, -100);
                Set Global Variable(realFall, 1);
                Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,5,selffall,{0},{1},{2}", Global.crossing, Global.steps, Global.curCell));
            Else If(Or(Compare(Subtract(Total Time Elapsed, Global.tStart), >, 40), Compare(Global.steps, >=, 40)));
                Set Global Variable(outcome, 3);
            Else;
                Set Global Variable(seenCell, Add(Multiply(Min(Max(Round To Integer(Divide(Subtract(Z Component Of(Position Of(Global.learner)), -19.5), 2), Down), 0), 3), 12), Min(Max(Round To Integer(Divide(Subtract(X Component Of(Position Of(Global.learner)), -12), 2), Down), 0), 11)));
                If(Compare(Global.seenCell, !=, Global.curCell));
                    "One step. Draw first, log the draw, then use it."
                    Set Global Variable(roll, Random Integer(1, 10));
                    Set Global Variable(mv, Random Integer(0, 3));
                    Set Global Variable(stepR, -1);
                    Set Global Variable(newCell, Global.seenCell);
                    If(Compare(Round To Integer(Divide(Global.curCell, 12), Down), ==, 0));
                        Modify Global Variable(edgeSteps, Add, 1);
                    End;
                    If(And(Global.boopsOn, Compare(Global.roll, ==, 1)));
                        Modify Global Variable(boops, Add, 1);
                        Set Global Variable(r2, Add(Round To Integer(Divide(Global.curCell, 12), Down), Value In Array(Array(0, 0, 1, -1), Global.mv)));
                        Set Global Variable(c2, Add(Modulo(Global.curCell, 12), Value In Array(Array(1, -1, 0, 0), Global.mv)));
                        If(Compare(Global.r2, <, 0));
                            Set Global Variable(newCell, -1);
                        Else If(Or(Or(Compare(Global.c2, <, 0), Compare(Global.c2, >, 11)), Compare(Global.r2, >, 3)));
                            Set Global Variable(newCell, Global.curCell);
                        Else;
                            Set Global Variable(newCell, Add(Multiply(Global.r2, 12), Global.c2));
                        End;
                        Play Effect(All Players(All Teams), Ring Explosion, Color(Green), Position Of(Global.learner), 3);
                        Small Message(All Players(All Teams), Custom String("Boop. Lucio sends you one pad {0}.", Value In Array(Global.dirName, Global.mv)));
                        If(Compare(Global.newCell, <, 0));
                            Teleport(Global.learner, Vector(Add(Add(-12, Multiply(2, Modulo(Global.curCell, 12))), 1), 0.5, -21.5));
                            Apply Impulse(Global.learner, Backward, 6, To World, Cancel Contrary Motion XYZ);
                        Else;
                            Teleport(Global.learner, Vector(Add(Add(-12, Multiply(2, Modulo(Global.newCell, 12))), 1), 0, Add(Add(-19.5, Multiply(2, Round To Integer(Divide(Global.newCell, 12), Down))), 1)));
                        End;
                    End;
                    If(Compare(Global.newCell, <, 0));
                        Modify Global Variable(stepR, Add, -100);
                        Set Global Variable(outcome, 2);
                    Else If(Compare(Global.newCell, ==, 11));
                        Modify Global Variable(stepR, Add, 50);
                        Set Global Variable(outcome, 1);
                    End;
                    Modify Global Variable(steps, Add, 1);
                    Modify Global Variable(ret, Add, Global.stepR);
                    Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,5,step,{0},{1},{2}", Global.crossing, Global.steps, Custom String("{0},{1},{2}", Global.curCell, Global.seenCell, Custom String("{0},{1},{2}", Global.roll, Global.mv, Custom String("{0},{1},{2}", Global.newCell, Global.stepR, If-Then-Else(Global.boopsOn, 1, 0))))));
                    If(Compare(Global.newCell, >=, 0));
                        Set Global Variable(curCell, Global.newCell);
                    End;
                    Set Global Variable(hudRun, Custom String("Now: {0} lane, {1} steps, {2}", Value In Array(Global.laneName, Global.lane), Global.steps, Custom String("{0} boops, return {1}", Global.boops, Global.ret)));
                End;
            End;
        End;
        Stop Throttle In Direction(Global.learner);
        If(And(Compare(Global.outcome, ==, 2), Compare(Global.realFall, ==, 0)));
            Wait Until(Compare((Global.learner).fell, ==, True), 3);
            Set Global Variable(realFall, If-Then-Else((Global.learner).fell, 1, 0));
        End;
        "cross: crossing, lane, boops on, steps, boops, outcome (1 goal, 2 fell, 3 time up), return, steps from edge pads,\nfell for real in the game (1/0)"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,5,cross,{0},{1},{2}", Global.crossing, Global.lane, Custom String("{0},{1},{2}", If-Then-Else(Global.boopsOn, 1, 0), Global.steps, Custom String("{0},{1},{2}", Global.boops, Global.outcome, Custom String("{0},{1},{2}", Global.ret, Global.edgeSteps, Global.realFall)))));
        Set Global Variable At Index(hudRes, Subtract(Global.crossing, 1), Custom String("{0}. {1}: {2}", Global.crossing, Value In Array(Global.laneName, Global.lane), Custom String("{0} steps, {1} boops, {2}", Global.steps, Global.boops, Custom String("{0}, return {1}", Value In Array(Global.outName, Global.outcome), Global.ret))));
        Set Global Variable(hudRun, Custom String(""));
        If(Compare(Global.outcome, ==, 2));
            Wait(0.4, Ignore Condition);
            Call Subroutine(putHome);
        End;
    }
}
