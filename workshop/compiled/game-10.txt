rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "The first seconds of a match are their own thing: the engine starts the map and every rule sets up at once. That\nburst is kept apart from the load during play (the peak reading looks back two seconds, and each game's own setup starts at about 5 s)."
        If(Not(Global.gameOver));
            If(Compare(Total Time Elapsed, <, 10));
                Modify Global Variable(loadPeakStart, Max, Server Load Peak);
            Else If(Global.ready);
                Modify Global Variable(loadSamples, Add, 1);
                Modify Global Variable(loadSumRun, Add, Server Load Average);
                Modify Global Variable(loadPeakRun, Max, Server Load Peak);
            End;
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        "2026-11-22: 4 seconds, not 1.5. Every game starts its own setup when `ready` turns true; at 1.5 s that landed on\ntop of the engine's own start-of-match work and the server load read 125 to 139 for a moment."
        Wait(4, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        Set Global Variable(learner, Host Player);
        Teleport(Global.learner, Vector(0, 0, -12));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Forward, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,10,1,0"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(0, 0, -12));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Forward, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: the learner fell off the island") {

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,10,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,10,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        "The moment the question is on screen, so the log shows how long the prediction took (and so a script that plays\nthe human build knows when to press, as a person does by looking)."
        Set Global Variable(qShown, Total Time Elapsed);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,Q,10,{0}", Global.qId));
        Wait Until(Or(Compare(Global.qAnswer, >=, 0), Global.gameOver), 90);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,10,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1},{2}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0), Subtract(Total Time Elapsed, Global.qShown))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(5, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

rule ("Game 10: her odds of aiming left this peek") {
    event {
        Subroutine;
        oddsNow;
    }
    actions {
        If(Compare(Global.phase, ==, 1));
            "The fixed Widowmaker: she always holds right."
            Set Global Variable(herBR, 0);
            Set Global Variable(herQ, 0);
        Else;
            "Half the time she answers his habit (aim left once he peeks left more than 2 times in 7), half the time she\nrepeats her own habit (her share of left aims so far). Integer test, so the log can recompute it exactly."
            Set Global Variable(herBR, If-Then-Else(Compare(Multiply(7, Global.nL), >, Multiply(2, Global.nN)), 1, 0));
            Set Global Variable(herQ, Divide(Add(Multiply(Global.herBR, Global.aA), Global.aL), Multiply(2, Global.aA)));
        End;
    }
}

rule ("Game 10: rebuild the HUD strings (only when something changes)") {
    event {
        Subroutine;
        redraw;
    }
    actions {
        Set Global Variable(barHer, Multiply(100, Global.herQ));
        Set Global Variable(barHim, Divide(Multiply(100, Global.nL), Max(Global.nN, 1)));
        "Bar labels stay under 44 characters, or the game wraps them under the bar onto the next row."
        Set Global Variable(barHerTxt, Custom String("her LEFT odds now: {0}%{1}", Round To Integer(Multiply(100, Global.herQ), To Nearest), If-Then-Else(Compare(Global.phase, >=, 3), Custom String("  (safe mix 57%)"), Custom String(""))));
        Set Global Variable(barHimTxt, Custom String("you went LEFT {0}/{1}: {2}", Global.nL, Global.nN, Custom String("{0}%{1}", Round To Integer(Global.barHim, To Nearest), If-Then-Else(Compare(Global.phase, >=, 3), Custom String("  (safe mix 29%)"), Custom String("")))));
        Set Global Variable(scoreTxt, Custom String("your points {0}    hers {1}    peek {2} of 14", Global.score, Subtract(0, Global.score), Global.peekN));
        If(Compare(Global.phase, ==, 1));
            Set Global Variable(infoTxt, Custom String("This Widowmaker never changes: she always holds RIGHT."));
        Else If(Compare(Global.phase, ==, 2));
            Set Global Variable(infoTxt, Custom String("She aims at your habit half the time, else at her own."));
        Else;
            Set Global Variable(infoTxt, Custom String("The die says {0}. It says LEFT 2 times in 7.", If-Then-Else(Compare(Global.dieSays, ==, 1), Custom String("LEFT"), Custom String("RIGHT"))));
        End;
    }
}

rule ("Game 10: one peek. Both choose at once, then the result shows in the world") {
    event {
        Subroutine;
        peek;
    }
    actions {
        Call Subroutine(putHome);
        Set Facing(Global.learner, Forward, To World);
        Set Facing(Global.widow, Direction Towards(Global.widowPos, Vector(0, 0, -2)), To World);
        Modify Global Variable(peekN, Add, 1);
        Modify Global Variable(phasePeeks, Add, 1);
        Call Subroutine(oddsNow);
        Set Global Variable(dieRoll, -1);
        Set Global Variable(dieSays, -1);
        If(Compare(Global.phase, ==, 3));
            Set Global Variable(dieRoll, Random Integer(0, 6));
            Set Global Variable(dieSays, If-Then-Else(Compare(Global.dieRoll, <, 2), 1, 0));
        End;
        Call Subroutine(redraw);
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Wait Until(Not(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire)))), 3);
        Set Global Variable(pick, -1);
        Set Global Variable(tmo, 0);
        Set Global Variable(pickTxt, Custom String("PEEK {0} OF 14: LEFT CLICK left, RIGHT CLICK right", Global.peekN));
        Set Global Variable(pickOpen, True);
        Wait Until(Or(Compare(Global.pick, >=, 0), Global.gameOver), 120);
        Set Global Variable(pickOpen, False);
        If(Compare(Global.pick, <, 0));
            Set Global Variable(pick, 0);
            Set Global Variable(tmo, 1);
        End;
        If(Compare(Global.phase, ==, 1));
            Set Global Variable(drawU, -1);
            Set Global Variable(aim, 0);
        Else;
            Set Global Variable(drawU, Random Integer(0, 99));
            Set Global Variable(aim, If-Then-Else(Compare(Multiply(Multiply(Global.drawU, 2), Global.aA), <, Multiply(100, Add(Multiply(Global.herBR, Global.aA), Global.aL))), 1, 0));
        End;
        Set Global Variable(pay, Value In Array(Global.payTab, Add(Multiply(Global.pick, 2), Global.aim)));
        Set Global Variable(ev, Add(Multiply(Global.herQ, Value In Array(Global.payTab, Add(Multiply(Global.pick, 2), 1))), Multiply(Subtract(1, Global.herQ), Value In Array(Global.payTab, Multiply(Global.pick, 2)))));
        "peek: n, phase, pick, his lefts so far, his peeks so far, her best answer, her lefts, her aims, her odds, draw,\naim, his points, his expected points against her odds, die roll, what the die said, timed out"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,10,peek,{0},{1},{2}", Global.peekN, Global.phase, Custom String("{0},{1},{2}", Global.pick, Global.nL, Custom String("{0},{1},{2}", Global.nN, Global.herBR, Custom String("{0},{1},{2}", Global.aL, Global.aA, Custom String("{0},{1},{2}", Global.herQ, Global.drawU, Custom String("{0},{1},{2}", Global.aim, Global.pay, Custom String("{0},{1},{2}", Global.ev, Global.dieRoll, Custom String("{0},{1}", Global.dieSays, Global.tmo)))))))));
        Modify Global Variable(score, Add, Global.pay);
        Modify Global Variable(evSum, Add, Global.ev);
        Modify Global Variable(phasePay, Add, Global.pay);
        If(And(Compare(Global.phase, ==, 3), Compare(Global.pick, ==, Global.dieSays)));
            Modify Global Variable(followed, Add, 1);
        End;
        Modify Global Variable(nL, Add, Global.pick);
        Modify Global Variable(nN, Add, 1);
        If(Compare(Global.phase, >=, 2));
            Modify Global Variable(aL, Add, Global.aim);
            Modify Global Variable(aA, Add, 1);
        End;
        "The result, in the world: he steps out at his corner, she turns her scope to hers."
        Teleport(Global.learner, If-Then-Else(Compare(Global.pick, ==, 1), Global.leftPos, Global.rightPos));
        Wait(0.15, Ignore Condition);
        Set Facing(Global.learner, Direction Towards(Eye Position(Global.learner), Eye Position(Global.widow)), To World);
        Set Facing(Global.widow, Direction Towards(Eye Position(Global.widow), Add(If-Then-Else(Compare(Global.aim, ==, 1), Global.leftPos, Global.rightPos), Vector(0, 1.4, 0))), To World);
        Start Holding Button(Global.widow, Button(Secondary Fire));
        If(Compare(Global.pick, ==, Global.aim));
            Create Beam Effect(All Players(All Teams), Bad Beam, Eye Position(Global.widow), Eye Position(Global.learner), Color(Red), None);
            Set Global Variable(beam, Last Created Entity);
            Play Effect(All Players(All Teams), Bad Explosion, Color(Red), Position Of(Global.learner), 1.5);
            Big Message(All Players(All Teams), Custom String("CAUGHT. She was holding {0}: {1} for you", If-Then-Else(Compare(Global.aim, ==, 1), Custom String("LEFT"), Custom String("RIGHT")), Global.pay));
        Else;
            Play Effect(All Players(All Teams), Good Explosion, Color(Green), Position Of(Global.learner), 1.5);
            Big Message(All Players(All Teams), Custom String("CLEAN PEEK. She was holding {0}: +{1} for you", If-Then-Else(Compare(Global.aim, ==, 1), Custom String("LEFT"), Custom String("RIGHT")), Global.pay));
        End;
        Set Global Variable(coach, Custom String("Peek {0}: you went {1}, she held {2}", Global.peekN, If-Then-Else(Compare(Global.pick, ==, 1), Custom String("LEFT"), Custom String("RIGHT")), Custom String("{0}. {1} points, total {2}.", If-Then-Else(Compare(Global.aim, ==, 1), Custom String("LEFT"), Custom String("RIGHT")), Global.pay, Global.score)));
        If(Compare(Global.phase, ==, 2));
            Set Global Variable(coach, Custom String("Peek {0}: {1} points. Against her odds, that peek was worth {2} on average.", Global.peekN, Global.pay, Global.ev));
        End;
        Call Subroutine(redraw);
        Wait(3, Ignore Condition);
        Stop Holding Button(Global.widow, Button(Secondary Fire));
        If(Compare(Global.beam, !=, Null));
            Destroy Effect(Global.beam);
            Set Global Variable(beam, Null);
        End;
    }
}

rule ("Game 10: two learners play 300 peeks (fictitious play: each answers the other's habit counts)") {
    event {
        Subroutine;
        selfPlay;
    }
    actions {
        Set Global Variable(spNL, 1);
        Set Global Variable(spN, 2);
        Set Global Variable(spML, 1);
        Set Global Variable(spM, 2);
        Set Global Variable(spLast, -1);
        Set Global Variable(spFlips, 0);
        Set Global Variable(spPay, 0);
        Set Global Variable(spSnap, Empty Array);
        For Global Variable(spT, 1, 301, 1);
            "Tracer: peek left when her left share is under 4 in 7. Widowmaker: aim left when his is over 2 in 7."
            Set Global Variable(spTr, If-Then-Else(Compare(Multiply(7, Global.spML), <, Multiply(4, Global.spM)), 1, 0));
            Set Global Variable(spWi, If-Then-Else(Compare(Multiply(7, Global.spNL), >, Multiply(2, Global.spN)), 1, 0));
            If(And(Compare(Global.spLast, >=, 0), Compare(Global.spTr, !=, Global.spLast)));
                Modify Global Variable(spFlips, Add, 1);
            End;
            Set Global Variable(spLast, Global.spTr);
            Modify Global Variable(spPay, Add, Value In Array(Global.payTab, Add(Multiply(Global.spTr, 2), Global.spWi)));
            Modify Global Variable(spNL, Add, Global.spTr);
            Modify Global Variable(spN, Add, 1);
            Modify Global Variable(spML, Add, Global.spWi);
