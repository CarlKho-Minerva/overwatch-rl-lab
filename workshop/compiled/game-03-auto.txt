rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        "The automatic check: a bot plays the learner's part. The host stands aside, out of the way and unseen."
        Create Dummy Bot(Hero(Tracer), Team 1, -1, Vector(-6, 0, -14), Left);
        Wait(1, Ignore Condition);
        Set Global Variable(learner, First Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
        Set Status(Host Player, Null, Phased Out, 99999);
        Set Invisible(Host Player, All);
        Teleport(Host Player, Vector(0, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Forward, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,3,1,1"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(-6, 0, -14));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Forward, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,3,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,3,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        Wait(1.2, Ignore Condition);
        Set Global Variable(qAnswer, Global.qAuto);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,3,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(0.6, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

rule ("Game 3: the rules (the model). Pad npP, move npA -> landing pad npS, reward npR") {
    event {
        Subroutine;
        nextPad;
    }
    actions {
        Set Global Variable(npRow, Add(Round To Integer(Divide(Global.npP, 5), Down), Value In Array(Array(0, 1, 0, -1), Global.npA)));
        Set Global Variable(npCol, Add(Modulo(Global.npP, 5), Value In Array(Array(1, 0, -1, 0), Global.npA)));
        Set Global Variable(npS, If-Then-Else(Or(Or(Or(Compare(Global.npRow, <, 0), Compare(Global.npRow, >, 4)), Compare(Global.npCol, <, 0)), Compare(Global.npCol, >, 4)), Global.npP, Add(Multiply(Global.npRow, 5), Global.npCol)));
        Set Global Variable(npR, Subtract(Add(-1, If-Then-Else(Compare(Global.npS, ==, Global.goal), 10, 0)), If-Then-Else(Array Contains(Global.pits, Global.npS), 10, 0)));
    }
}

rule ("Game 3: one-step look ahead from pad lp: qs[move] = reward + old number of the landing pad") {
    event {
        Subroutine;
        lookAhead;
    }
    actions {
        For Global Variable(la, 0, 4, 1);
            Set Global Variable(npP, Global.lp);
            Set Global Variable(npA, Global.la);
            Call Subroutine(nextPad);
            Set Global Variable At Index(qs, Global.la, Add(Global.npR, Value In Array(Global.V, Global.npS)));
        End;
        Set Global Variable(qBest, Last Of(Sorted Array(Global.qs, Current Array Element)));
    }
}

rule ("Game 3: one synchronous sweep of value iteration (the new table is built from the old one)") {
    event {
        Subroutine;
        sweepOnce;
    }
    actions {
        Set Global Variable(Vnew, Global.V);
        Set Global Variable(nChg, 0);
        Set Global Variable(maxChg, 0);
        For Global Variable(wp, 0, 25, 1);
            Set Global Variable At Index(chg, Global.wp, False);
            Set Global Variable At Index(newPos, Global.wp, False);
            If(And(Compare(Global.wp, !=, Global.goal), Not(Array Contains(Global.pits, Global.wp))));
                Set Global Variable(lp, Global.wp);
                Call Subroutine(lookAhead);
                Set Global Variable At Index(Vnew, Global.wp, Global.qBest);
                Modify Global Variable(looks, Add, 4);
                If(Compare(Absolute Value(Subtract(Global.qBest, Value In Array(Global.V, Global.wp))), >, 0.001));
                    Set Global Variable At Index(chg, Global.wp, True);
                    Modify Global Variable(nChg, Add, 1);
                    Modify Global Variable(maxChg, Max, Absolute Value(Subtract(Global.qBest, Value In Array(Global.V, Global.wp))));
                End;
                Set Global Variable At Index(newPos, Global.wp, And(Compare(Global.qBest, >, 0), Compare(Value In Array(Global.V, Global.wp), <=, 0)));
            End;
            If(Compare(Modulo(Global.wp, 5), ==, 4));
                Wait(0.016, Ignore Condition);
            End;
        End;
        Set Global Variable(V, Global.Vnew);
        Modify Global Variable(sweepN, Add, 1);
    }
}

rule ("Game 3: wipe the whiteboard (every number back to 0)") {
    event {

            Set Global Variable At Index(lastCh, Global.sp, 0);
            Set Global Variable At Index(flip, Global.sp, False);
        End;

}

rule ("Game 3: plan silently from a clean board until a sweep changes nothing (for the answer key)") {
    event {
        Subroutine;
        settle;
    }
    actions {
        Call Subroutine(clearBoard);
        While(And(Compare(Global.nChg, >, 0), Compare(Global.sweepN, <, 30)));
            Call Subroutine(sweepOnce);
            For Global Variable(sp, 0, 25, 1);
                If(Value In Array(Global.chg, Global.sp));
                    Set Global Variable At Index(lastCh, Global.sp, Global.sweepN);
                End;
            End;
        End;
    }
}

rule ("Game 3: the plan: on each pad, the move with the best look ahead (keep the old line if it is still best)") {
    event {
        Subroutine;
        makeArrows;
    }
    actions {
        Set Global Variable(wrongN, 0);
        For Global Variable(mp, 0, 25, 1);
            Set Global Variable At Index(flip, Global.mp, False);
            If(Or(Compare(Global.mp, ==, Global.goal), Array Contains(Global.pits, Global.mp)));
                Set Global Variable At Index(arrow, Global.mp, -1);
            Else;
                Set Global Variable(lp, Global.mp);
                Call Subroutine(lookAhead);
                If(And(And(Global.keepOld, Compare(Value In Array(Global.arrowOld, Global.mp), >=, 0)), Compare(Value In Array(Global.qs, Value In Array(Global.arrowOld, Global.mp)), >=, Subtract(Global.qBest, 0.001))));
                    Set Global Variable At Index(arrow, Global.mp, Value In Array(Global.arrowOld, Global.mp));
                Else;
                    Set Global Variable At Index(arrow, Global.mp, Index Of Array Value(Global.qs, Global.qBest));
                End;
                If(And(And(Global.keepOld, Compare(Value In Array(Global.arrowOld, Global.mp), >=, 0)), Compare(Value In Array(Global.qs, Value In Array(Global.arrowOld, Global.mp)), <, Subtract(Global.qBest, 0.001))));
                    Modify Global Variable(wrongN, Add, 1);
                    Set Global Variable At Index(flip, Global.mp, True);
                End;
            End;
            If(Compare(Modulo(Global.mp, 5), ==, 4));
                Wait(0.016, Ignore Condition);
            End;
        End;
    }
}

rule ("Game 3: what following the lines from the start earns, by the rules") {
    event {
        Subroutine;
        simWalk;
    }
    actions {
        Set Global Variable(curP, 2);
        Set Global Variable(walkSim, 0);
        For Global Variable(wk, 0, 40, 1);
            If(Or(Compare(Global.curP, ==, Global.goal), Array Contains(Global.pits, Global.curP)));
                Break;
            End;
            Set Global Variable(npP, Global.curP);
            Set Global Variable(npA, Value In Array(Global.arrow, Global.curP));

            Modify Global Variable(walkSim, Add, Global.npR);
            Set Global Variable(curP, Global.npS);
        End;
    }
}

rule ("Game 3: redraw the whiteboard: rings, one number per pad, and the plan's lines") {
    event {
        Subroutine;
        drawBoard;
    }
    actions {
        "Only this board's own texts: the core's floating question is an in-world text too, and destroying every text\nwould remove it for the rest of the game."
        For Global Variable(dp, 0, Count Of(Global.padTxt), 1);
            Destroy In-World Text(Value In Array(Global.padTxt, Global.dp));
        End;
        Set Global Variable(padTxt, Empty Array);
        Destroy All Effects;
        Wait(0.016, Ignore Condition);
        Create Effect(All Players(All Teams), Ring, Color(Sky Blue), Value In Array(Global.padPos, 2), 1, None);
        Create Effect(All Players(All Teams), Ring, Color(Green), Value In Array(Global.padPos, Global.goal), 1.8, None);
        Create Effect(All Players(All Teams), Ring, Color(Red), Value In Array(Global.padPos, First Of(Global.pits)), 1.8, None);
        Create Effect(All Players(All Teams), Ring, Color(Red), Value In Array(Global.padPos, Value In Array(Global.pits, 1)), 1.8, None);
        For Global Variable(dp, 0, 25, 1);
            Create In-World Text(All Players(All Teams), If-Then-Else(Compare(Global.dp, ==, Global.goal), Custom String("GOAL +10"), If-Then-Else(Array Contains(Global.pits, Global.dp), If-Then-Else(And(Global.markPit, Compare(Global.dp, ==, Value In Array(Global.pits, 1))), Custom String("THIS PIT MOVES"), Custom String("PIT -10")), Custom String("{0}", Round To Integer(Value In Array(Global.V, Global.dp), To Nearest)))), Add(Value In Array(Global.padPos, Global.dp), Up), 2.5, Do Not Clip, None, If-Then-Else(Compare(Global.dp, ==, Global.goal), Color(Green), If-Then-Else(Array Contains(Global.pits, Global.dp), Custom Color(255, 112, 122, 255), If-Then-Else(Value In Array(Global.newPos, Global.dp), Color(Yellow), If-Then-Else(Compare(Value In Array(Global.V, Global.dp), >, 0), Color(Lime Green), Color(White))))), Default Visibility);
            Modify Global Variable(padTxt, Append To Array, Last Text ID);
            If(And(Global.showArrows, Compare(Value In Array(Global.arrow, Global.dp), >=, 0)));
                Create Beam Effect(All Players(All Teams), Grapple Beam, Add(Value In Array(Global.padPos, Global.dp), Vector(0, 0.3, 0)), Add(Add(Value In Array(Global.padPos, Global.dp), Multiply(Value In Array(Global.dirv, Value In Array(Global.arrow, Global.dp)), 2)), Vector(0, 0.3, 0)), If-Then-Else(Value In Array(Global.flip, Global.dp), Color(Yellow), Color(Green)), None);
            End;
            If(Compare(Modulo(Global.dp, 5), ==, 4));
                Wait(0.016, Ignore Condition);
            End;
        End;
        Set Global Variable(boardTxt, Custom String("sweep {0}   pads changed {1}   biggest change {2}", Global.sweepN, If-Then-Else(Compare(Global.sweepN, >, 0), Global.nChg, 0), If-Then-Else(Compare(Global.sweepN, >, 0), Round To Integer(Global.maxChg, To Nearest), 0)));

rule ("Game 3: run") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.ready != False;
    }
    actions {
        Disallow Button(Global.learner, Button(Ability 1));
        Disallow Button(Global.learner, Button(Ability 2));
        Disallow Button(Global.learner, Button(Ultimate));
        For Global Variable(sp, 0, 25, 1);
            Set Global Variable At Index(padPos, Global.sp, Vector(Add(-14, Multiply(4, Modulo(Global.sp, 5))), 0, Add(-14, Multiply(4, Round To Integer(Divide(Global.sp, 5), Down)))));
            Set Global Variable At Index(arrow, Global.sp, -1);
            Set Global Variable At Index(arrowOld, Global.sp, -1);
        End;
        Set Facing(Global.learner, Forward, To World);
        "map: start, goal, the fixed pit, the pit that will move"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,3,map,2,{0},{1},{2}", Global.goal, First Of(Global.pits), Value In Array(Global.pits, 1)));
        "---- Self-tests: the planner checks its own arithmetic on this map, silently, before the round ----"
        Call Subroutine(clearBoard);
        Set Global Variable(looks, 0);
        Call Subroutine(sweepOnce);
        Set Global Variable(chkName, Custom String("one sweep: pad beside the goal"));
        Set Global Variable(chkExp, 9);
        Set Global Variable(chkGot, Value In Array(Global.V, 22));
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("one sweep: start pad"));
        Set Global Variable(chkExp, -1);
        Set Global Variable(chkGot, Value In Array(Global.V, 2));
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("look-aheads in one sweep (22 pads x 4 moves)"));
        Set Global Variable(chkExp, 88);
        Set Global Variable(chkGot, Global.looks);
        Call Subroutine(check);
        Call Subroutine(settle);
        Set Global Variable(nSweeps, Global.sweepN);
        Set Global Variable(dStart, Value In Array(Global.lastCh, 2));
        Set Global Variable(dMax, Last Of(Sorted Array(Global.lastCh, Current Array Element)));
        Set Global Variable(vStart, Value In Array(Global.V, 2));
        Set Global Variable(chkName, Custom String("sweeps until a sweep changes nothing"));
        Set Global Variable(chkExp, 8);
        Set Global Variable(chkGot, Global.nSweeps);
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("start pad's settled number"));
        Set Global Variable(chkExp, 5);
        Set Global Variable(chkGot, Global.vStart);
        Call Subroutine(check);
        Set Global Variable(keepOld, False);
        Call Subroutine(makeArrows);
        Call Subroutine(simWalk);
        Set Global Variable(chkName, Custom String("walking the lines earns the start pad's number"));
        Set Global Variable(chkExp, Global.vStart);
        Set Global Variable(chkGot, Global.walkSim);
        Call Subroutine(check);
        Set Global Variable(arrowOld, Global.arrow);
        Set Global Variable At Index(pits, 1, 7);
        Call Subroutine(settle);
        Set Global Variable(keepOld, True);
        Call Subroutine(makeArrows);
        Set Global Variable(chkName, Custom String("old lines made wrong by moving the pit to pad 7"));
        Set Global Variable(chkExp, 5);
        Set Global Variable(chkGot, Global.wrongN);
        Call Subroutine(check);
        "pre: sweeps until nothing changes, sweeps until the start pad hears the goal, the same for the farthest pad,\nthe start pad's settled number, what walking the lines earns by the rules"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,3,pre,{0},{1},{2}", Global.nSweeps, Global.dStart, Custom String("{0},{1},{2}", Global.dMax, Global.vStart, Global.walkSim)));
        Set Global Variable At Index(pits, 1, 13);
        Set Global Variable(keepOld, False);
        For Global Variable(sp, 0, 25, 1);
            Set Global Variable At Index(arrow, Global.sp, -1);
            Set Global Variable At Index(arrowOld, Global.sp, -1);
        End;
        Call Subroutine(clearBoard);
        Set Global Variable(looks, 0);
        Set Global Variable(looks0, 0);
        Set Global Variable(showArrows, False);
        Call Subroutine(drawBoard);
        "---- Round 1: the rules, and the first sweep ----"
        Set Global Variable(ctrl, Custom String("Answer: LEFT CLICK / RIGHT CLICK / MELEE.   F: one sweep.   WASD: walk."));
        Set Global Variable(coach, Custom String("Whiteboard before the round: you know this map's rules, so plan the dive instead of dying to find out."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(coach, Custom String("Each step costs 1. The goal pays 10. A red pit costs 10 and ends the dive. Every pad starts at 0."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 1);
        Set Global Variable(qText, Custom String("After ONE sweep, the pad next to the goal shows?"));
        Set Global Variable(qOpts, Array(Custom String("9"), Custom String("10"), Custom String("-1")));
        Set Global Variable(qCorrect, 0);
        Set Global Variable(qAuto, 0);
        Set Global Variable(qWhy, Custom String("-1 for the step, +10 for the goal, plus the goal's 0: lecture 2's one-step rule."));
        Call Subroutine(ask);
        Set Global Variable(runId, 1);
        Set Global Variable(coach, Custom String("Press F: each pad looks at its 4 moves (step reward + the number it lands on) and keeps the best."));
        While(Compare(Global.sweepN, <, 30));
            Call Subroutine(waitF);
            Call Subroutine(sweepOnce);
            Call Subroutine(drawBoard);
            Call Subroutine(logBoard);
            Set Global Variable(coach, Custom String("Sweep {0}: {1} pads changed. Yellow pads just heard about the goal. Press F again.", Global.sweepN, Global.nChg));
            If(Compare(Global.nChg, ==, 0));
                Break;
            End;
            If(Compare(Global.sweepN, ==, 1));
                Wait(0.4, Ignore Condition);
                Set Global Variable(qId, 2);
                Set Global Variable(qText, Custom String("Your start pad shows -1. Which sweep turns it positive?"));
                Set Global Variable(qOpts, Array(Custom String("sweep 2"), Custom String("sweep {0}", Global.dStart), Custom String("sweep {0}", Multiply(2, Global.dStart))));
                Set Global Variable(qCorrect, 1);
                Set Global Variable(qAuto, 1);
                Set Global Variable(qWhy, Custom String("Good news moves one pad per sweep, and your pad is {0} steps from the goal.", Global.dStart));
                Call Subroutine(ask);
                Set Global Variable(coach, Custom String("Press F for sweep 2."));
            Else If(Compare(Global.sweepN, ==, 2));
                Wait(0.4, Ignore Condition);
                Set Global Variable(qId, 3);
                Set Global Variable(qText, Custom String("Does ONE plan score exactly these numbers?"));
                Set Global Variable(qOpts, Array(Custom String("the best plan"), Custom String("a worse plan"), Custom String("no plan")));
                Set Global Variable(qCorrect, 2);
                Set Global Variable(qAuto, 0);
                Set Global Variable(qWhy, Custom String("A -2 pad ringed by -2 pads scores -3 under any plan. Midway numbers are half-done sums."));
                Call Subroutine(ask);
                Set Global Variable(coach, Custom String("Press F for sweep 3."));
            Else If(Compare(Global.sweepN, ==, 3));
                Wait(0.4, Ignore Condition);
                Set Global Variable(qId, 4);
                Set Global Variable(qText, Custom String("Which sweep is the first to change nothing?"));
                Set Global Variable(qOpts, Array(Custom String("sweep {0}", Global.dMax), Custom String("sweep {0}", Add(Global.dMax, 1)), Custom String("never")));
                Set Global Variable(qCorrect, 1);
                Set Global Variable(qAuto, 1);
                Set Global Variable(qWhy, Custom String("The farthest pad settles on sweep {0}. Sweep {1} changes nothing: that is how you know.", Global.dMax, Add(Global.dMax, 1)));
                Call Subroutine(ask);
                Set Global Variable(coach, Custom String("Keep pressing F until a sweep changes nothing."));
            End;
        End;
        "---- Round 2: the plan appears ----"
        Set Global Variable(nSweeps, Global.sweepN);
        Set Global Variable(keepOld, False);
        Call Subroutine(makeArrows);
        Set Global Variable(showArrows, True);
        Call Subroutine(drawBoard);
        Call Subroutine(logArrows);
        "looks: sweeps, pads that get a number, look-aheads in all"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,3,looks,{0},{1},{2}", Global.nSweeps, Subtract(24, Count Of(Global.pits)), Global.looks));
        Set Global Variable(coach, Custom String("Nothing changed: plan done. Green line = the step to the best number next door (policy improvement)."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(coach, Custom String("{0} sweeps x {1} pads x 4 moves = {2} look-aheads. A real match has far too many states to sweep.", Global.nSweeps, Subtract(24, Count Of(Global.pits)), Global.looks));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 5);
        Set Global Variable(qText, Custom String("Start pad says {0}. Walk the lines. Your score?", Round To Integer(Value In Array(Global.V, 2), To Nearest)));
        Set Global Variable(qOpts, Array(Custom String("10"), Custom String("{0}", Round To Integer(Value In Array(Global.V, 2), To Nearest)), Custom String("luck decides")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("Walk it and see."));
        Call Subroutine(ask);
        "---- Round 3: walk the plan ----"
        Call Subroutine(putHome);
        Set Global Variable(ctrl, Custom String("WASD: walk the green lines from the blue ring to the goal. Each pad costs 1."));
        Set Global Variable(coach, Custom String("Walk the plan. You can leave it if you like; the board counts every pad you step on."));
        Set Global Variable(walkRet, 0);
        Set Global Variable(walkSteps, 0);
        Set Global Variable(walkEnd, 0);
        Set Global Variable(walkRestarts, 0);
        Set Global Variable(prevP, 2);
        Set Global Variable(walkT0, Total Time Elapsed);
        Set Global Variable(lastMove, Total Time Elapsed);
        Set Global Variable(walking, True);
        While(Global.walking);
            If((Global.learner).fell);
                Call Subroutine(putHome);
                Set Global Variable(walkRet, 0);
                Set Global Variable(walkSteps, 0);
                Set Global Variable(prevP, 2);
                Modify Global Variable(walkRestarts, Add, 1);
                Set Global Variable(coach, Custom String("Off the island. Back to the start; the walk starts over."));
            End;
            If(Compare(Subtract(Total Time Elapsed, Global.lastMove), >, 0.6));
                Set Global Variable(npP, Global.prevP);
                Set Global Variable(npA, Value In Array(Global.arrow, Global.prevP));
                Call Subroutine(nextPad);
                Teleport(Global.learner, Value In Array(Global.padPos, Global.npS));
                Set Global Variable(lastMove, Total Time Elapsed);
            End;
            Set Global Variable(curP, Add(Min(Max(Round To Integer(Divide(Add(X Component Of(Position Of(Global.learner)), 16), 4), Down), 0), 4), Multiply(5, Min(Max(Round To Integer(Divide(Add(Z Component Of(Position Of(Global.learner)), 16), 4), Down), 0), 4))));
            If(Compare(Global.curP, !=, Global.prevP));
                Modify Global Variable(walkSteps, Add, Add(Absolute Value(Subtract(Round To Integer(Divide(Global.curP, 5), Down), Round To Integer(Divide(Global.prevP, 5), Down))), Absolute Value(Subtract(Modulo(Global.curP, 5), Modulo(Global.prevP, 5)))));
                Modify Global Variable(walkRet, Add, Subtract(Add(Subtract(0, Add(Absolute Value(Subtract(Round To Integer(Divide(Global.curP, 5), Down), Round To Integer(Divide(Global.prevP, 5), Down))), Absolute Value(Subtract(Modulo(Global.curP, 5), Modulo(Global.prevP, 5))))), If-Then-Else(Compare(Global.curP, ==, Global.goal), 10, 0)), If-Then-Else(Array Contains(Global.pits, Global.curP), 10, 0)));
                Set Global Variable(prevP, Global.curP);
                If(Compare(Global.curP, ==, Global.goal));
                    Set Global Variable(walkEnd, 1);
                    Set Global Variable(walking, False);
                Else If(Array Contains(Global.pits, Global.curP));
                    Set Global Variable(walkEnd, 2);
                    Set Global Variable(walking, False);
                End;
            End;
            If(Compare(Subtract(Total Time Elapsed, Global.walkT0), >, 90));
                Set Global Variable(walking, False);
            End;
            Wait(0.05, Ignore Condition);
        End;
        "walk: pads entered, what he scored, the start pad's number, end (1 goal, 2 pit, 0 time ran out), restarts after a fall"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,3,walk,{0},{1},{2}", Global.walkSteps, Global.walkRet, Custom String("{0},{1},{2}", Value In Array(Global.V, 2), Global.walkEnd, Global.walkRestarts)));
        If(Compare(Global.walkEnd, ==, 1));
            Set Global Variable(coach, Custom String("You scored {0}. The start pad said {1}. Its number is what this plan earns: policy evaluation.", Round To Integer(Global.walkRet, To Nearest), Round To Integer(Value In Array(Global.V, 2), To Nearest)));
        Else If(Compare(Global.walkEnd, ==, 2));
            Set Global Variable(coach, Custom String("A pit: you scored {0}. Following the lines would have earned {1}, the start pad's number.", Round To Integer(Global.walkRet, To Nearest), Round To Integer(Value In Array(Global.V, 2), To Nearest)));
        Else;
            Set Global Variable(coach, Custom String("Time ran out. Following the lines earns {0}, the start pad's number.", Round To Integer(Value In Array(Global.V, 2), To Nearest)));
        End;
        Wait(0.4, Ignore Condition);
        "---- Round 4: the map changes ----"
        Set Global Variable(arrowOld, Global.arrow);
        Set Global Variable(Vsave, Global.V);
        Set Global Variable(pitFrom, Value In Array(Global.pits, 1));
        Set Global Variable(pitOk, False);
        Set Global Variable(pitTries, 0);
        "2026-11-22: this used to say \"the right-hand pit\", which was the left-hand one from the start pad and\nneither from where the walk ends. The board now labels the pit that moves."
        Set Global Variable(markPit, True);
        Call Subroutine(drawBoard);
        Set Global Variable(ctrl, Custom String("WASD: go to a pad.   R: move the pit marked THIS PIT MOVES to that pad."));
        Set Global Variable(coach, Custom String("The enemy moved. Stand on any pad and press R: the pit marked THIS PIT MOVES jumps there."));
        While(Not(Global.pitOk));
            Set Global Variable(gotR, False);
            Set Global Variable(wantR, True);
            Teleport(Global.learner, Value In Array(Global.padPos, 7));
            Wait(0.6, Ignore Condition);
            Set Global Variable(gotR, True);
            Wait Until(Compare(Global.gotR, ==, True), 60);
            Set Global Variable(wantR, False);
            Modify Global Variable(pitTries, Add, 1);
            Set Global Variable(pitTo, Add(Min(Max(Round To Integer(Divide(Add(X Component Of(Position Of(Global.learner)), 16), 4), Down), 0), 4), Multiply(5, Min(Max(Round To Integer(Divide(Add(Z Component Of(Position Of(Global.learner)), 16), 4), Down), 0), 4))));
            If(And(Compare(Global.pitTries, >=, 4), Or(Or(Compare(Global.pitTo, ==, 2), Compare(Global.pitTo, ==, Global.goal)), Array Contains(Global.pits, Global.pitTo))));
                Set Global Variable(pitTo, 7);
            End;
            If(Or(Or(Compare(Global.pitTo, ==, 2), Compare(Global.pitTo, ==, Global.goal)), Array Contains(Global.pits, Global.pitTo)));
                Set Global Variable(coach, Custom String("Pick a pad that is not the blue start, the goal or a pit, then press R."));
            Else;
                Set Global Variable At Index(pits, 1, Global.pitTo);
                Call Subroutine(settle);
                "Refuse a pad that would wall another pad in (a pad that could never reach the goal)."
                Set Global Variable(pitOk, True);
                For Global Variable(sp, 0, 25, 1);
                    If(And(And(Compare(Global.sp, !=, Global.goal), Not(Array Contains(Global.pits, Global.sp))), Compare(Value In Array(Global.V, Global.sp), <=, 0)));
                        Set Global Variable(pitOk, False);
                    End;
                End;
                If(Not(Global.pitOk));
                    Set Global Variable At Index(pits, 1, Global.pitFrom);
                    Set Global Variable(coach, Custom String("That would wall a pad in with no way to the goal. Pick another pad."));
                End;
            End;
        End;
        Set Global Variable(markPit, False);
        Set Global Variable(keepOld, True);
        Call Subroutine(makeArrows);
        "pit: the pad it left, the pad it moved to, old lines now wrong (answer key), presses of R"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,3,pit,{0},{1},{2}", Global.pitFrom, Global.pitTo, Custom String("{0},{1}", Global.wrongN, Global.pitTries)));
        Set Global Variable(V, Global.Vsave);
        Set Global Variable(arrow, Global.arrowOld);
        For Global Variable(sp, 0, 25, 1);
            Set Global Variable At Index(flip, Global.sp, False);
            Set Global Variable At Index(newPos, Global.sp, False);
        End;
        Call Subroutine(drawBoard);
        Set Global Variable(qId, 6);
        Set Global Variable(qText, Custom String("The pit moved. How many old lines are now wrong?"));
        Set Global Variable(qOpts, Array(Custom String("none"), Custom String("1 to 3"), Custom String("4 or more")));
        Set Global Variable(qCorrect, If-Then-Else(Compare(Global.wrongN, ==, 0), 0, If-Then-Else(Compare(Global.wrongN, <=, 3), 1, 2)));
        Set Global Variable(qAuto, Global.qCorrect);
        Set Global Variable(qWhy, Custom String("Press F and watch: lines that change turn yellow."));
        Call Subroutine(ask);
        "---- Round 5: plan again ----"
        Set Global Variable(ctrl, Custom String("F: wipe the board and plan again (the sweeps run by themselves)."));
        Set Global Variable(coach, Custom String("Press F: wipe the whiteboard and sweep the new map until nothing changes."));
        Call Subroutine(waitF);
        Set Global Variable(runId, 2);
        Call Subroutine(clearBoard);
        Set Global Variable(showArrows, False);
        Set Global Variable(looks0, Global.looks);
        While(And(Compare(Global.nChg, >, 0), Compare(Global.sweepN, <, 30)));
            Call Subroutine(sweepOnce);
            Call Subroutine(drawBoard);
            Call Subroutine(logBoard);
            Set Global Variable(coach, Custom String("Sweep {0}: {1} pads changed.", Global.sweepN, Global.nChg));
            Wait(0.3, Ignore Condition);
        End;
        Set Global Variable(keepOld, True);
        Call Subroutine(makeArrows);
        Set Global Variable(showArrows, True);
        Call Subroutine(drawBoard);
        Call Subroutine(logArrows);
        "replan: sweeps, old lines that flipped"
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,3,replan,{0},{1}", Global.sweepN, Global.wrongN));
        Set Global Variable(coach, Custom String("{0} old lines flipped (yellow). Only pads whose best route ran past the new pit change their minds.", Global.wrongN));
        Set Global Variable(ctrl, Custom String(""));
        Wait(0.4, Ignore Condition);
        "---- Debrief ----"
        Set Global Variable(deb, Array(Custom String("You did: planned the dive one sweep per F, walked the plan, then moved a pit."), Custom String("Called: dynamic programming, here value iteration: look one step ahead, repeat."), Custom String("In an LLM agent: it can plan tool calls only if it knows what each tool does.")));
        Set Global Variable(showDebrief, True);
        Set Global Variable(coach, Custom String("Predictions right: {0} of {1}. Simplified: moves never slip, and 25 pads is tiny next to a real game.", Global.predRight, Global.predTotal));
        Wait(0.4, Ignore Condition);
        Call Subroutine(finish);
    }
