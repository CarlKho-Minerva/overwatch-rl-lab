rule ("Initialize global variables") {
    event {
        Ongoing - Global;
    }
    actions {
        Set Global Variable(logQ, Empty Array);
        Set Global Variable(ready, False);
        Set Global Variable(gameOver, False);
        Set Global Variable(coach, Custom String(""));
        Set Global Variable(qActive, False);
        Set Global Variable(qText, Custom String(""));
        Set Global Variable(qOpts, Empty Array);
        Set Global Variable(qLine, Custom String(""));
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qRight, False);
        Set Global Variable(qWhy, Custom String(""));
        Set Global Variable(chkName, Custom String(""));
        Set Global Variable(pad, Vector(-6, 0, -15));
    }
}

rule ("Initialize player variables") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    actions {
        Set Player Variable(Event Player, fell, False);
    }
}

rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: every human player gets the game's hero") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Is Dummy Bot(Event Player) == False;
    }
    actions {
        Start Forcing Player To Be Hero(Event Player, Hero(Tracer));
    }
}

rule ("Core: hide the match clock, which sits under the question line") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Is Dummy Bot(Event Player) == False;
        Has Spawned(Event Player) == True;
    }
    actions {
        Disable Game Mode HUD(Event Player);
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        "The automatic check: a bot plays the learner's part. The host stands aside, out of the way and unseen."
        Create Dummy Bot(Hero(Tracer), Team 1, -1, Vector(-12, 0, -15), Left);
        Wait(1, Ignore Condition);
        Set Global Variable(learner, First Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
        Set Status(Host Player, Null, Phased Out, 99999);
        Set Invisible(Host Player, All);
        Teleport(Host Player, Vector(0, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Left, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,0,1,1"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(-12, 0, -15));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Left, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: the learner fell off the island") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Event Player == Global.learner;
        Global.ready != False;
        Or(Compare(Y Component Of(Position Of(Event Player)), <, -1.5), Is Dead(Event Player)) == True;
    }
    actions {
        Set Player Variable(Event Player, fell, True);
        Modify Player Variable(Event Player, fellCount, Add, 1);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,0,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,0,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        Wait(1.2, Ignore Condition);
        Set Global Variable(qAnswer, Global.qAuto);
        Set Global Variable(qActive, False);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,0,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(0.6, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
    }
}

rule ("Core: end the game and write the closing line") {
    event {
        Subroutine;
        finish;
    }
    actions {
        Set Global Variable(gameOver, True);
        "Server load as the game itself reports it (a percentage; near 100 the game may shut the mode down): seconds\nsampled, the mean over the whole game, the highest peak. Written by both builds."
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,L,0,{0},{1},{2}", Global.loadSamples, Round To Integer(Divide(Global.loadSumRun, Max(Global.loadSamples, 1)), To Nearest), Round To Integer(Global.loadPeakRun, To Nearest)));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,X,0,{0},{1},{2}", Total Time Elapsed, Global.predRight, Custom String("{0},{1},{2}", Global.predTotal, Global.selfOk, Global.selfBad)));
    }
}

rule ("Core: the screen") {
    event {
        Ongoing - Global;
    }
    actions {
        Create HUD Text(All Players(All Teams), Custom String("RL 0: CORE CHECK"), Null, Null, Left, 0, Color(Orange), Color(White), Color(White), Visible To and String, Default Visibility);
        Create HUD Text(All Players(All Teams), Null, Global.coach, Null, Left, 1, Color(White), Color(White), Color(White), Visible To and String, Default Visibility);
        Create HUD Text(All Players(All Teams), Null, Null, Custom String("predictions right: {0} of {1}", Global.predRight, Global.predTotal), Left, 2, Color(White), Color(White), Color(Sky Blue), Visible To and String, Default Visibility);
        "it. At the top of the screen it ran into whatever a game had put on the left and right (seen in the game,\n2026-11-22: three placements tried). In the world it is clear of every HUD line."
        Create In-World Text(If-Then-Else(Global.qActive, All Players(All Teams), Null), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Create In-World Text(If-Then-Else(Global.qActive, All Players(All Teams), Null), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
    }
}

rule ("Game 0: run") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.ready != False;
    }
    actions {
        Create Effect(All Players(All Teams), Ring, Color(Green), Global.pad, 1.5, None);
        Set Global Variable(chkName, Custom String("two plus two"));
        Set Global Variable(chkExp, 4);
        Set Global Variable(chkGot, 4);
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("bayes two readings"));
        Set Global Variable(chkExp, 0.941);
        Set Global Variable(chkGot, 0.941176470588235);
        Call Subroutine(check);
        Set Global Variable(coach, Custom String("Walk onto the green ring."));
        Start Throttle In Direction(Global.learner, Left, 1, To World, Replace existing throttle, None);
        Wait Until(Compare(Distance Between(Position Of(Global.learner), Global.pad), <, 1.5), 20);
        Stop Throttle In Direction(Global.learner);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,0,pad,{0},{1}", Compare(Distance Between(Position Of(Global.learner), Global.pad), <, 1.5), Total Time Elapsed));
        Set Global Variable(qId, 1);
        Set Global Variable(qText, Custom String("A sensor that is right 8 times in 10 says WALL twice. How sure should she be?"));
        Set Global Variable(qOpts, Array(Custom String("about 80%"), Custom String("about 94%"), Custom String("certain")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("0.8 x 0.8 against 0.2 x 0.2: about 94%."));
        Call Subroutine(ask);
        Call Subroutine(putHome);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,0,home,{0},{1}", Compare(Distance Between(Position Of(Global.learner), Vector(-12, 0, -15)), <, 1.5), (Global.learner).fellCount));
        Call Subroutine(finish);
        Set Global Variable(coach, Custom String("Done. Core check finished."));
    }
}
