rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        "The automatic check: a bot plays the learner's part. The host stands aside, out of the way and unseen."
        Create Dummy Bot(Hero(Soldier: 76), Team 1, -1, Vector(0, 0, -8), Left);
        Wait(1, Ignore Condition);
        Set Global Variable(learner, First Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
        Set Status(Host Player, Null, Phased Out, 99999);
        Set Invisible(Host Player, All);
        Teleport(Host Player, Vector(0, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Forward, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,7,1,1"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(0, 0, -8));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Forward, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: the learner fell off the island") {
    event {

    }
    conditions {
        Event Player == Global.learner;

    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,7,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,7,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        "The moment the question is on screen, so the log shows how long the prediction took (and so a script that plays\nthe human build knows when to press, as a person does by looking)."
        Set Global Variable(qShown, Total Time Elapsed);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,Q,7,{0}", Global.qId));
        Wait(1.2, Ignore Condition);
        Set Global Variable(qAnswer, Global.qAuto);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,7,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1},{2}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0), Subtract(Total Time Elapsed, Global.qShown))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(0.6, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

rule ("Game 7: the bot's odds from its preferences (softmax)") {
    event {
        Subroutine;
        softmax;
    }
    actions {
        Set Global Variable(smMax, Max(First Of(Global.botH), Max(Value In Array(Global.botH, 1), Value In Array(Global.botH, 2))));
        Set Global Variable(smE, Mapped Array(Global.botH, Raise To Power(2.718281828, Subtract(Current Array Element, Global.smMax))));
        Set Global Variable(smS, Add(Add(First Of(Global.smE), Value In Array(Global.smE, 1)), Value In Array(Global.smE, 2)));
        Set Global Variable(botP, Mapped Array(Global.smE, Divide(Current Array Element, Global.smS)));
    }
}

rule ("Game 7: REINFORCE with a baseline: push the pick by (reward minus usual) times its surprise") {
    event {
        Subroutine;
        learn;
    }
    actions {
        "botP holds the odds the pick was drawn from."
        Set Global Variable(adv, Subtract(Global.bR, Global.base));
        Set Global Variable(dPick, Multiply(Multiply(Global.alpha, Global.adv), Subtract(1, Value In Array(Global.botP, Global.bPick))));
        For Global Variable(li, 0, 3, 1);
            Modify Global Variable At Index(botH, Global.li, Add, Multiply(Multiply(Global.alpha, Global.adv), Subtract(If-Then-Else(Compare(Global.li, ==, Global.bPick), 1, 0), Value In Array(Global.botP, Global.li))));
        End;
        Modify Global Variable(base, Add, Multiply(0.25, Subtract(Global.bR, Global.base)));
        Call Subroutine(softmax);
    }
}

        Set Global Variable(barTxt, Array(Custom String("bot picks TRACER {0}%", Round To Integer(Multiply(First Of(Global.botP), 100), To Nearest)), Custom String("bot picks WIDOWMAKER {0}%", Round To Integer(Multiply(Value In Array(Global.botP, 1), 100), To Nearest)), Custom String("bot picks PHARAH {0}%", Round To Integer(Multiply(Value In Array(Global.botP, 2), 100), To Nearest))));
        Set Global Variable(oddsHead, Custom String("nudge size {0}   its usual reward (baseline) {1}", Global.alpha, Global.base));
        Set Global Variable(exF, Global.botP);
        Set Global Variable(exBot, Max(Subtract(Value In Array(Global.exF, 1), Value In Array(Global.exF, 2)), Max(Subtract(Value In Array(Global.exF, 2), First Of(Global.exF)), Subtract(First Of(Global.exF), Value In Array(Global.exF, 1)))));

    }

rule ("Game 7: wait for the player's pick (left click, right click, melee), outside of questions") {
    event {
        Subroutine;
        getPick;
    }
    actions {
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        Set Global Variable(youPick, -1);
        "A button still held from the last answer must not count as a pick."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 5);
        Wait(0.25, Ignore Condition);
        Set Global Variable(pickOpen, True);
        Wait(0.4, Ignore Condition);
        Set Global Variable(youPick, Value In Array(Global.autoPicks, Global.rnd));
        Set Global Variable(pickOpen, False);
        If(Compare(Global.youPick, <, 0));
            "2026-11-22: in the same-hero rounds a random pick had a 2 in 3 chance of not matching, and the round then\nasked again, two minutes at a time. A timed-out pick there is the hero already chosen."
            Set Global Variable(youPick, If-Then-Else(And(Compare(Global.phase, ==, 0), Compare(Global.habit, >=, 0)), Global.habit, Random Integer(0, 2)));
            Modify Global Variable(timedOut, Add, 1);
            Set Global Variable(coach, Custom String("No pick in two minutes, so one was made for you."));
        End;
    }
}

rule ("Game 7: one round: the bot locks in from its odds, the player picks, reveal, the bot learns") {
    event {
        Subroutine;
        playRound;
    }
    actions {
        Call Subroutine(softmax);
        Set Global Variable(pBefore, Global.botP);
        "The bot draws first, from its odds only (it never sees his pick). Logged so the test can redo it."
        Set Global Variable(drawK, Random Integer(0, 999));
        Set Global Variable(drawU, Divide(Add(Global.drawK, 0.5), 1000));
        Set Global Variable(bPick, If-Then-Else(Compare(Global.drawU, <, First Of(Global.botP)), 0, If-Then-Else(Compare(Global.drawU, <, Add(First Of(Global.botP), Value In Array(Global.botP, 1))), 1, 2)));
        Set Global Variable(botFx, Vector(0, -60, 0));
        Set Global Variable(youFx, Vector(0, -60, 0));
        Set Global Variable(coach, Custom String("Round {0}: the bot has locked in its pick, hidden. Make yours.", Add(Global.rnd, 1)));
        Call Subroutine(getPick);
        If(Compare(Global.phase, ==, 0));
            If(Compare(Global.habit, <, 0));
                Set Global Variable(habit, Global.youPick);
            End;
            While(Compare(Global.youPick, !=, Global.habit));
                Set Global Variable(coach, Custom String("Same hero for these rounds: keep picking {0}.", Value In Array(Global.heroName, Global.habit)));
                Call Subroutine(getPick);
            End;
        End;
        Set Global Variable(youFx, Value In Array(Global.youPos, Global.youPick));
        Set Global Variable(botFx, Value In Array(Global.botPos, Global.bPick));
        Set Global Variable(bR, If-Then-Else(Compare(Global.youPick, ==, Modulo(Add(Global.bPick, 1), 3)), 1, If-Then-Else(Compare(Global.bPick, ==, Modulo(Add(Global.youPick, 1), 3)), -1, 0)));
        If(Compare(Global.bR, >, 0));
            Modify Global Variable(winsBot, Add, 1);
        Else If(Compare(Global.bR, <, 0));
            Modify Global Variable(winsYou, Add, 1);
        Else;
            Modify Global Variable(draws, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,7,pick,{0},{1},{2}", Add(Global.rnd, 1), Global.phase, Custom String("{0},{1},{2}", Global.youPick, Global.drawK, Custom String("{0},{1}", Global.bPick, Global.bR))));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,7,odds,{0},{1},{2}", Add(Global.rnd, 1), First Of(Global.pBefore), Custom String("{0},{1},{2}", Value In Array(Global.pBefore, 1), Value In Array(Global.pBefore, 2), Custom String("{0},{1}", Global.alpha, Global.base))));
        Set Global Variable At Index(info, 1, Custom String("Round {0}: you {1}, bot {2}", Add(Global.rnd, 1), Value In Array(Global.heroName, Global.youPick), Custom String("{0}: {1}. You {2}", Value In Array(Global.heroName, Global.bPick), If-Then-Else(Compare(Global.bR, >, 0), Custom String("Bot wins"), If-Then-Else(Compare(Global.bR, <, 0), Custom String("You win"), Custom String("Draw"))), Custom String("{0}, bot {1}, draws {2}.", Global.winsYou, Global.winsBot, Global.draws))));
        Set Global Variable(coach, Custom String("{0} had {1}%, scored {2}", Value In Array(Global.heroName, Global.bPick), Round To Integer(Multiply(Value In Array(Global.pBefore, Global.bPick), 100), To Nearest), Custom String("{0} vs its usual {1}: preference moves {2}", Global.bR, Global.base, Custom String("{0} x {1} x {2}", Global.alpha, Subtract(Global.bR, Global.base), Custom String("{0} = {1}.", Subtract(1, Value In Array(Global.pBefore, Global.bPick)), Multiply(Multiply(Global.alpha, Subtract(Global.bR, Global.base)), Subtract(1, Value In Array(Global.pBefore, Global.bPick))))))));
        Call Subroutine(learn);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,7,pref,{0},{1},{2}", Add(Global.rnd, 1), First Of(Global.botH), Custom String("{0},{1},{2}", Value In Array(Global.botH, 1), Value In Array(Global.botH, 2), Global.base)));
        If(Compare(Global.phase, >, 0));
            Modify Global Variable At Index(cntB, Global.youPick, Add, 1);
            Modify Global Variable(nB, Add, 1);
            Set Global Variable(youF, Array(Divide(First Of(Global.cntB), Global.nB), Divide(Value In Array(Global.cntB, 1), Global.nB), Divide(Value In Array(Global.cntB, 2), Global.nB)));
            Set Global Variable(exF, Global.youF);
