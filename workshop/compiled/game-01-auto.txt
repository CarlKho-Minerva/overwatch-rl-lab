rule ("Core: write the queued log lines in one frame, once a second") {
    event {
        Ongoing - Global;
    }
    actions {
        Wait(1, Ignore Condition);
        If(Compare(Count Of(Global.logQ), >, 0));
            Set Global Variable(flushN, Min(Count Of(Global.logQ), 120));
            Enable Inspector Recording;
            For Global Variable(flushI, 0, Global.flushN, 1);
                Log To Inspector(Value In Array(Global.logQ, Global.flushI));
            End;
            Disable Inspector Recording;
            Set Global Variable(logQ, Array Slice(Global.logQ, Global.flushN, 1000));
        End;
        "2026-11-22: the game's own load readings only look back two seconds, so they are read here once a second and\nkept for the whole game (the closing reading alone showed the quiet debrief, not the busy middle)."
        If(And(Global.ready, Not(Global.gameOver)));
            Modify Global Variable(loadSamples, Add, 1);
            Modify Global Variable(loadSumRun, Add, Server Load Average);
            Modify Global Variable(loadPeakRun, Max, Server Load Peak);
        End;
        Loop;
    }
}

rule ("Core: set up the learner") {
    event {
        Ongoing - Global;
    }
    actions {
        Disable Inspector Recording;
        Pause Match Time;
        Wait(1.5, Ignore Condition);
        Wait Until(And(Has Spawned(Host Player), Is Alive(Host Player)), 60);
        "The automatic check: a bot plays the learner's part. The host stands aside, out of the way and unseen."
        Create Dummy Bot(Hero(Tracer), Team 1, -1, Vector(0, 0, -3), Left);
        Wait(1, Ignore Condition);
        Set Global Variable(learner, First Of(Filtered Array(All Players(Team 1), Is Dummy Bot(Current Array Element))));
        Set Status(Host Player, Null, Phased Out, 99999);
        Set Invisible(Host Player, All);
        Teleport(Host Player, Vector(0, 0, 0));
        Wait(0.5, Ignore Condition);
        "2026-11-22: five games left a person looking wherever he spawned, sometimes away from the board."
        Set Facing(Global.learner, Forward, To World);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,M,1,1,1"));
        Set Global Variable(ready, True);
    }
}

rule ("Core: put the learner back at the start (after a fall, or between rounds)") {
    event {
        Subroutine;
        putHome;
    }
    actions {
        If(Is Dead(Global.learner));
            Resurrect(Global.learner);
            Wait(0.25, Ignore Condition);
            If(Is Dead(Global.learner));
                Respawn(Global.learner);
                Wait Until(Is Alive(Global.learner), 4);
            End;
        End;
        Teleport(Global.learner, Vector(0, 0, -3));
        Wait(0.2, Ignore Condition);
        Set Facing(Global.learner, Forward, To World);
        Set Player Variable(Global.learner, fell, False);
    }
}

rule ("Core: the learner fell off the island") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Event Player == Global.learner;
        Global.ready != False;
        Or(Compare(Y Component Of(Position Of(Event Player)), <, -1.5), Is Dead(Event Player)) == True;

    actions {
        Set Player Variable(Event Player, fell, True);
        Modify Player Variable(Event Player, fellCount, Add, 1);
    }
}

rule ("Core: one self-test line") {
    event {
        Subroutine;
        check;
    }
    actions {
        If(Compare(Absolute Value(Subtract(Global.chkExp, Global.chkGot)), <, 0.011));
            Modify Global Variable(selfOk, Add, 1);
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,1,{0},{1},{2},1", Global.chkName, Global.chkExp, Global.chkGot));
        Else;
            Modify Global Variable(selfBad, Add, 1);
            Set Global Variable(coach, Custom String("SELF-TEST FAILED: {0} (expected {1}, got {2})", Global.chkName, Global.chkExp, Global.chkGot));
            Modify Global Variable(logQ, Append To Array, Custom String("RLG,T,1,{0},{1},{2},0", Global.chkName, Global.chkExp, Global.chkGot));
        End;
    }
}

rule ("Core: ask the learner to predict, wait for the answer, say why, log it") {
    event {
        Subroutine;
        ask;
    }
    actions {
        Set Global Variable(qAnswer, -1);
        Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1)));
        If(Compare(Count Of(Global.qOpts), >, 2));
            Set Global Variable(qLine, Custom String("LEFT CLICK: {0}      RIGHT CLICK: {1}      MELEE: {2}", First Of(Global.qOpts), Value In Array(Global.qOpts, 1), Value In Array(Global.qOpts, 2)));
        End;
        Disallow Button(Global.learner, Button(Primary Fire));
        Disallow Button(Global.learner, Button(Secondary Fire));
        Disallow Button(Global.learner, Button(Melee));
        "switched off while a question is up. This is the ability switch, not the button rule, so a game that keeps\nBlink off for good with disallowButton(Button.ABILITY_1) is not undone when the question ends."
        Set Ability 1 Enabled(Global.learner, False);
        "2026-11-22: a fire button still held when a question opened answered it at once (the answer rules fire the moment\ntheir conditions are true). Wait until all three are let go, for at most 8 seconds, before showing the question."
        Wait Until(Not(Or(Or(Is Button Held(Global.learner, Button(Primary Fire)), Is Button Held(Global.learner, Button(Secondary Fire))), Is Button Held(Global.learner, Button(Melee)))), 8);
        Wait(0.4, Ignore Condition);
        Set Global Variable(qActive, True);
        "It is made for each question and removed after it: made once at the start, it was wiped for good by any game\nthat redrew its board with destroyAllInWorldText (games 2, 3 and 6 did), and a person then saw no question."
        Create In-World Text(All Players(All Teams), Global.qText, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Up), 1.9, Do Not Clip, Visible To Position and String, Color(Yellow), Default Visibility);
        Set Global Variable(qTxtTop, Last Text ID);
        Create In-World Text(All Players(All Teams), Global.qLine, Add(Add(Eye Position(Global.learner), Multiply(Facing Direction Of(Global.learner), 5)), Vector(0, 0.45, 0)), 1.5, Do Not Clip, Visible To Position and String, Color(White), Default Visibility);
        Set Global Variable(qTxtLow, Last Text ID);
        "The moment the question is on screen, so the log shows how long the prediction took (and so a script that plays\nthe human build knows when to press, as a person does by looking)."
        Set Global Variable(qShown, Total Time Elapsed);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,Q,1,{0}", Global.qId));
        Wait(1.2, Ignore Condition);
        Set Global Variable(qAnswer, Global.qAuto);
        Set Global Variable(qActive, False);
        Destroy In-World Text(Global.qTxtTop);
        Destroy In-World Text(Global.qTxtLow);
        Set Global Variable(qRight, Compare(Global.qAnswer, ==, Global.qCorrect));
        Modify Global Variable(predTotal, Add, 1);
        If(Global.qRight);
            Modify Global Variable(predRight, Add, 1);
        End;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,A,1,{0},{1},{2}", Global.qId, Global.qAnswer, Custom String("{0},{1},{2}", Global.qCorrect, If-Then-Else(Global.qRight, 1, 0), Subtract(Total Time Elapsed, Global.qShown))));
        Set Global Variable(coach, Custom String("{0} {1}", If-Then-Else(Global.qRight, Custom String("RIGHT."), If-Then-Else(Compare(Global.qAnswer, <, 0), Custom String("NO ANSWER."), Custom String("NOT THIS TIME."))), Global.qWhy));
        Wait(0.6, Ignore Condition);
        Allow Button(Global.learner, Button(Primary Fire));
        Allow Button(Global.learner, Button(Secondary Fire));
        Allow Button(Global.learner, Button(Melee));
        Set Ability 1 Enabled(Global.learner, True);
    }
}

rule ("Game 1: the stand-in walks to a ring (automatic check only)") {
    event {
        Subroutine;
        autoGo;
    }
    actions {
        Start Throttle In Direction(Global.learner, Direction Towards(Position Of(Global.learner), Global.autoTgt), 1, To World, Replace existing throttle, Direction and Magnitude);
        Wait Until(Or(Compare(Distance Between(Position Of(Global.learner), Global.autoTgt), <, 0.8), Not(Global.roundOn)), 8);
        Stop Throttle In Direction(Global.learner);
        If(And(Global.roundOn, Compare(Distance Between(Position Of(Global.learner), Global.autoTgt), >, 1.200000000000000)));
            Teleport(Global.learner, Global.autoTgt);
        End;
        Wait(0.5, Ignore Condition);
    }
}

rule ("Game 1: one round, from the start until the +50 ring or the time limit") {
    event {
        Subroutine;
        playRound;
    }
    actions {
        Call Subroutine(putHome);
        Set Facing(Global.learner, Forward, To World);
        Set Global Variable(touched, Array(False, False, False, False, False));
        Set Global Variable(found, 0);
        Set Global Variable(trapHit, 0);
        Set Global Variable(secs, 0);
        Set Global Variable(total, 0);
        Set Global Variable(last, 0);
        Set Global Variable(lastSrc, Custom String("round {0} starts", Global.rnd));
        Call Subroutine(refreshScore);
        Set Global Variable(coach, Custom String("Round {0}. Go.", Global.rnd));
        Set Global Variable(roundOn, True);
        For Global Variable(ki, 0, Count Of(Global.autoRoute), 1);
            If(Global.roundOn);
                Set Global Variable(autoTgt, Value In Array(Global.ringPos, Value In Array(Global.autoRoute, Global.ki)));
                Call Subroutine(autoGo);
            End;
        End;
        Wait Until(Not(Global.roundOn), 20);
        If(Global.roundOn);
            Set Global Variable(roundOn, False);
            Set Global Variable(coach, Custom String("Time is up for round {0}.", Global.rnd));
        End;
        Stop Throttle In Direction(Global.learner);
        Wait(0.3, Ignore Condition);
    }
}

rule ("Game 1: the late bill, the round's total, its self-test and its log line") {
    event {
        Subroutine;
        settleRound;
    }
    actions {
        If(Global.trapHit);
            Modify Global Variable(total, Subtract, 20);
            Set Global Variable(last, -20);
            Set Global Variable(lastSrc, Custom String("a late bill"));
            Call Subroutine(refreshScore);
            Small Message(Global.learner, Custom String("-20, a late bill for a ring you touched"));
        End;
        Set Global Variable(chkName, Custom String("round {0} total adds up", Global.rnd));
        Set Global Variable(chkExp, Subtract(Subtract(Multiply(50, Global.found), Global.secs), Multiply(20, Global.trapHit)));
        Set Global Variable(chkGot, Global.total);
        Call Subroutine(check);
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,1,round,{0},{1},{2}", Global.rnd, Global.found, Custom String("{0},{1},{2}", Global.secs, Global.trapHit, Global.total)));
        Set Global Variable At Index(totals, Subtract(Global.rnd, 1), Global.total);
        Set Global Variable(hudRounds, Custom String("round totals   1: {0}   2: {1}   3: {2}", First Of(Global.totals), Value In Array(Global.totals, 1), Custom String("{0}   4: {1}", Value In Array(Global.totals, 2), Value In Array(Global.totals, 3))));
    }
}

rule ("Game 1: every second of a round costs 1") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.roundOn != False;
    }
    actions {
        While(Global.roundOn);
            Wait(1, Ignore Condition);
            If(Global.roundOn);
                Modify Global Variable(secs, Add, 1);
                Modify Global Variable(total, Subtract, 1);
                Set Global Variable(last, -1);
                Set Global Variable(lastSrc, Custom String("a second passed"));
                Call Subroutine(refreshScore);
            End;
        End;
    }
}

rule ("Game 1: touching a ring") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.roundOn != False;
    }
    actions {
        While(Global.roundOn);
            For Global Variable(ri, 0, 5, 1);
                If(And(And(Global.roundOn, Not(Value In Array(Global.touched, Global.ri))), Compare(Distance Between(Position Of(Global.learner), Value In Array(Global.ringPos, Global.ri)), <, 1.6)));
                    Set Global Variable At Index(touched, Global.ri, True);
                    Set Global Variable(lastSrc, Custom String("a ring"));
                    If(Compare(Global.ri, ==, Global.goalRing));
                        Modify Global Variable(total, Add, 50);
                        Set Global Variable(last, 50);
                        Set Global Variable(found, 1);
                        Set Global Variable(roundOn, False);
                        Small Message(Global.learner, Custom String("+50"));
                    Else;
                        If(Compare(Global.ri, ==, Global.trapRing));
                            Set Global Variable(trapHit, 1);
                        End;
                        Set Global Variable(last, 0);
                        Small Message(Global.learner, Custom String("0"));
                    End;
                    Call Subroutine(refreshScore);
                    Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,1,touch,{0},{1},{2}", Global.rnd, Global.ri, Custom String("{0},{1}", Global.last, Global.secs)));
                End;
            End;
            Wait(0.1, Ignore Condition);
        End;
    }
}

rule ("Game 1: run") {
    event {
        Ongoing - Global;
    }
    conditions {
        Global.ready != False;
    }
    actions {
        "The five rings sit on a circle of 13 m around the start, so no ring is closer than another."
        For Global Variable(ki, 0, 5, 1);
            Modify Global Variable(ringPos, Append To Array, Add(Vector(0, 0, -3), Vector(Multiply(13, Cosine From Degrees(Add(90, Multiply(72, Global.ki)))), 0, Multiply(13, Sine From Degrees(Add(90, Multiply(72, Global.ki)))))));
        End;
        Set Global Variable(chkName, Custom String("ring 0 is 13 m from the start"));
        Set Global Variable(chkExp, 13);
        Set Global Variable(chkGot, Distance Between(First Of(Global.ringPos), Vector(0, 0, -3)));
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("ring 3 is 13 m from the start"));
        Set Global Variable(chkExp, 13);
        Set Global Variable(chkGot, Distance Between(Value In Array(Global.ringPos, 3), Vector(0, 0, -3)));
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("quick round: +50 in 3 s, no bill"));
        Set Global Variable(chkExp, 47);
        Set Global Variable(chkGot, 47);
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("slow round: +50 in 12 s, with the bill"));
        Set Global Variable(chkExp, 18);
        Set Global Variable(chkGot, 18);
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("no +50 in 120 s, no bill"));
        Set Global Variable(chkExp, -120);
        Set Global Variable(chkGot, -120);
        Call Subroutine(check);
        Set Global Variable(chkName, Custom String("gap between those two rounds"));
        Set Global Variable(chkExp, 29);
        Set Global Variable(chkGot, 29);
        Call Subroutine(check);
        Call Subroutine(drawRings);
        "The hidden layout: drawn first, logged, then used."
        Set Global Variable(goalRing, Random Integer(0, 4));
        Set Global Variable(trapRing, Random Value In Array(Remove From Array(Array(0, 1, 2, 3, 4), Global.goalRing)));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,1,layout,1,{0},{1}", Global.goalRing, Global.trapRing));
        Set Global Variable(coach, Custom String("Like your first minutes on a new map: no callouts, no coach, only the scoreboard."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(coach, Custom String("Five rings, all alike. The yellow number is all you get."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 1);
        Set Global Variable(qText, Custom String("The round starts after you answer. If you stand still, what will the number do?"));
        Set Global Variable(qOpts, Array(Custom String("stay at 0"), Custom String("go down"), Custom String("go up")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("It drops 1 every second. The score says hurry, but never where to go."));
        Call Subroutine(ask);
        "Round 1: find out what the score rewards."
        Set Global Variable(rnd, 1);
        Set Global Variable(autoRoute, Array(Global.trapRing, First Of(Filtered Array(Array(0, 1, 2, 3, 4), And(Compare(Current Array Element, !=, Global.goalRing), Compare(Current Array Element, !=, Global.trapRing)))), Global.goalRing));
        Call Subroutine(playRound);
        Set Global Variable(r1Found, Global.found);
        Set Global Variable(coach, Custom String("Round 1 is over."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 2);
        Set Global Variable(qText, Custom String("The round is over. Will your total change before round 2 starts?"));
        Set Global Variable(qOpts, Array(Custom String("no, it is final"), Custom String("yes, it can still change")));
        Set Global Variable(qCorrect, If-Then-Else(Global.trapHit, 1, 0));
        Set Global Variable(qAuto, 0);
        If(Global.trapHit);
            Set Global Variable(qWhy, Custom String("One ring you touched sends a -20 bill, late. Which one? The score does not say."));
        Else;
            Set Global Variable(qWhy, Custom String("No bill this time. One ring you have not touched sends a -20 bill when the round ends."));
        End;
        Call Subroutine(ask);
        Call Subroutine(settleRound);
        Set Global Variable(r1Total, Global.total);
        Wait(0.4, Ignore Condition);
        Set Global Variable(coach, Custom String("You acted, then read the number, then acted again: action, observation, reward, in a loop."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(coach, Custom String("No one showed you the right move. Learning from that number alone is reinforcement learning."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 3);
        If(Global.r1Found);
            Set Global Variable(qText, Custom String("Round 2 has the same hidden layout. Go straight to the ring that paid: your total will be..."));
        Else;
            Set Global Variable(qText, Custom String("Round 2, same hidden layout. Go straight to a ring you have not tried. If it pays, your total is..."));
        End;
        Set Global Variable(qOpts, Array(Custom String("exactly +50"), Custom String("a little under +50"), Custom String("about round 1's")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("+50 minus the seconds it takes. Same map, fewer tries: you should beat round 1."));
        Call Subroutine(ask);
        "Round 2: same layout, use what you know."
        Set Global Variable(rnd, 2);
        Set Global Variable(autoRoute, Array(Global.goalRing));
        Call Subroutine(playRound);
        Call Subroutine(settleRound);
        Set Global Variable(r2Total, Global.total);
        Set Global Variable(hudGap, Custom String("round 2 minus round 1: {0}, what the score taught", Subtract(Global.r2Total, Global.r1Total)));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,1,gap,{0},{1},{2}", Global.r1Total, Global.r2Total, Subtract(Global.r2Total, Global.r1Total)));
        If(Compare(Global.r2Total, >, Global.r1Total));
            Set Global Variable(coach, Custom String("Round 2 beat round 1 by {0}. Same map, more score: that is learning from reward.", Subtract(Global.r2Total, Global.r1Total)));
        Else;
            Set Global Variable(coach, Custom String("Round 2 did not beat round 1 this time. The score only teaches if you use what it showed."));
        End;
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 4);
        Set Global Variable(qText, Custom String("Round 3: if you go straight to the ring that paid, what will you learn about the other rings?"));
        Set Global Variable(qOpts, Array(Custom String("nothing new"), Custom String("which one is best")));
        Set Global Variable(qCorrect, 0);
        Set Global Variable(qAuto, 0);
        Set Global Variable(qWhy, Custom String("Using what you know earns now but tests nothing. Trying things costs seconds but finds more."));
        Call Subroutine(ask);
        "Round 3: the rings vanish before the clock starts, so he acts on what he remembers."
        Set Global Variable(rnd, 3);
        Call Subroutine(putHome);
        Set Facing(Global.learner, Forward, To World);
        Set Status(Global.learner, Null, Rooted, 9999);
        Disallow Button(Global.learner, Button(Ability 1));
        Set Global Variable(coach, Custom String("Round 3. Look around now: the rings vanish in 2 seconds."));
        Wait(2, Ignore Condition);
        Destroy All Effects;
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,1,vanish,{0}", Global.rnd));
        Set Global Variable(qId, 5);
        Set Global Variable(qText, Custom String("The rings are gone. What are you steering by now?"));
        Set Global Variable(qOpts, Array(Custom String("what the screen shows"), Custom String("what I remember"), Custom String("the score")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("Memory. History is every second, ring and bill so far. What you kept of it is your state."));
        Call Subroutine(ask);
        Clear Status(Global.learner, Rooted);
        Allow Button(Global.learner, Button(Ability 1));
        Set Global Variable(autoRoute, Array(Global.goalRing));
        Call Subroutine(playRound);
        Call Subroutine(settleRound);
        Set Global Variable(coach, Custom String("Round 3: {0}, with nothing on screen to steer by. Your state came from your history.", Global.total));
        Wait(0.4, Ignore Condition);
        "Round 4: the island looks the same, but the hidden payouts are redrawn and the +50 ring moves."
        Call Subroutine(drawRings);
        Set Global Variable(oldGoal, Global.goalRing);
        Set Global Variable(goalRing, Random Value In Array(Remove From Array(Array(0, 1, 2, 3, 4), Global.oldGoal)));
        Set Global Variable(trapRing, Random Value In Array(Remove From Array(Array(0, 1, 2, 3, 4), Global.goalRing)));
        Modify Global Variable(logQ, Append To Array, Custom String("RLG,E,1,layout,4,{0},{1}", Global.goalRing, Global.trapRing));
        Set Global Variable(coach, Custom String("Round 4. The rings are back."));
        Wait(0.4, Ignore Condition);
        Set Global Variable(qId, 6);
        Set Global Variable(qText, Custom String("Round 4 looks the same, but something hidden may have changed. Will your old ring pay +50?"));
        Set Global Variable(qOpts, Array(Custom String("yes, the island looks the same"), Custom String("I cannot tell from what I see")));
        Set Global Variable(qCorrect, 1);
        Set Global Variable(qAuto, 1);
        Set Global Variable(qWhy, Custom String("The payouts were redrawn. Same view, different hidden state: what you see is not everything."));
        Call Subroutine(ask);
        Set Global Variable(rnd, 4);
        Set Global Variable(autoRoute, Array(Global.oldGoal, Global.goalRing));
        Call Subroutine(playRound);
        Call Subroutine(settleRound);
        Set Global Variable(coach, Custom String("Round 4: {0}. It looked like round 2, but it was not: that is partial observability.", Global.total));
        Wait(0.4, Ignore Condition);
        "Debrief."
        Set Global Variable At Index(db, 0, Custom String("What you did: found the paying ring from a number alone, then ran to it blind."));
        Set Global Variable At Index(db, 1, Custom String("What it is called: learning from reward, with late feedback. Your memory was your state."));
        Set Global Variable At Index(db, 2, Custom String("In a language-model agent: a task scored only at the end; the transcript is its history."));
        Set Global Variable At Index(db, 3, Custom String("Your predictions: {0} of {1} right.", Global.predRight, Global.predTotal));
        Set Global Variable(debrief, True);
        Set Global Variable(coach, Custom String("Simplified: you always knew where you stood, and the rules held still. Real agents see less."));
        Wait(0.4, Ignore Condition);
        Call Subroutine(finish);
        Set Global Variable(coach, Custom String("Done. Game 1 finished."));
    }
